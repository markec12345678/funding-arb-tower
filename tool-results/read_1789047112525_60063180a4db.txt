    84→    84→def _api_call(
    85→    85→    method: str, path: str, params: Optional[dict] = None, signed: bool = False
    86→    86→) -> Any:
    87→    87→    base_params = dict(params or {})
    88→    88→    if signed:
    89→    89→        # Credential validation: fail immediately on missing credentials; never send private requests with empty credentials
    90→    90→        key, secret = _get_key(), _get_secret()
    91→    91→        if not (key and secret):
    92→    92→            raise RuntimeError(
    93→    93→                "Binance API credentials missing: please set BINANCE_API_KEY / BINANCE_API_SECRET, "
    94→    94→                "or configure them in ~/.funding-arb/credentials.json under env."
    95→    95→            )
    96→    96→    # GET can be safely retried; POST (orders) are not retried to avoid duplicate orders
    97→    97→    retries = 3 if method == "GET" else 1
    98→    98→    last_err: Optional[Exception] = None
    99→    99→    for attempt in range(retries):
   100→   100→        if signed:
   101→   101→            params2 = dict(base_params)
   102→   102→            params2["timestamp"] = int(time.time() * 1000)  # Refresh timestamp on retry
   103→   103→            query = urllib.parse.urlencode(params2)
   104→   104→            sig = _sign(query)
   105→   105→            base_url = "https://fapi.binance.com" if path.startswith("/fapi/") else BASE
   106→   106→            url = f"{base_url}{path}?{query}&signature={sig}"
   107→   107→            headers = {"X-MBX-APIKEY": _get_key()}
   108→   108→        else:
   109→   109→            query = urllib.parse.urlencode(base_params) if base_params else ""
   110→   110→            base_url = "https://fapi.binance.com" if path.startswith("/fapi/") else BASE
   111→   111→            url = f"{base_url}{path}" + (f"?{query}" if query else "")
   112→   112→            headers = {}
   113→   113→        req = urllib.request.Request(url, headers=headers, method=method)
   114→   114→        try:
   115→   115→            with urllib.request.urlopen(req, timeout=15) as resp:
   116→   116→                return json.loads(resp.read().decode())
   117→   117→        except Exception as e:
   118→   118→            last_err = e
   119→   119→            if method == "GET" and attempt < retries - 1:
   120→   120→                time.sleep(0.5 * (attempt + 1))
   121→   121→                continue
   122→   122→            raise
   123→   123→    raise last_err if last_err else RuntimeError("binance _api_call failed")

   198→   198→    def _ensure_fapi_exchange_info(self, cache_sec: int = 3600) -> None:
   199→   199→        global _fapi_exchange_info_loaded_at, _fapi_exchange_info_rules
   200→   200→        now = time.time()
   201→   201→        if (
   202→   202→            _fapi_exchange_info_rules
   203→   203→            and (now - _fapi_exchange_info_loaded_at) < cache_sec
   204→   204→        ):
   205→   205→            return
   206→   206→        info = _api_call("GET", "/fapi/v1/exchangeInfo")
   207→   207→        rules: dict[str, dict[str, Any]] = {}
   208→   208→        for s in info.get("symbols", []):
   209→   209→            s_pair = str(s.get("symbol", "")).upper()
   210→   210→            if not s_pair:
   211→   211→                continue
   212→   212→            s_min_base = 0.0
   213→   213→            s_min_usdt = 0.0
   214→   214→            s_qty_prec = int(s.get("quantityPrecision", 3))
   215→   215→            s_quote_prec = int(s.get("pricePrecision", 2))
   216→   216→            for f in s.get("filters", []):
   217→   217→                ftype = f.get("filterType")
   218→   218→                if ftype == "LOT_SIZE":
   219→   219→                    s_min_base = float(f.get("minQty", 0))
   220→   220→                elif ftype in ("MIN_NOTIONAL", "NOTIONAL"):
   221→   221→                    s_min_usdt = float(f.get("notional", f.get("minNotional", 0)) or 0)
   222→   222→            if s_min_base <= 0:
   223→   223→                s_min_base = 10 ** (-s_qty_prec)
   224→   224→            rules[s_pair] = {
   225→   225→                "symbol": s_pair,
   226→   226→                "min_trade_usdt": s_min_usdt,
   227→   227→                "min_trade_base": s_min_base,
   228→   228→                "quantity_precision": s_qty_prec,
   229→   229→                "quote_precision": s_quote_prec,
   230→   230→                "status": s.get("status", ""),
   231→   231→            }
   232→   232→        _fapi_exchange_info_rules = rules
   233→   233→        _fapi_exchange_info_loaded_at = now

   248→   248→    def fetch_symbol_rules(
   249→   249→        self, pair: str, cache_sec: int = 3600
   250→   250→    ) -> Optional[dict[str, Any]]:
   251→   251→        now = time.time()
   252→   252→        cached = _symbol_rules_cache.get(pair)
   253→   253→        if cached and (now - cached[0]) < cache_sec:
   254→   254→            return dict(cached[1])
   255→   255→        try:
   256→   256→            self._ensure_spot_exchange_info(cache_sec)
   257→   257→            sym = _exchange_info_symbols.get(pair.upper())
   258→   258→            if not sym:
   259→   259→                return dict(cached[1]) if cached else None
   260→   260→            info = {"symbols": [sym]}
   261→   261→        except Exception:
   262→   262→            return dict(cached[1]) if cached else None
   263→   263→        symbols = info.get("symbols") or []
   264→   264→        if not symbols:
   265→   265→            return dict(cached[1]) if cached else None
   266→   266→        sym = symbols[0]
   267→   267→        min_base = 0.0
   268→   268→        min_usdt = 0.0
   269→   269→        qty_prec = 8
   270→   270→        quote_prec = 8
   271→   271→        for f in sym.get("filters", []):
   272→   272→            ftype = f.get("filterType")
   273→   273→            if ftype == "LOT_SIZE":
   274→   274→                min_base = float(f.get("minQty", 0))
   275→   275→                step = f.get("stepSize", "0.00000001")
   276→   276→                qty_prec = len(step.rstrip("0").split(".")[-1]) if "." in step else 0
   277→   277→            elif ftype in ("NOTIONAL", "MIN_NOTIONAL"):
   278→   278→                min_usdt = float(f.get("minNotional", f.get("notional", 0)) or 0)
   279→   279→            elif ftype == "PRICE_FILTER":
   280→   280→                tick = f.get("tickSize", "0.01")
   281→   281→                quote_prec = len(tick.rstrip("0").split(".")[-1]) if "." in tick else 2
   282→   282→        if min_base <= 0:
   283→   283→            min_base = 10 ** (-qty_prec)
   284→   284→        rules = {
   285→   285→            "symbol": pair,
   286→   286→            "min_trade_usdt": min_usdt,
   287→   287→            "min_trade_base": min_base,
   288→   288→            "quantity_precision": qty_prec,
   289→   289→            "quote_precision": quote_prec,
   290→   290→            "status": sym.get("status", ""),
   291→   291→        }
   292→   292→        _symbol_rules_cache[pair] = (now, rules)
   293→   293→        return dict(rules)

   295→   295→    def fetch_futures_symbol_rules(
   296→   296→        self, pair: str, cache_sec: int = 3600
   297→   297→    ) -> dict[str, Any] | None:
   298→   298→        now = time.time()
   299→   299→        cached = _futures_rules_cache.get(pair)
   300→   300→        if cached and (now - cached[0]) < cache_sec:
   301→   301→            return dict(cached[1])
   302→   302→        try:
   303→   303→            self._ensure_fapi_exchange_info(cache_sec)
   304→   304→        except Exception:
   305→   305→            return dict(cached[1]) if cached else None
   306→   306→        rules = _fapi_exchange_info_rules.get(pair.upper())
   307→   307→        if rules:
   308→   308→            _futures_rules_cache[pair] = (now, dict(rules))
   309→   309→            return dict(rules)
   310→   310→        return dict(cached[1]) if cached else None

   312→   312→    def transfer_asset(
   313→   313→        self, asset: str, amount: float, from_account: str, to_account: str
   314→   314→    ) -> bool:
   315→   315→        """Transfer between main spot and UM futures."""
   316→   316→        # 1: spot -> UM futures, 2: UM futures -> spot
   317→   317→        transfer_type = (
   318→   318→            1
   319→   319→            if from_account == "spot" and to_account == "futures"
   320→   320→            else 2
   321→   321→            if from_account == "futures" and to_account == "spot"
   322→   322→            else None
   323→   323→        )
   324→   324→        if not transfer_type:
   325→   325→            return False
   326→   326→
   327→   327→        try:
   328→   328→            res = _api_call(
   329→   329→                "POST",
   330→   330→                "/sapi/v1/futures/transfer",
   331→   331→                {
   332→   332→                    "type": transfer_type,
   333→   333→                    "asset": asset,
   334→   334→                    "amount": f"{amount:.8f}".rstrip("0").rstrip("."),
   335→   335→                },
   336→   336→                signed=True,
   337→   337→            )
   338→   338→            return "tranId" in res
   339→   339→        except Exception:
   340→   340→            return False

   356→   356→        try:
   357→   357→            f_data = _api_call("GET", "/fapi/v2/account", signed=True)
   358→   358→            for asset in f_data.get("assets", []):
   359→   359→                if str(asset.get("asset", "")).upper() == "USDT":
   360→   360→                    futures_usdt = float(asset.get("availableBalance", "0"))
   361→   361→                    break
   362→   362→        except Exception as e:
   363→   363→            print(f"fetch_usdt_account_balances futures error: {e}", file=sys.stderr)
   364→   364→            raise e
   365→   365→
   366→   366→        return {"spot": spot_usdt, "futures": futures_usdt}

   368→   368→    def fetch_asset_market(
   369→   369→        self, asset: str, quote: str = "USDT", cfg: dict[str, Any] | None = None
   370→   370→    ) -> dict[str, Any]:
   371→   371→        pair = make_pair(asset, quote)
   372→   372→        price = self.get_ticker(pair)
   373→   373→        rules = self.fetch_symbol_rules(pair)
   374→   374→        if rules is None:
   375→   375→            return {
   376→   376→                "symbol": asset,
   377→   377→                "price": price,
   378→   378→                "rules_error": True,
   379→   379→                "venue": self.venue_id,
   380→   380→            }
   381→   381→        limits = rules_for_price(rules, price)
   382→   382→
   383→   383→        cfg = cfg or {}
   384→   384→        tf = resolve_timeframes(cfg)
   385→   385→
   386→   386→        klines_1d = [
   387→   387→            parse_kline_ohlcv(k)
   388→   388→            for k in self.get_klines(pair, tf["slow"]["interval"], tf["slow"]["limit"])
   389→   389→            if k
   390→   390→        ]
   391→   391→        klines_4h = [
   392→   392→            parse_kline_ohlcv(k)
   393→   393→            for k in self.get_klines(pair, tf["mid"]["interval"], tf["mid"]["limit"])
   394→   394→            if k
   395→   395→        ]
   396→   396→        klines_1w = [
   397→   397→            parse_kline_ohlcv(k)
   398→   398→            for k in self.get_klines(
   399→   399→                pair, tf["macro"]["interval"], tf["macro"]["limit"]
   400→   400→            )
   401→   401→            if k
   402→   402→        ]
   403→   403→        return {
   404→   404→            "symbol": asset,
   405→   405→            "pair": pair,
   406→   406→            "price": price,
   407→   407→            "rules_error": False,
   408→   408→            "venue": self.venue_id,
   409→   409→            "symbol_rules": rules,
   410→   410→            **limits,
   411→   411→            "klines_1d": klines_1d,
   412→   412→            "klines_4h": klines_4h,
   413→   413→            "klines_1w": klines_1w,
   414→   414→        }

   416→   416→    def fetch_balances(self, coins: list[str]) -> dict[str, float]:
   417→   417→        # Do not swallow exceptions: balance fetch failure must propagate to abort, never return all zeros
   418→   418→        # (otherwise strategy misjudges account as empty and only uses cash orders)
   419→   419→        balances: dict[str, float] = {c: 0.0 for c in coins}
   420→   420→
   421→   421→        data = _api_call("GET", "/api/v3/account", signed=True)
   422→   422→        for asset in data.get("balances", []):
   423→   423→            coin = str(asset.get("asset", "")).upper()
   424→   424→            if coin in balances:
   425→   425→                balances[coin] = float(asset.get("free", "0"))
   426→   426→
   427→   427→        # Fetch futures balance if USDT is requested
   428→   428→        if "USDT" in balances:
   429→   429→            try:
   430→   430→                f_data = _api_call("GET", "/fapi/v2/account", signed=True)
   431→   431→                for asset in f_data.get("assets", []):
   432→   432→                    if str(asset.get("asset", "")).upper() == "USDT":
   433→   433→                        balances["USDT"] += float(asset.get("marginBalance", "0"))
   434→   434→            except Exception as e:
   435→   435→                print(f"fetch_futures_balances error: {e}", file=sys.stderr)
   436→   436→
   437→   437→        return balances

   439→   439→    def initialize_futures_symbol(self, pair: str) -> None:
   440→   440→        """Initialize futures configuration (marginType, leverage, positionSide) for a specific pair."""
   441→   441→        if pair in _initialized_symbols:
   442→   442→            return

   472→   472→    def fetch_live_state(self, assets: list[str]) -> dict[str, Any]:
   473→   473→        """Fetch unified global state: spot balances, margin debt, futures margin, futures positions."""
   474→   474→        # 1. Spot Balances
   475→   475→        spot_balances: dict[str, float] = {c: 0.0 for c in assets}
   476→   476→        try:
   477→   477→            data = _api_call("GET", "/api/v3/account", signed=True)
   478→   478→            for asset in data.get("balances", []):
   479→   479→                coin = str(asset.get("asset", "")).upper()
   480→   480→                if coin in spot_balances:
   481→   481→                    spot_balances[coin] = float(asset.get("free", "0")) + float(
   482→   482→                        asset.get("locked", "0")
   483→   483→                    )
   484→   484→        except Exception as e:
   485→   485→            print(f"fetch_live_state spot error: {e}", file=sys.stderr)
   486→   486→            raise e  # Must propagate if spot fails
   487→   487→
   488→   488→        # 2. Cross Margin Balances & Debt (for Reverse Arb)
   489→   489→        margin_balances: dict[str, float] = {c: 0.0 for c in assets}
   490→   490→        try:
   491→   491→            margin_data = _api_call("GET", "/sapi/v1/margin/account", signed=True)
   492→   492→            for asset in margin_data.get("userAssets", []):
   493→   493→                coin = str(asset.get("asset", "")).upper()
   494→   494→                if coin in margin_balances:
   495→   495→                    free = float(asset.get("free", "0")) + float(
   496→   496→                        asset.get("locked", "0")
   497→   497→                    )
   498→   498→                    borrowed = float(asset.get("borrowed", "0"))
   499→   499→                    interest = float(asset.get("interest", "0"))
   500→   500→                    # Net balance = free - borrowed - interest
   501→   501→                    margin_balances[coin] = free - borrowed - interest
   502→   502→        except Exception as e:
   503→   503→            # If cross-margin account doesn't exist, ignore
   504→   504→            pass
   505→   505→
   506→   506→        # Combine Spot and Margin balances
   507→   507→        combined_balances = {c: spot_balances[c] + margin_balances[c] for c in assets}
   508→   508→
   509→   509→        # 3. Futures Margin (USDT)
   510→   510→        futures_usdt = 0.0
   511→   511→        try:
   512→   512→            f_data = _api_call("GET", "/fapi/v2/account", signed=True)
   513→   513→            for asset in f_data.get("assets", []):
   514→   514→                if str(asset.get("asset", "")).upper() == "USDT":
   515→   515→                    # Use walletBalance instead of marginBalance to avoid double-counting unrealized PnL
   516→   516→                    futures_usdt = float(asset.get("walletBalance", "0"))
   517→   517→        except Exception as e:
   518→   518→            print(f"fetch_live_state futures account error: {e}", file=sys.stderr)
   519→   519→            raise e
   520→   520→
   521→   521→        if "USDT" in combined_balances:
   522→   522→            combined_balances["USDT"] += futures_usdt
   523→   523→
   524→   524→        # 4. Futures Positions
   525→   525→        positions = {}
   526→   526→        try:
   527→   527→            pos_data = _api_call("GET", "/fapi/v2/positionRisk", signed=True)
   528→   528→            for pos in pos_data:
   529→   529→                amt = float(pos.get("positionAmt", "0"))
   530→   530→                if abs(amt) > 1e-9:
   531→   531→                    sym_raw = pos.get("symbol", "")
   532→   532→                    base_sym = sym_raw
   533→   533→                    if sym_raw.endswith("USDT"):
   534→   534→                        base_sym = sym_raw[:-4]
   535→   535→
   536→   536→                    entry = float(pos.get("entryPrice", "0"))
   537→   537→                    unrealized = float(pos.get("unRealizedProfit", "0"))
   538→   538→                    liq_price = float(pos.get("liquidationPrice", "0"))
   539→   539→                    lev = float(pos.get("leverage", "1"))
   540→   540→                    positions[base_sym] = {
   541→   541→                        "amount": abs(amt),
   542→   542→                        "side": "long" if amt > 0 else "short",
   543→   543→                        "entry_price": entry,
   544→   544→                        "unrealized_pnl": unrealized,
   545→   545→                        "liq_price": liq_price,
   546→   546→                        "leverage": lev,
   547→   547→                    }
   548→   548→        except Exception as e:
   549→   549→            print(f"fetch_live_state positionRisk error: {e}", file=sys.stderr)
   550→   550→            raise e
   551→   551→
   552→   552→        return {"balances": combined_balances, "futures_positions": positions}

   642→   642→    def place_margin_order(
   643→   643→        self,

   652→   652→
   653→   653→        side_effect:
   654→   654→          - MARGIN_BUY : auto-borrow missing assets (SELL = borrow to sell)

   660→   660→            qty = f"{amount_base:.{quantity_precision}f}"

   675→   675→                signed=True,
   676→   676→            )

   700→   700→        except urllib.error.HTTPError as e:
   701→   701→            try:
   702→   702→                body = e.read().decode()
   703→   703→            except Exception:
   704→   704→                body = ""
   705→   705→            return False, {"error": f"HTTP {e.code}: {body[:200]}"}
