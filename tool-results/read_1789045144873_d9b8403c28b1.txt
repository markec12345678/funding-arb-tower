   157→def _fetch_hyperliquid(symbol: str) -> dict[str, float]:
   158→    """Hyperliquid fees via the public `userFees` info endpoint (account-wide).
   159→
   160→    Effective rate = base tier rate × (1 − staking discount) × (1 − referral
   161→    discount), per the docs (multiplicative). With HYPERLIQUID_ADDRESS set the
   162→    rate reflects that account's HYPE-staking tier; otherwise the zero address
   163→    returns the base schedule. Cached account-wide (fee is not per-symbol).
   164→    """
   165→    global _HL_FEE_CACHE
   166→    now = time.time()
   167→    if _HL_FEE_CACHE and now - _HL_FEE_CACHE[0] < _HL_FEE_TTL_SEC:
   168→        return dict(_HL_FEE_CACHE[1])
   169→
   170→    import json as _json
   171→    import urllib.request
   172→
   173→    addr = (
   174→        os.environ.get("HYPERLIQUID_ADDRESS")
   175→        or os.environ.get("HYPERLIQUID_PUBLIC_ADDRESS")
   176→        or "0x0000000000000000000000000000000000000000"
   177→    ).strip()
   178→    body = _json.dumps({"type": "userFees", "user": addr}).encode()
   179→    req = urllib.request.Request(
   180→        "https://api.hyperliquid.xyz/info",
   181→        data=body,
   182→        headers={"Content-Type": "application/json"},
   183→    )
   184→    with urllib.request.urlopen(req, timeout=15) as resp:
   185→        d = _json.loads(resp.read().decode())
   186→
   187→    taker = float(d.get("userCrossRate", 0.00045) or 0.00045)
   188→    maker = float(d.get("userAddRate", 0.00015) or 0.00015)
   189→    staking = float((d.get("activeStakingDiscount") or {}).get("discount", 0) or 0)
   190→    referral = float(d.get("activeReferralDiscount", 0) or 0)
   191→    rates = {
   192→        "taker_pct": taker * 100.0 * (1 - staking) * (1 - referral),
   193→        "maker_pct": maker * 100.0 * (1 - staking),
   194→    }
   195→    _HL_FEE_CACHE = (now, rates)
   196→    return dict(rates)

   337→def fetch_futures_fee_rates(
   338→    venue: str,
   339→    symbol: str,
   340→    *,
   341→    use_cache: bool = True,
   342→) -> dict[str, float]:
   343→    """Fetch {taker_pct, maker_pct} for one venue + symbol."""
   344→    v = venue.lower()
   345→    sym = normalize_symbol(symbol)
   346→    key = (v, sym)
   347→    now = time.time()
   348→    if use_cache and key in _CACHE:
   349→        ts, rates = _CACHE[key]
   350→        if now - ts < _CACHE_TTL_SEC:
   351→            return dict(rates)
   352→
   353→    fetcher = _FETCHERS.get(v)
   354→    if fetcher:
   355→        try:
   356→            rates = fetcher(sym)
   357→            _CACHE[key] = (now, rates)
   358→            return dict(rates)
   359→        except Exception:
   360→            pass
   361→
   362→    return {
   363→        "taker_pct": default_taker_pct(v),
   364→        "maker_pct": DEFAULT_MAKER_PCT.get(v, 0.02),
   365→    }

   399→def prefetch_spot_fee_rates(
   400→    pairs: list[tuple[str, str]],
   401→    *,
   402→    workers: int = 8,
   403→) -> dict[tuple[str, str], dict[str, float]]:
   404→    """Parallel prefetch spot fee rates. pairs = [(venue, symbol), ...].
   405→
   406→    Checks _SPOT_CACHE first; only fetches stale/missing entries.
   407→    """
   408→    unique = list({(v.lower(), normalize_symbol(s)) for v, s in pairs})
   409→    now = time.time()
   410→
   411→    cached: dict[tuple[str, str], dict[str, float]] = {}
   412→    stale: list[tuple[str, str]] = []
   413→    for item in unique:
   414→        entry = _SPOT_CACHE.get(item)
   415→        if entry and now - entry[0] < _CACHE_TTL_SEC:
   416→            cached[item] = dict(entry[1])
   417→        else:
   418→            stale.append(item)
   419→
   420→    if not stale:
   421→        return cached
   422→
   423→    def _one(item: tuple[str, str]) -> tuple[tuple[str, str], dict[str, float]]:
   424→        v, sym = item
   425→        return (v, sym), fetch_spot_fee_rates(v, sym, use_cache=False)
   426→
   427→    raw = run_io_parallel(stale, _one, max_workers=workers, swallow_errors=True)
   428→    for k, v in raw.items():
   429→        _SPOT_CACHE[k] = (time.time(), v)
   430→    out: dict[tuple[str, str], dict[str, float]] = dict(cached)
   431→    for v, sym in stale:
   432→        out[(v, sym)] = raw.get((v, sym)) or {
   433→            "taker_pct": default_spot_taker_pct(v),
   434→            "maker_pct": default_spot_taker_pct(v),
   435→        }
   436→    return out

   439→def carry_two_leg_fee_pct(
   440→    venue: str,
   441→    symbol: str,
   442→    *,
   443→    futures_cache: dict[tuple[str, str], dict[str, float]] | None = None,
   444→    spot_cache: dict[tuple[str, str], dict[str, float]] | None = None,
   445→) -> tuple[float, float, float]:
   446→    """Cash-and-carry open cost: (spot_pct, futures_pct, total_pct) in percentage points."""
   447→    v = venue.lower()
   448→    sym = normalize_symbol(symbol)
   449→    key = (v, sym)
   450→    if futures_cache and key in futures_cache:
   451→        futures_pct = float(futures_cache[key].get("taker_pct", default_taker_pct(v)))
   452→    else:
   453→        futures_pct = float(fetch_futures_fee_rates(v, sym)["taker_pct"])
   454→    if spot_cache and key in spot_cache:
   455→        spot_pct = float(spot_cache[key].get("taker_pct", default_spot_taker_pct(v)))
   456→    else:
   457→        spot_pct = float(fetch_spot_fee_rates(v, sym)["taker_pct"])
   458→    return spot_pct, futures_pct, spot_pct + futures_pct

   461→def prefetch_futures_fee_rates(
   462→    pairs: list[tuple[str, str]],
   463→    *,
   464→    workers: int = 8,
   465→) -> dict[tuple[str, str], dict[str, float]]:
   466→    """Parallel prefetch. pairs = [(venue, symbol), ...].
   467→
   468→    Checks the per-symbol _CACHE first; only fetches pairs whose cache entry
   469→    is stale or missing. Exchange fee rates rarely change (_CACHE_TTL_SEC=3600),
   470→    so repeated scans within an hour skip all HTTP work here.
   471→    """
   472→    unique = list({(v.lower(), normalize_symbol(s)) for v, s in pairs})
   473→    now = time.time()
   474→
   475→    # Fast path: serve everything from cache if fresh.
   476→    cached: dict[tuple[str, str], dict[str, float]] = {}
   477→    stale: list[tuple[str, str]] = []
   478→    for item in unique:
   479→        entry = _CACHE.get(item)
   480→        if entry and now - entry[0] < _CACHE_TTL_SEC:
   481→            cached[item] = dict(entry[1])
   482→        else:
   483→            stale.append(item)
   484→
   485→    if not stale:
   486→        return cached  # all fresh, zero HTTP
   487→
   488→    def _one(item: tuple[str, str]) -> tuple[tuple[str, str], dict[str, float]]:
   489→        v, sym = item
   490→        return (v, sym), fetch_futures_fee_rates(v, sym, use_cache=False)
   491→
   492→    raw = run_io_parallel(stale, _one, max_workers=workers, swallow_errors=True)
   493→    for k, v in raw.items():
   494→        _CACHE[k] = (time.time(), v)
   495→    # Merge cached + freshly fetched
   496→    out: dict[tuple[str, str], dict[str, float]] = dict(cached)
   497→    for v, sym in stale:
   498→        out[(v, sym)] = raw.get((v, sym)) or {
   499→            "taker_pct": default_taker_pct(v),
   500→            "maker_pct": DEFAULT_MAKER_PCT.get(v, 0.02),
   501→        }
   502→    return out

   505→def offline_fee_cache_from_by_base(
   506→    by_base: dict[str, dict[str, dict[str, Any]]],
   507→) -> dict[tuple[str, str], dict[str, float]]:
   508→    """Build venue-default fee cache without network (tests / offline backtest)."""
   509→    out: dict[tuple[str, str], dict[str, float]] = {}
   510→    for base, venue_map in by_base.items():
   511→        for venue, info in venue_map.items():
   512→            v = venue.lower()
   513→            sym = normalize_symbol(str(info.get("symbol") or f"{base}USDT"))
   514→            out[(v, sym)] = {
   515→                "taker_pct": default_taker_pct(v),
   516→                "maker_pct": DEFAULT_MAKER_PCT.get(v, 0.02),
   517→            }
   518→    return out

   526→    """Build fee cache from scan_pure_futures_spreads `by_base` structure."""
   527→    pairs: list[tuple[str, str]] = []
   528→    for base, venue_map in by_base.items():
   529→        for venue, info in venue_map.items():
   530→            sym = str(info.get("symbol") or f"{base}USDT")
   531→            pairs.append((venue, sym))
   532→    return prefetch_futures_fee_rates(pairs, workers=workers)
   533→
   534→
   535→def taker_fee_pct(
   536→    venue: str,
   537→    symbol: str,
   538→    *,
   539→    fee_cache: dict[tuple[str, str], dict[str, float]] | None = None,
   540→    config_overrides: dict[str, float] | None = None,
   541→) -> float:
   542→    """Taker fee in percentage points for venue+symbol."""
   543→    v = venue.lower()
   544→    sym = normalize_symbol(symbol)
   545→    if config_overrides and v in config_overrides:
   546→        return float(config_overrides[v])
   547→    if fee_cache is not None:
   548→        cached = fee_cache.get((v, sym))
   549→        if cached:
   550→            return float(cached.get("taker_pct", default_taker_pct(v)))
   551→    return fetch_futures_fee_rates(v, sym)["taker_pct"]
   552→
   553→
   554→def pair_open_taker_fee_pct(
   555→    long_venue: str,
   556→    long_symbol: str,
   557→    short_venue: str,
   558→    short_symbol: str,
   559→    *,
   560→    fee_cache: dict[tuple[str, str], dict[str, float]] | None = None,
   561→    config_overrides: dict[str, float] | None = None,
   562→) -> tuple[float, float, float]:
   563→    """Returns (long_fee_pct, short_fee_pct, total_open_fee_pct)."""
   564→    long_fee = taker_fee_pct(
   565→        long_venue, long_symbol, fee_cache=fee_cache, config_overrides=config_overrides
   566→    )
   567→    short_fee = taker_fee_pct(
   568→        short_venue,
   569→        short_symbol,
   570→        fee_cache=fee_cache,
   571→        config_overrides=config_overrides,
   572→    )
   573→    return long_fee, short_fee, long_fee + short_fee

   584→        "BINANCE_TRADE_API_KEY",

   594→def venue_has_credentials(venue: str) -> bool:
   595→    """True when any known API key env var is set for this venue."""
   596→    for key in _VENUE_CRED_KEYS.get(venue.lower(), []):
   597→        if os.environ.get(key):
   598→            return True
   599→    return False

   601→
   602→def parse_fee_policy(cfg: dict[str, Any] | None = None) -> dict[str, Any]:
   603→    """Normalize fee policy from strategy config.
   604→
   605→    Idempotent: accepts either raw strategy config keys (fee_mode /
   606→    venue_fee_tiers) or an already-parsed policy (mode / venue_tiers), so
   607→    callers can safely re-parse a policy passed down from the server layer.
   608→    """
   609→    cfg = cfg or {}
   610→    tiers = cfg.get("venue_fee_tiers")
   611→    if not isinstance(tiers, dict):
   612→        tiers = cfg.get("venue_tiers")
   613→    mode = cfg.get("fee_mode") or cfg.get("mode") or "auto"
   614→    return {
   615→        "mode": str(mode),
   616→        "venue_tiers": dict(tiers) if isinstance(tiers, dict) else {},
   617→    }
   618→
   619→
   620→def venue_uses_api(venue: str, policy: dict[str, Any] | None = None) -> bool:
   621→    """Whether this venue should fetch fees from API (account or public).
   622→
   623→    Perp-DEX venues with public fee endpoints (no keys) always fetch live, so
   624→    the scanner reflects real venue fees instead of stale hardcoded defaults.
   625→    The user can still force the VIP-tier table via fee_mode='vip_tier'.
   626→    """
   627→    policy = policy or parse_fee_policy()
   628→    mode = policy.get("mode", "auto")
   629→    if mode == "vip_tier":
   630→        return False
   631→    if venue.lower() in _PUBLIC_FEE_VENUES:
   632→        return True
   633→    if mode == "api":
   634→        return venue_has_credentials(venue)
   635→    return venue_has_credentials(venue)

   638→def _tier_taker_pct(venue: str, leg: str, policy: dict[str, Any]) -> float:
   639→    tier_id = policy.get("venue_tiers", {}).get(venue.lower())
   640→    rates = tier_rates(venue, tier_id)
   641→    if leg == "spot":
   642→        return float(rates["spot_taker_pct"])
   643→    return float(rates["futures_taker_pct"])

   646→def resolve_venue_fee(
   647→    venue: str,
   648→    *,
   649→    leg: str = "futures",
   650→    symbol: str = "BTCUSDT",
   651→    policy: dict[str, Any] | None = None,
   652→) -> dict[str, Any]:
   653→    """Resolve taker fee for one venue with source metadata."""
   654→    policy = policy or parse_fee_policy()
   655→    v = venue.lower()
   656→    sym = normalize_symbol(symbol)
   657→    if venue_uses_api(v, policy):
   658→        try:
   659→            if leg == "spot":
   660→                rates = fetch_spot_fee_rates(v, sym)
   661→            else:
   662→                rates = fetch_futures_fee_rates(v, sym)
   663→            return {
   664→                "taker_pct": float(rates["taker_pct"]),
   665→                "maker_pct": float(rates.get("maker_pct", 0.02)),
   666→                "source": "api",
   667→                "tier": None,
   668→            }
   669→        except Exception:
   670→            pass
   671→    tier_id = policy.get("venue_tiers", {}).get(v) or (
   672→        "vip0" if v in ("binance", "bitget", "bybit", "okx") else "default"
   673→    )
   674→    taker = _tier_taker_pct(v, leg, policy)
   675→    return {
   676→        "taker_pct": taker,
   677→        "maker_pct": DEFAULT_MAKER_PCT.get(v, 0.02),
   678→        "source": "tier",
   679→        "tier": tier_id,
   680→    }

   682→
   683→def build_policy_futures_cache(
   684→    by_base: dict[str, dict[str, dict[str, Any]]],
   685→    policy: dict[str, Any] | None = None,
   686→    *,
   687→    workers: int = 8,
   688→) -> dict[tuple[str, str], dict[str, float]]:
   689→    """Futures fee cache honoring auto API vs VIP tier per venue."""
   690→    policy = policy or parse_fee_policy()
   691→    cache: dict[tuple[str, str], dict[str, float]] = {}
   692→    api_pairs: list[tuple[str, str]] = []
   693→
   694→    # Precompute which venues use API fees — avoids ~500 os.environ scans
   695→    # (one per (base, venue) pair) when most venues repeat across bases.
   696→    all_venues = {
   697→        venue.lower() for venue_map in by_base.values() for venue in venue_map
   698→    }
   699→    api_venues = {v for v in all_venues if venue_uses_api(v, policy)}
   700→
   701→    for base, venue_map in by_base.items():
   702→        for venue, info in venue_map.items():
   703→            v = venue.lower()
   704→            sym = normalize_symbol(str(info.get("symbol") or f"{base}USDT"))
   705→            key = (v, sym)
   706→            if v in api_venues:
   707→                api_pairs.append(key)
   708→            else:
   709→                cache[key] = {
   710→                    "taker_pct": _tier_taker_pct(v, "futures", policy),
   711→                    "maker_pct": DEFAULT_MAKER_PCT.get(v, 0.02),
   712→                }
   713→
   714→    if api_pairs:
   715→        api_cache = prefetch_futures_fee_rates(api_pairs, workers=workers)
   716→        cache.update(api_cache)
   717→
   718→    return cache
   719→

   721→def build_policy_carry_caches(
   722→    venue: str,
   723→    symbols: list[str],
   724→    policy: dict[str, Any] | None = None,
   725→    *,
   726→    workers: int = 8,
   727→) -> tuple[
   728→    dict[tuple[str, str], dict[str, float]], dict[tuple[str, str], dict[str, float]]
   729→]:
   730→    """Spot + futures caches for cash-and-carry scanner."""
   731→    policy = policy or parse_fee_policy()
   732→    v = venue.lower()
   733→    pairs = [(v, normalize_symbol(s)) for s in symbols]
   734→    futures_cache: dict[tuple[str, str], dict[str, float]] = {}
   735→    spot_cache: dict[tuple[str, str], dict[str, float]] = {}
   736→
   737→    if venue_uses_api(v, policy):
   738→        futures_cache = prefetch_futures_fee_rates(pairs, workers=workers)
   739→        spot_cache = prefetch_spot_fee_rates(pairs, workers=workers)
   740→    else:
   741→        spot_taker = _tier_taker_pct(v, "spot", policy)
   742→        futures_taker = _tier_taker_pct(v, "futures", policy)
   743→        maker = DEFAULT_MAKER_PCT.get(v, 0.02)
   744→        for key in pairs:
   745→            spot_cache[key] = {"taker_pct": spot_taker, "maker_pct": spot_taker}
   746→            futures_cache[key] = {"taker_pct": futures_taker, "maker_pct": maker}
   747→
   748→    return futures_cache, spot_cache

   751→def futures_config_overrides(
   752→    venues: list[str],
   753→    policy: dict[str, Any] | None = None,
   754→) -> dict[str, float] | None:
   755→    """Per-venue taker overrides for venues not using API (pair_open_taker_fee_pct)."""
   756→    policy = policy or parse_fee_policy()
   757→    overrides: dict[str, float] = {}
   758→    for venue in venues:
   759→        v = venue.lower()
   760→        if not venue_uses_api(v, policy):
   761→            overrides[v] = _tier_taker_pct(v, "futures", policy)
   762→    return overrides or None
