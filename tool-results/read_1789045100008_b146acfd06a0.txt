    47→def _ensure_env() -> None:
    48→    global _env_loaded
    49→    if _env_loaded:
    50→        return
    51→    if os.environ.get("BYBIT_API_KEY"):
    52→        _env_loaded = True
    53→        return
    54→    try:
    55→        with open(CONFIG_PATH) as f:
    56→            for k, v in json.load(f).get("env", {}).items():
    57→                if v and k not in os.environ:
    58→                    os.environ[k] = str(v)
    59→    except (OSError, json.JSONDecodeError):
    60→        pass
    61→    _env_loaded = True

    64→def _get_key() -> str:

    74→def _sign(payload: str) -> str:
    75→    return hmac.new(
    76→        _get_secret().encode(), payload.encode(), hashlib.sha256
    77→    ).hexdigest()

    80→def _api_call(
    81→    method: str, path: str, params: Optional[dict] = None, body: Optional[dict] = None
    82→) -> dict:
    83→    key, secret = _get_key(), _get_secret()
    84→    if not (key and secret):
    85→        raise RuntimeError(
    86→            "Bybit API credentials missing: please set BYBIT_API_KEY / BYBIT_SECRET_KEY, "
    87→            "or configure them in ~/.funding-arb/credentials.json under env."
    88→        )
    89→
    90→    recv_window = "5000"
    91→    ts = str(int(time.time() * 1000))
    92→
    93→    if method == "GET" and params:
    94→        query = urllib.parse.urlencode(params)
    95→    else:
    96→        query = ""
    97→
    98→    body_str = json.dumps(body) if body else ""
    99→    full_path = path + ("?" + query if query else "")
   100→
   101→    prehash = ts + key + recv_window + (query if method == "GET" else body_str)
   102→    sig = _sign(prehash)
   103→
   104→    headers = {
   105→        "X-BAPI-API-KEY": key,
   106→        "X-BAPI-SIGN": sig,
   107→        "X-BAPI-TIMESTAMP": ts,
   108→        "X-BAPI-RECV-WINDOW": recv_window,
   109→        "Content-Type": "application/json",
   110→    }
   111→
   112→    retries = 3 if method == "GET" else 1
   113→    last_err: Optional[Exception] = None
   114→    for attempt in range(retries):
   115→        url = BASE + full_path
   116→        req = urllib.request.Request(url, headers=headers, method=method)
   117→        if body_str:
   118→            req.data = body_str.encode()
   119→        try:
   120→            with urllib.request.urlopen(req, timeout=15) as resp:
   121→                data = json.loads(resp.read().decode())
   122→            if data.get("retCode") != 0:
   123→                raise RuntimeError(
   124→                    f"Bybit API error: {data.get('retMsg', data.get('retCode'))}"
   125→                )
   126→            return data
   127→        except Exception as e:
   128→            last_err = e
   129→            if method == "GET" and attempt < retries - 1:
   130→                time.sleep(0.5 * (attempt + 1))
   131→                continue
   132→            raise
   133→    raise last_err if last_err else RuntimeError("bybit _api_call failed")

   160→    def get_all_spot_tickers(self, cache_sec: int = 5) -> dict[str, float]:
   161→        """Bulk spot last prices {PAIR: price}. Cached briefly for screener loops."""
   162→        global _spot_ticker_loaded_at, _spot_ticker_prices
   163→        now = time.time()
   164→        if _spot_ticker_prices and (now - _spot_ticker_loaded_at) < cache_sec:
   165→            return dict(_spot_ticker_prices)
   166→        try:
   167→            data = http_get_json(f"{BASE}/v5/market/tickers?category=spot")
   168→            _spot_ticker_prices = {
   169→                str(r.get("symbol", "")).upper(): float(r.get("lastPrice", 0) or 0)
   170→                for r in data.get("result", {}).get("list", [])
   171→                if r.get("symbol")
   172→            }
   173→            _spot_ticker_loaded_at = now
   174→        except Exception:
   175→            pass
   176→        return dict(_spot_ticker_prices)

   208→    def fetch_symbol_rules(
   209→        self, pair: str, cache_sec: int = 3600
   210→    ) -> Optional[dict[str, Any]]:
   211→        now = time.time()
   212→        cached = _symbol_rules_cache.get(pair)
   213→        if cached and (now - cached[0]) < cache_sec:
   214→            return dict(cached[1])
   215→        url = f"{BASE}/v5/market/instruments-info?category=spot&symbol={pair}"
   216→        try:
   217→            payload = http_get_json(url)
   218→        except Exception:
   219→            return dict(cached[1]) if cached else None
   220→        rows = payload.get("result", {}).get("list", [])
   221→        if not rows:
   222→            return dict(cached[1]) if cached else None
   223→        info = rows[0]
   224→        lot_filter = {}
   225→        min_notional = {}
   226→        for f in info.get("lotSizeFilter", []):
   227→            lot_filter = f if isinstance(f, dict) else {}
   228→            break
   229→        for f in info.get("minNotionalFilter", []):
   230→            min_notional = f if isinstance(f, dict) else {}
   231→            break
   232→        if not lot_filter:
   233→            lot_filter = info.get("lotSizeFilter") or {}
   234→        if not min_notional:
   235→            min_notional = info.get("minNotionalFilter") or {}
   236→
   237→        base_prec = (
   238→            len(
   239→                str(lot_filter.get("basePrecision", "0.000001"))
   240→                .rstrip("0")
   241→                .split(".")[-1]
   242→            )
   243→            if "." in str(lot_filter.get("basePrecision", "0.000001"))
   244→            else 6
   245→        )
   246→        min_base = float(lot_filter.get("minOrderQty", 0))
   247→        min_usdt = float(min_notional.get("minNotionalValue", 0))
   248→        rules = {
   249→            "symbol": pair,
   250→            "min_trade_usdt": min_usdt,
   251→            "min_trade_base": min_base,
   252→            "quantity_precision": base_prec,
   253→            "quote_precision": 2,
   254→            "status": "Trading" if info.get("status") == "Trading" else "",
   255→        }
   256→        _symbol_rules_cache[pair] = (now, rules)
   257→        return dict(rules)
   258→
   259→    def fetch_futures_symbol_rules(
   260→        self, pair: str, cache_sec: int = 3600
   261→    ) -> dict[str, Any] | None:
   262→        now = time.time()
   263→        cached = _futures_rules_cache.get(pair)
   264→        if cached and (now - cached[0]) < cache_sec:
   265→            return dict(cached[1])
   266→        url = f"{BASE}/v5/market/instruments-info?category=linear&symbol={pair}"
   267→        try:
   268→            payload = http_get_json(url)
   269→        except Exception:
   270→            return dict(cached[1]) if cached else None
   271→        rows = payload.get("result", {}).get("list", [])
   272→        if not rows:
   273→            return dict(cached[1]) if cached else None
   274→        info = rows[0]
   275→        lot_filter = info.get("lotSizeFilter") or {}
   276→        qty_step = str(lot_filter.get("qtyStep", "0.001"))
   277→        qty_prec = len(qty_step.rstrip("0").split(".")[-1]) if "." in qty_step else 3
   278→        min_base = float(lot_filter.get("minOrderQty", 0))
   279→        rules = {
   280→            "symbol": pair,
   281→            "min_trade_usdt": 0,
   282→            "min_trade_base": min_base,
   283→            "quantity_precision": qty_prec,
   284→            "quote_precision": 2,
   285→            "status": "Trading" if info.get("status") == "Trading" else "",
   286→        }
   287→        _futures_rules_cache[pair] = (now, rules)

   314→    def fetch_asset_market(
   315→        self, asset: str, quote: str = "USDT", cfg: dict[str, Any] | None = None
   316→    ) -> dict[str, Any]:
   317→        pair = make_pair(asset, quote)
   318→        price = self.get_ticker(pair)
   319→        rules = self.fetch_symbol_rules(pair)
   320→        if rules is None:
   321→            return {
   322→                "symbol": asset,
   323→                "price": price,
   324→                "rules_error": True,
   325→                "venue": self.venue_id,
   326→            }
   327→        limits = rules_for_price(rules, price)
   328→        cfg = cfg or {}
   329→        tf = resolve_timeframes(cfg)
   330→        klines_1d = [
   331→            parse_kline_ohlcv(k)
   332→            for k in self.get_klines(pair, tf["slow"]["interval"], tf["slow"]["limit"])
   333→            if k
   334→        ]
   335→        klines_4h = [
   336→            parse_kline_ohlcv(k)
   337→            for k in self.get_klines(pair, tf["mid"]["interval"], tf["mid"]["limit"])
   338→            if k
   339→        ]
   340→        klines_1w = [
   341→            parse_kline_ohlcv(k)
   342→            for k in self.get_klines(
   343→                pair, tf["macro"]["interval"], tf["macro"]["limit"]
   344→            )
   345→            if k
   346→        ]
   347→        return {
   348→            "symbol": asset,
   349→            "pair": pair,
   350→            "price": price,
   351→            "rules_error": False,
   352→            "venue": self.venue_id,
   353→            "symbol_rules": rules,
   354→            **limits,
   355→            "klines_1d": klines_1d,
   356→            "klines_4h": klines_4h,
   357→            "klines_1w": klines_1w,
   358→        }

   360→    def fetch_balances(self, coins: list[str]) -> dict[str, float]:
   361→        balances: dict[str, float] = {c: 0.0 for c in coins}
   362→        data = _api_call(
   363→            "GET", "/v5/account/wallet-balance", params={"accountType": "UNIFIED"}
   364→        )
   365→        for acct in data.get("result", {}).get("list", []):
   366→            for coin in acct.get("coin", []):
   367→                c = str(coin.get("coin", "")).upper()
   368→                if c in balances:
   369→                    raw = (
   370→                        coin.get("availableToWithdraw")
   371→                        or coin.get("walletBalance")
   372→                        or "0"
   373→                    )
   374→                    balances[c] = float(raw if raw else 0)
   375→        return balances

   377→    def fetch_live_state(self, assets: list[str]) -> dict[str, Any]:
   378→        balances = self.fetch_balances(assets)
   379→        positions: dict[str, dict[str, Any]] = {}
   380→        try:
   381→            pos_data = _api_call(
   382→                "GET",
   383→                "/v5/position/list",
   384→                params={"category": "linear", "settleCoin": "USDT"},
   385→            )
   386→            for pos in pos_data.get("result", {}).get("list", []):
   387→                amt = float(pos.get("size", 0) or 0)
   388→                if amt < 1e-9:
   389→                    continue
   390→                sym_raw = str(pos.get("symbol", "")).upper()
   391→                base = (
   392→                    sym_raw.replace("USDT", "") if sym_raw.endswith("USDT") else sym_raw
   393→                )
   394→                side_val = str(pos.get("side", "")).lower()
   395→                positions[base] = {
   396→                    "amount": amt,
   397→                    "side": "long" if side_val == "buy" else "short",
   398→                    "entry_price": float(pos.get("avgPrice", 0) or 0),
   399→                    "unrealized_pnl": float(pos.get("unrealisedPnl", 0) or 0),
   400→                    "leverage": float(pos.get("leverage", 1) or 1),
   401→                }
   402→        except Exception:
   403→            pass
   404→        return {"balances": balances, "futures_positions": positions}

   406→    def fetch_futures_positions(self, quote: str = "USDT") -> list[dict[str, Any]]:
   407→        """USDT perpetual positions list (single endpoint; raises on failure)."""
   408→        pos_data = _api_call(
   409→            "GET",
   410→            "/v5/position/list",
   411→            params={"category": "linear", "settleCoin": quote.upper()},
   412→        )
   413→        out: list[dict[str, Any]] = []
   414→        for pos in pos_data.get("result", {}).get("list", []) or []:
   415→            qty = abs(float(pos.get("size", 0) or 0))
   416→            if qty <= 1e-12:
   417→                continue
   418→            out.append(
   419→                {
   420→                    "symbol": str(pos.get("symbol", "")).upper(),
   421→                    "side": "long"
   422→                    if str(pos.get("side", "")).lower() == "buy"
   423→                    else "short",
   424→                    "qty": qty,
   425→                    "entry_price": float(pos.get("avgPrice", 0) or 0),
   426→                    "liq_price": float(pos.get("liqPrice", 0) or 0),
   427→                    "leverage": float(pos.get("leverage", 1) or 1),
   428→                    "unrealized_pnl": float(pos.get("unrealisedPnl", 0) or 0),

   469→    def fetch_borrow_rates(self, coins: list[str]) -> dict[str, float]:
   470→        """Fetch annualized borrow rates from Bybit spot margin. Returns {coin: rate_decimal}."""
   471→        rates: dict[str, float] = {c: 0.0 for c in coins}
   472→        for coin in coins:
   473→            try:
   474→                data = _api_call(
   475→                    "GET",
   476→                    "/v5/spot-margin-trade/data",
   477→                    params={"vipLevel": "No VIP", "currency": coin.upper()},
   478→                )
   479→                for vip_group in data.get("result", {}).get("vipCoinList", []):
   480→                    for item in vip_group.get("list", []):
   481→                        if str(item.get("currency", "")).upper() != coin.upper():
   482→                            continue
   483→                        if not item.get("borrowable"):
   484→                            continue
   485→                        hourly = float(item.get("hourlyBorrowRate", 0) or 0)
   486→                        if hourly > 0:
   487→                            rates[coin.upper()] = hourly * 24 * 365
   488→            except Exception:
   489→                pass
   490→        return rates

   494→    def supports_reverse_arbitrage(self) -> bool:
   495→        """UTA spot margin capability: assumed available without keys; with keys, probes toggle and attempts to enable if off."""
   496→        if not (_get_key() and _get_secret()):
   497→            return True
   498→        try:
   499→            data = _api_call("GET", "/v5/spot-margin-trade/state")
   500→            mode = str((data.get("result") or {}).get("spotMarginMode", "0"))
   501→            if mode == "1":
   502→                return True
   503→            _api_call(
   504→                "POST",
   505→                "/v5/spot-margin-trade/switch-mode",
   506→                body={"spotMarginMode": "1"},
   507→            )
   508→            return True
   509→        except Exception:
   510→            return False

   512→    def fetch_margin_debt(self, assets: list[str]) -> dict[str, float]:
   513→        """UTA per-coin liabilities (borrowAmount + accruedInterest), in base coin units."""
   514→        debt: dict[str, float] = {a.upper(): 0.0 for a in assets}
   515→        try:
   516→            data = _api_call(
   517→                "GET", "/v5/account/wallet-balance", params={"accountType": "UNIFIED"}
   518→            )
   519→            for acct in data.get("result", {}).get("list", []):
   520→                for coin in acct.get("coin", []):
   521→                    c = str(coin.get("coin", "")).upper()
   522→                    if c in debt:
   523→                        debt[c] = float(coin.get("borrowAmount", 0) or 0) + float(
   524→                            coin.get("accruedInterest", 0) or 0
   525→                        )
   526→        except Exception as e:
   527→            print(f"bybit fetch_margin_debt failed: {e}", file=sys.stderr)
   528→        return debt

   534→    def margin_repay(self, asset: str, amount: float) -> bool:
   535→        """UTA manual repayment: prefers official /v5/account/repay, falls back to quick-repayment on failure."""
   536→        coin = asset.upper()
   537→        amt = f"{amount:.8f}".rstrip("0").rstrip(".")
   538→        try:
   539→            _api_call(
   540→                "POST",
   541→                "/v5/account/repay",
   542→                body={"coin": coin, "amount": amt, "repaymentType": "FLEXIBLE"},
   543→            )
   544→            return True
   545→        except Exception as e:
   546→            print(
   547→                f"bybit margin repay {coin} via /account/repay failed: {e}",
   548→                file=sys.stderr,
   549→            )
   550→        try:
   551→            _api_call("POST", "/v5/account/quick-repayment", body={"coin": coin})
   552→            return True
   553→        except Exception as e:
   554→            print(f"bybit margin repay {coin} failed: {e}", file=sys.stderr)
   555→            return False

   557→    def place_margin_order(
   558→        self,
   559→        pair: str,
   560→        side: str,
   561→        amount_base: float,
   562→        quantity_precision: int = 6,
   563→        ref_price: float = 0.0,
   564→    ) -> tuple[bool, dict[str, Any]]:
   565→        """UTA spot margin market order (isLeverage=1).
   566→
   567→        sell: insufficient balance auto-borrows and sells; buy: bought coins auto-offset liabilities.
   568→        Both sides measured in base quantity (buy uses marketUnit=baseCoin).
   569→        """
   570→        client_oid = f"qmgn{int(time.time())}{random.randint(0, 9999)}"
   571→        sz = f"{amount_base:.{quantity_precision}f}".rstrip("0").rstrip(".")
   572→        body: dict[str, Any] = {
   573→            "category": "spot",
   574→            "symbol": pair,
   575→            "side": "Buy" if side == "buy" else "Sell",
   576→            "orderType": "Market",
   577→            "qty": sz or f"{amount_base:.{quantity_precision}f}",
   578→            "isLeverage": 1,
   579→            "orderLinkId": client_oid,
   580→        }
   581→        if side == "buy":
   582→            body["marketUnit"] = "baseCoin"
   583→        submit_ts = time.time()
   584→        try:
   585→            result = _api_call("POST", "/v5/order/create", body=body)
   586→            fill_ts = time.time()
   587→            order_id = result.get("result", {}).get("orderId", "?")
   588→            time.sleep(0.3)
   589→            detail = _api_call(
   590→                "GET",
   591→                "/v5/order/realtime",
   592→                params={"category": "spot", "orderId": order_id},
   593→            )
   594→            od = detail.get("result", {}).get("list", [{}])[0]
   595→            exec_price = float(od.get("avgPrice", 0) or ref_price)
   596→            exec_qty = float(od.get("cumExecQty", 0) or amount_base)
   597→            exec_quote = float(od.get("cumExecValue", 0) or exec_qty * exec_price)
   598→            slippage = (
   599→                round((exec_price - ref_price) / ref_price, 6)
   600→                if ref_price and exec_price
   601→                else None
   602→            )
   603→            return True, {
   604→                "order_id": order_id,
   605→                "exec_price": exec_price,
   606→                "exec_qty": exec_qty,
   607→                "exec_quote_usd": exec_quote,
   608→                "ref_price": ref_price,
   609→                "slippage": slippage,
   610→                "submit_ts": round(submit_ts, 3),
   611→                "fill_ts": round(fill_ts, 3),
   612→                "latency_ms": round((fill_ts - submit_ts) * 1000),
   613→                "order_status": od.get("orderStatus", ""),
   614→            }
   615→        except Exception as e:
   616→            return False, {"error": str(e)}

   792→    def execute_trades(
   793→        self,
   794→        trades: list[dict[str, Any]],
   795→        market: dict[str, dict[str, Any]],
   796→        dry_run: bool,
   797→    ) -> list[dict[str, Any]]:
   798→        results: list[dict[str, Any]] = []
   799→        for trade in trades:
   800→            symbol = trade["symbol"]
   801→            mkt = market.get(symbol, {})
   802→            pair = str(mkt.get("pair") or make_pair(symbol, "USDT"))
   803→            ref_price = float(mkt.get("price", 0))
   804→            record = dict(trade)
   805→            record["dry_run"] = dry_run
   806→            record["ref_price"] = ref_price
   807→            record["venue"] = self.venue_id
   808→            if dry_run:
   809→                record["status"] = "simulated"
   810→                record["order_id"] = None
   811→                record["slippage"] = 0.0
   812→                record["latency_ms"] = 0
   813→                results.append(record)
   814→                continue
   815→            is_margin = str(trade.get("account", "")).lower() == "margin"
   816→            if trade["type"] in ("buy", "sell") and is_margin:
   817→                # Reverse C&C spot leg uses UTA spot margin (isLeverage=1 auto borrow/repay).
   818→                ok, detail = self.place_margin_order(
   819→                    pair,
   820→                    trade["type"],
   821→                    trade["amount_base"],
   822→                    int(mkt.get("quantity_precision", 6)),
   823→                    ref_price=ref_price,
   824→                )
   825→            elif trade["type"] == "buy":
   826→                ok, detail = self.place_buy(
   827→                    pair,
   828→                    trade["amount_usdt"],
   829→                    int(mkt.get("quote_precision", 2)),
   830→                    ref_price=ref_price,
   831→                )
   832→            elif trade["type"] == "sell":
   833→                ok, detail = self.place_sell(
   834→                    pair,
   835→                    trade["amount_base"],
   836→                    int(mkt.get("quantity_precision", 6)),
   837→                    ref_price=ref_price,
   838→                )
   839→            elif trade["type"] in (
   840→                "open_short",
   841→                "close_long",
   842→                "close_short",
   843→                "open_long",
   844→            ):
   845→                ok, detail = self.place_futures_order(
   846→                    pair,
   847→                    trade["type"],
   848→                    trade["amount_base"],
   849→                    int(
   850→                        trade.get("quantity_precision")
   851→                        or mkt.get("quantity_precision", 3)
   852→                    ),
   853→                    ref_price=ref_price,
   854→                )
   855→            else:
   856→                ok, detail = False, {"error": f"Unknown trade type {trade['type']}"}
   857→            record["status"] = "filled" if ok else "failed"
   858→            if ok:
   859→                record.update(
   860→                    {
   861→                        "order_id": detail.get("order_id"),
   862→                        "exec_price": detail.get("exec_price"),
   863→                        "exec_qty": detail.get("exec_qty"),
   864→                        "exec_quote_usd": detail.get("exec_quote_usd"),
   865→                        "slippage": detail.get("slippage"),
   866→                        "latency_ms": detail.get("latency_ms"),
   867→                        "submit_ts": detail.get("submit_ts"),
   868→                        "fill_ts": detail.get("fill_ts"),
   869→                        "order_status": detail.get("order_status"),
   870→                        "error": None,
   871→                    }
   872→                )
   873→            else:
   874→                record["order_id"] = None
   875→                record["error"] = detail.get("error", str(detail))
   876→            results.append(record)
   877→        return results
