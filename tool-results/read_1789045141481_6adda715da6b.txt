   101→def _balance_needs(route: CrossRoute, trade_usd: float) -> list[BalanceNeed]:
   102→    """Estimate USDT requirements per venue (conservative: each leg needs trade_usd + 10% buffer)."""
   103→    buffer = trade_usd * 0.1
   104→    needs: list[BalanceNeed] = []
   105→    if route.direction == "forward":
   106→        fut_need = trade_usd + buffer
   107→        spot_need = trade_usd + buffer
   108→        needs.append(
   109→            BalanceNeed(
   110→                venue=route.futures_venue,
   111→                role="futures_margin",
   112→                required_usd=fut_need,
   113→                available_usd=_venue_usdt_balance(route.futures_venue),
   114→            )
   115→        )
   116→        if route.futures_venue != route.spot_venue:
   117→            needs.append(
   118→                BalanceNeed(
   119→                    venue=route.spot_venue,
   120→                    role="spot_buy",
   121→                    required_usd=spot_need,
   122→                    available_usd=_venue_usdt_balance(route.spot_venue),
   123→                )
   124→            )
   125→        elif route.same_venue:
   126→            combined = trade_usd * 2.1 + buffer
   127→            needs[0].required_usd = combined
   128→            needs[0].role = "futures+spot"
   129→    else:
   130→        fut_need = trade_usd + buffer
   131→        margin_need = trade_usd * 0.5 + buffer
   132→        needs.append(
   133→            BalanceNeed(
   134→                venue=route.futures_venue,
   135→                role="futures_margin",
   136→                required_usd=fut_need,
   137→                available_usd=_venue_usdt_balance(route.futures_venue),
   138→            )
   139→        )
   140→        if route.futures_venue != route.spot_venue:
   141→            needs.append(
   142→                BalanceNeed(
   143→                    venue=route.spot_venue,
   144→                    role="margin_collateral",
   145→                    required_usd=margin_need,
   146→                    available_usd=_venue_usdt_balance(route.spot_venue),
   147→                )
   148→            )
   149→    for n in needs:
   150→        n.deficit_usd = max(0.0, n.required_usd - n.available_usd)
   151→    return needs

   154→def _plan_transfers(
   155→    balances: list[BalanceNeed],
   156→    venues: tuple[str, ...],
   157→    trade_usd: float,
   158→    *,
   159→    dry_run: bool,
   160→) -> list[dict[str, Any]]:
   161→    """Plan USDT transfers from venues with surplus to those with deficit."""
   162→    plans: list[dict[str, Any]] = []
   163→    donors = sorted(
   164→        [b for b in balances if b.available_usd > b.required_usd + 5],
   165→        key=lambda x: -(x.available_usd - x.required_usd),
   166→    )
   167→    for need in balances:
   168→        if need.sufficient:
   169→            continue
   170→        amount = need.deficit_usd
   171→        donor_venue = None
   172→        for d in donors:
   173→            if d.venue == need.venue:
   174→                continue
   175→            surplus = d.available_usd - d.required_usd
   176→            if surplus >= amount:
   177→                donor_venue = d.venue
   178→                break
   179→        if donor_venue is None:
   180→            for v in venues:
   181→                if v == need.venue:
   182→                    continue
   183→                bal = _venue_usdt_balance(v)
   184→                if bal >= amount + 5:
   185→                    donor_venue = v
   186→                    break
   187→        if donor_venue is None:
   188→            plans.append(
   189→                {
   190→                    "from": None,
   191→                    "to": need.venue,
   192→                    "amount": round(amount, 2),
   193→                    "viable": False,
   194→                    "note": f"no donor with >= {amount:.2f} USDT",
   195→                }
   196→            )
   197→            continue
   198→        xfer_plan = build_plan(donor_venue, need.venue, "USDT", amount, dry_run=dry_run)
   199→        if xfer_plan is None:
   200→            plans.append(
   201→                {
   202→                    "from": donor_venue,
   203→                    "to": need.venue,
   204→                    "amount": round(amount, 2),
   205→                    "viable": False,
   206→                    "note": "no common chain route",
   207→                }
   208→            )
   209→            continue
   210→        plans.append({**xfer_plan.to_dict(), "viable": xfer_plan.route.viable})
   211→    return plans

   214→def _pick_route(
   215→    routes: dict[str, list[CrossRoute]],
   216→    *,
   217→    base: str | None,
   218→    direction: Direction,
   219→    min_all_in: float,
   220→) -> CrossRoute | None:
   221→    pool: list[CrossRoute] = []
   222→    if direction in ("forward", "auto"):
   223→        pool.extend(routes.get("forward") or [])
   224→    if direction in ("reverse", "auto"):
   225→        pool.extend(routes.get("reverse") or [])
   226→    if base:
   227→        pool = [r for r in pool if r.base.upper() == base.upper()]
   228→    pool = [r for r in pool if r.net_edge_all_in_pct >= min_all_in]
   229→    pool.sort(key=lambda r: -r.net_edge_all_in_pct)
   230→    return pool[0] if pool else None

   233→def _execution_notes(route: CrossRoute, trade_usd: float) -> list[str]:
   234→    notes: list[str] = []
   235→    if route.same_venue:
   236→        tpl = VENUE_TEMPLATES.get(route.futures_venue)
   237→        notes.append(
   238→            f"Same-venue {route.futures_venue}: run run_cash_and_carry "
   239→            f"(template={tpl or 'n/a'}) base={route.base} trade_usd≈{trade_usd}"
   240→        )
   241→    else:
   242→        if route.direction == "forward":
   243→            notes.append(
   244→                f"Cross-venue FORWARD: {route.futures_venue} open short {route.base} perp "
   245→                f"+ {route.spot_venue} spot buy {route.base} (~{trade_usd} USD)"
   246→            )
   247→        else:
   248→            notes.append(
   249→                f"Cross-venue REVERSE: {route.futures_venue} open long {route.base} perp "
   250→                f"+ {route.spot_venue} margin borrow-sell {route.base}"
   251→            )
   252→        if route.transfer_chain:
   253→            notes.append(
   254→                f"Funding route: prefer {route.transfer_chain.upper()} "
   255→                f"(est fee {route.transfer_fee_usdt:.4f} USDT / {route.transfer_fee_pct:.3f}%)"
   256→            )
   257→        notes.append(
   258→            "Cross-venue auto-trade: --run-executor simulates both legs, --run-executor --live-trades opens live"
   259→        )
   260→    return notes

   263→def build_orchestration_plan(
   264→    pool: UnifiedFundingPool,
   265→    routes: dict[str, list[CrossRoute]],
   266→    *,
   267→    base: str | None,
   268→    direction: Direction,
   269→    trade_usd: float,
   270→    min_all_in: float,
   271→    dry_run: bool,
   272→) -> OrchestrationPlan | None:
   273→    route = _pick_route(routes, base=base, direction=direction, min_all_in=min_all_in)
   274→    if route is None:
   275→        return None
   276→    balances = _balance_needs(route, trade_usd)
   277→    transfers = _plan_transfers(balances, pool.venues, trade_usd, dry_run=dry_run)
   278→    notes = _execution_notes(route, trade_usd)
   279→    return OrchestrationPlan(
   280→        route=route,
   281→        trade_usd=trade_usd,
   282→        balances=balances,
   283→        transfers=transfers,
   284→        execution_notes=notes,
   285→        dry_run=dry_run,
   286→    )

   289→def _print_plan(plan: OrchestrationPlan) -> None:
   290→    r = plan.route
   291→    print(f"\n{'=' * 72}")
   292→    print(f"ORCHESTRATION  {r.direction.upper()}  {r.base}  trade_usd={plan.trade_usd}")
   293→    print(f"{'=' * 72}")
   294→    legs = f"fut={r.futures_venue}  spot/margin={r.spot_venue}"
   295→    same = "same-venue" if r.same_venue else "cross-venue"
   296→    print(
   297→        f"  route: {same}  {legs}\n"
   298→        f"  funding={r.funding_rate_pct:+.4f}%  net={r.net_edge_pct:+.4f}%  "
   299→        f"all-in={r.net_edge_all_in_pct:+.4f}%"
   300→    )
   301→    if not r.same_venue and r.transfer_chain:
   302→        print(
   303→            f"  transfer: {r.transfer_chain.upper()}  "
   304→            f"fee≈{r.transfer_fee_usdt:.4f} USDT ({r.transfer_fee_pct:.3f}%)"
   305→        )
   306→    print("\n  BALANCES:")
   307→    for b in plan.balances:
   308→        flag = "OK" if b.sufficient else f"NEED +{b.deficit_usd:.2f}"
   309→        print(
   310→            f"    {b.venue:8s} [{b.role:16s}]  "
   311→            f"avail={b.available_usd:.2f}  req={b.required_usd:.2f}  {flag}"
   312→        )
   313→    if plan.transfers:
   314→        print("\n  TRANSFERS:")
   315→        for i, t in enumerate(plan.transfers, 1):
   316→            if not t.get("viable", True):
   317→                print(f"    [{i}] SKIP  {t.get('note', 'not viable')}")
   318→                continue
   319→            print(
   320→                f"    [{i}] {t.get('from_venue')}→{t.get('to_venue')}  "
   321→                f"{t.get('amount')} USDT via {str(t.get('canonical', '')).upper()}  "
   322→                f"fee={t.get('total_fee', 0):.4f}"
   323→            )
   324→    print("\n  EXECUTION:")
   325→    for line in plan.execution_notes:
   326→        print(f"    · {line}")
   327→    if plan.dry_run:
   328→        print("\n  [DRY-RUN] No transfers or orders submitted")

   371→def _run_transfers(
   372→    plan: OrchestrationPlan,
   373→    *,
   374→    execute: bool,
   375→    poll_deposit: bool,
   376→    poll_timeout: int,
   377→) -> list[str]:
   378→    logs: list[str] = []
   379→    if not execute:
   380→        return logs
   381→    for t in plan.transfers:
   382→        if not t.get("viable"):
   383→            logs.append(f"skip transfer: {t.get('note')}")
   384→            continue
   385→        from_v = t.get("from_venue")
   386→        to_v = t.get("to_venue")
   387→        amount = float(t.get("amount", 0))
   388→        if not from_v or not to_v or amount <= 0:
   389→            continue
   390→        xfer = build_plan(from_v, to_v, "USDT", amount, dry_run=False)
   391→        if xfer is None or not xfer.route.viable:
   392→            logs.append(f"transfer {from_v}→{to_v} aborted: not viable")
   393→            continue
   394→        since_ms = int(time.time() * 1000)
   395→        step_logs, result = execute_plan(xfer)
   396→        logs.extend(step_logs)
   397→        if result and result.ok and poll_deposit:
   398→            ok, recs = poll_deposit_until(
   399→                to_v,
   400→                "USDT",
   401→                xfer.route.net_est,
   402→                since_ms,
   403→                timeout_s=poll_timeout,
   404→            )
   405→            logs.append(
   406→                f"poll deposit {to_v}: {'OK' if ok else 'TIMEOUT'} ({len(recs)} records)"
   407→            )
   408→    return logs

   420→def _run_pure_futures_mode(args: argparse.Namespace) -> None:
   421→    """--pure-futures mode: scan perp spreads → optional open → optional watcher start."""
   422→    from cli.scan_pure_futures_spreads import scan_pure_futures_spreads  # noqa: E402
   423→    from core.strategy_config import (  # noqa: E402
   424→        apply_strategy_to_pure_futures_cfg,
   425→        load_strategy_config,
   426→        min_edge_for_row_factory,
   427→        strategy_edge_thresholds,
   428→        strategy_fee_policy,
   429→    )
   430→    from execution.pure_futures_executor import open_pure_futures_pair  # noqa: E402
   431→    from execution.settle_mismatch_planner import (  # noqa: E402
   432→        effective_trade_usd,
   433→        filter_candidates_with_mismatch,
   434→    )
   435→
   436→    strat = load_strategy_config()
   437→    cli_venues = [v.strip().lower() for v in args.venues.split(",") if v.strip()]
   438→
   439→    # Load config for pure-futures if available
   440→    cfg: dict[str, Any] = {}
   441→    cfg_path = Path(args.config) if args.config else None
   442→    if cfg_path and cfg_path.exists():
   443→        cfg = json.loads(cfg_path.read_text(encoding="utf-8"))
   444→    else:
   445→        default_cfg = ROOT.parent / PURE_FUTURES_CONFIG
   446→        if default_cfg.exists():
   447→            cfg = json.loads(default_cfg.read_text(encoding="utf-8"))
   448→
   449→    cfg = apply_strategy_to_pure_futures_cfg(cfg)
   450→    pfa = cfg.get("pureFuturesArbitrage") or {}
   451→    if cli_venues:
   452→        pfa["venues"] = cli_venues
   453→        cfg = {**cfg, "pureFuturesArbitrage": pfa}
   454→    if args.trade_usd:
   455→        pfa["tradeUsdPerPair"] = float(args.trade_usd)
   456→        cfg = {**cfg, "pureFuturesArbitrage": pfa}
   457→
   458→    venues = [str(v).lower() for v in pfa.get("venues", cli_venues or DEFAULT_VENUES)]
   459→    min_spread = float(pfa.get("minSpreadPct", strat.get("min_spread_annual", 0.04)))
   460→    min_edge, min_edge_1h, min_edge_mismatch = strategy_edge_thresholds(strat)
   461→    row_min_edge = min_edge_for_row_factory(min_edge, min_edge_1h, min_edge_mismatch)
   462→    fee_policy = strategy_fee_policy(strat)
   463→
   464→    trade_usd = float(pfa.get("tradeUsdPerPair", args.trade_usd))
   465→    max_pairs = int(pfa.get("maxConcurrentPairs", 3))
   466→    exit_edge = float(pfa.get("exitThresholdPct", 0.01))
   467→    max_mark_spread = float(pfa.get("maxMarkSpreadPct", 1.0))
   468→    allow_mismatch = bool(pfa.get("allowSettleMismatch", False))
   469→
   470→    t0 = time.time()
   471→    print(f"Scanning {len(venues)} venues for pure-futures spreads...", file=sys.stderr)
   472→    scan_loose_edge = min_edge
   473→    if min_edge_1h is not None:
   474→        scan_loose_edge = min(min_edge, min_edge_1h)
   475→    scan = scan_pure_futures_spreads(
   476→        venues=venues,
   477→        min_spread=min_spread,
   478→        min_edge=scan_loose_edge,
   479→        max_mark_spread_pct=max_mark_spread,
   480→        workers=args.workers,
   481→        fee_policy=fee_policy,
   482→    )
   483→    elapsed = time.time() - t0
   484→    print(f"Scan done in {elapsed:.1f}s", file=sys.stderr)
   485→
   486→    if args.json:
   487→        # JSON output
   488→        print(json.dumps(scan, ensure_ascii=False, indent=2))
   489→    else:
   490→        _print_pure_futures_plan(scan)
   491→
   492→    # Run executor: open top-N pairs
   493→    if args.run_executor:
   494→        all_rows = list(scan.get("forward", [])) + list(scan.get("reverse", []))
   495→        candidates = [
   496→            r
   497→            for r in all_rows
   498→            if float(r.get("net_edge_pct", 0) or 0) >= row_min_edge(r)
   499→        ]
   500→        candidates = filter_candidates_with_mismatch(
   501→            candidates,
   502→            allow_mismatch=allow_mismatch,
   503→            max_cumulative_outflow_pct=0.5,
   504→            min_adjusted_edge_pct=min_edge,
   505→            min_edge_for_row=row_min_edge,
   506→        )
   507→        candidates.sort(
   508→            key=lambda x: (
   509→                -float(x.get("adjusted_net_edge_pct", 0) or x.get("net_edge_pct", 0))
   510→            )
   511→        )
   512→
   513→        opened = 0
   514→        for row in candidates[:max_pairs]:
   515→            dry_run = not args.live_trades
   516→            row_trade_usd = effective_trade_usd(trade_usd, row)
   517→            res = open_pure_futures_pair(
   518→                str(row["base"]),
   519→                str(row["long_venue"]),
   520→                str(row["short_venue"]),
   521→                row_trade_usd,
   522→                dry_run=dry_run,
   523→                direction=str(row.get("direction", "forward")),
   524→                max_mark_spread_pct=max_mark_spread,
   525→                config=cfg,
   526→                capital_buffer_pct=float(row.get("capital_buffer_pct", 0) or 0),
   527→            )
   528→            status = "OK" if res.ok else f"FAIL({res.state})"
   529→            edge = float(
   530→                row.get("adjusted_net_edge_pct", 0) or row.get("net_edge_pct", 0)
   531→            )
   532→            print(
   533→                f"  OPEN {row['base']} long@{row['long_venue']} short@{row['short_venue']} "
   534→                f"edge={edge:+.4f}% usd={row_trade_usd:.0f} → {status} {res.position_id}"
   535→            )
   536→            if res.ok:
   537→                opened += 1
   538→            for log in res.logs:
   539→                print(f"    · {log}")
   540→
   541→        if not args.live_trades:
   542→            print(f"\n  [DRY-RUN] {opened} pairs simulated")
   543→
   544→    # Start watcher if requested
   545→    if getattr(args, "auto_spread_watch", False):
   546→        from execution.pure_futures_watcher import main as watcher_main  # noqa: E402
   547→
   548→        # Rebuild argv for watcher
   549→        watcher_cfg = args.config or str(ROOT.parent / PURE_FUTURES_CONFIG)
   550→        watcher_argv = [
   551→            "--config",
   552→            watcher_cfg,
   553→            "--interval",
   554→            str(getattr(args, "watch_interval", 60)),
   555→        ]
   556→        if not args.live_trades:
   557→            watcher_argv.append("--dry-run")
   558→        if args.verbose:
   559→            watcher_argv.append("--verbose")
   560→
   561→        print(f"\n  Starting watcher (config={watcher_cfg})...", file=sys.stderr)
   562→        sys.argv = ["pure_futures_watcher"] + watcher_argv
   563→        raise SystemExit(watcher_main())
