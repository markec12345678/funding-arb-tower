     1→#!/usr/bin/env python3
     2→"""Pure Perpetual Futures Spread Arbitrage Scanner — Cross-exchange perpetual contract funding rate spread scanner。
     3→
     4→No spot/borrow needed; only compares funding rate differences across exchanges for USDT perpetuals：
     5→  - Forward: long at the venue with the highest rate, short at the venue with the lowest rate
     6→  - Reverse: long at the venue with the lowest (most negative) rate, short at the venue with the highest (or less negative) rate
     7→
     8→Key difference from cash-and-carry scanner：both legs are perp（lower taker fees, no spot slippage），
     9→Legs can be split across different exchanges; no need for one exchange to support both spot and perp.
    10→
    11→Usage:
    12→  python3 scripts/cli/scan_pure_futures_spreads.py
    13→  python3 scripts/cli/scan_pure_futures_spreads.py --venues binance,bybit,okx,bitget
    14→  python3 scripts/cli/scan_pure_futures_spreads.py --include-dex             # include all perp DEX venues
    15→  python3 scripts/cli/scan_pure_futures_spreads.py --min-spread 0.1 --verbose
    16→  python3 scripts/cli/scan_pure_futures_spreads.py --json
    17→  python3 scripts/cli/scan_pure_futures_spreads.py --watch 5  # scan every 5 minutes and append to JSONL
    18→
    19→Architecture note:
    20→  This is Phase 1 — data-driven scan only. No order execution. The output answers:
    21→  - How many pairs have spreads > 0.1% net?
    22→  - What is the average duration of a spread above threshold?
    23→  - Which venue pairs dominate?
    24→"""

    92→def _mins_to_settle(ts_ms: int) -> str:
    93→    if ts_ms <= 0:
    94→        return "?"
    95→    mins = (ts_ms - time.time() * 1000) / 60000
    96→    if mins < 0:
    97→        return "now"
    98→    if mins < 60:
    99→        return f"{mins:.0f}m"
   100→    return f"{mins / 60:.1f}h"

   120→def fetch_all_fee_rate_rows_by_base(
   121→    venues: list[str],
   122→    workers: int,
   123→    *,
   124→    cache_ttl_sec: float = 0.0,
   125→) -> dict[str, dict[str, dict[str, Any]]]:
   126→    """Fetch funding info from each venue, return {base: {venue: {rate, next_ts, interval_h, mark}}}.
   127→
   128→    If cache_ttl_sec > 0, results are cached at the venue level for that duration.
   129→    Repeated calls within the TTL window return cached data without hitting the network.
   130→    This is useful for the scanner loop and positions enrichment which both call this.
   131→    """
   132→    # Unified cross-scan cache: avoid re-fetching DEX venues on rapid repeated calls
   133→    if cache_ttl_sec > 0:
   134→        now = time.time()
   135→        cached_venues: list[str] = []
   136→        out: dict[str, dict[str, dict[str, Any]]] = {}
   137→        for v in venues:
   138→            entry = _FETCH_ALL_CACHE.get(v)
   139→            if entry and now - entry[0] < cache_ttl_sec:
   140→                for base, info in entry[1].items():
   141→                    if base not in out:
   142→                        out[base] = {}
   143→                    out[base][v] = info
   144→            else:
   145→                cached_venues.append(v)
   146→        if not cached_venues:
   147→            return out  # all cached
   148→        venues = cached_venues
   149→    else:
   150→        out = {}
   151→
   152→    def _fetch_one(venue: str) -> tuple[str, dict[str, dict[str, Any]]]:
   153→        fp = get_funding_provider(venue)
   154→        rows = fp.fetch_all("USDT")
   155→        imap = fp.fetch_interval_map("USDT")
   156→        by_base: dict[str, dict[str, Any]] = {}
   157→        for r in rows:
   158→            sym = str(r.get("symbol", "")).upper()
   159→            if not sym.endswith("USDT"):
   160→                continue
   161→            base = _base_from_symbol(sym)
   162→            if not base or base in SYMBOL_BLACKLIST:
   163→                continue
   164→            interval_h = float(imap.get(sym, 8.0) or 8.0)
   165→            next_ts = int(r.get("next_funding_ts", 0) or 0)
   166→            by_base[base] = {
   167→                "symbol": sym,
   168→                "rate_pct": float(r.get("rate_pct", 0.0)),
   169→                "interval_h": interval_h,
   170→                "next_funding_ts": next_ts,
   171→                "last_settle_ts": infer_last_settle_ts(next_ts, interval_h),
   172→                "mark_price": float(r.get("mark_price", 0.0) or 0.0),
   173→                "index_price": float(r.get("index_price", 0.0) or 0.0),
   174→            }
   175→        return venue, by_base
   176→
   177→    fetched = run_io_parallel(
   178→        venues, _fetch_one, max_workers=workers, swallow_errors=True, timeout=30.0
   179→    )
   180→    for venue, by_base in fetched.items():
   181→        # Store in cache
   182→        if cache_ttl_sec > 0:
   183→            _FETCH_ALL_CACHE[venue] = (time.time(), by_base)
   184→        for base, info in by_base.items():
   185→            if base not in out:
   186→                out[base] = {}
   187→            out[base][venue] = info
   188→    return out

   191→def _backfill_missing_settle_times(
   192→    by_base: dict[str, dict[str, dict[str, Any]]],
   193→    venues: list[str],
   194→    workers: int,
   195→) -> None:
   196→    """For entries with next_funding_ts==0 (Bitget bulk), backfill via per-symbol fetch_current."""
   197→    missing: list[tuple[str, str, str]] = []  # (venue, symbol, base)
   198→    for base, venue_map in by_base.items():
   199→        for venue in venues:
   200→            info = venue_map.get(venue)
   201→            if info and not info.get("next_funding_ts"):
   202→                missing.append((venue, info["symbol"], base))
   203→    if not missing:
   204→        return
   205→
   206→    def _fetch_one(
   207→        args: tuple[str, str, str],
   208→    ) -> tuple[tuple[str, str, str], dict[str, Any]]:
   209→        venue, symbol, _base = args
   210→        fp = get_funding_provider(venue)
   211→        snap = fp.fetch_current(symbol)
   212→        return args, snap
   213→
   214→    snap_map = run_io_parallel(
   215→        missing, _fetch_one, max_workers=workers, swallow_errors=True
   216→    )
   217→    for venue, symbol, base in missing:
   218→        snap = snap_map.get((venue, symbol, base))
   219→        if not isinstance(snap, dict):
   220→            continue
   221→        info = by_base[base][venue]
   222→        next_ts = int(snap.get("next_funding_ts", 0) or 0)
   223→        if next_ts > 0:
   224→            info["next_funding_ts"] = next_ts
   225→        last_ts = int(snap.get("last_settle_ts", 0) or 0)
   226→        if last_ts > 0:
   227→            info["last_settle_ts"] = last_ts
   228→        elif next_ts > 0:
   229→            info["last_settle_ts"] = infer_last_settle_ts(
   230→                next_ts, float(info.get("interval_h", 8.0) or 8.0)
   231→            )
   232→        idx = float(snap.get("index_price", 0) or 0)
   233→        if idx > 0:
   234→            info["index_price"] = idx
   235→        mark = float(snap.get("mark_price", 0) or 0)
   236→        if mark > 0 and not info.get("mark_price"):
   237→            info["mark_price"] = mark

   240→def _scan_spreads(
   241→    by_base: dict[str, dict[str, dict[str, Any]]],
   242→    min_spread: float,
   243→    min_edge: float,
   244→    fee_cache: dict[tuple[str, str], dict[str, float]] | None = None,
   245→    max_mark_spread_pct: float = 1.0,
   246→) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
   247→    """Compute pairwise perp-perp spreads across venues, return (forward, reverse)."""
   248→    forward: list[dict[str, Any]] = []
   249→    reverse: list[dict[str, Any]] = []
   250→
   251→    # Hoist now_ms out of the inner loop — funding timestamps resolve to
   252→    # seconds/minutes, so one timestamp per scan is identical in effect.
   253→    now_ms = int(time.time() * 1000)
   254→
   255→    for base, venue_map in by_base.items():
   256→        vlist = sorted(venue_map.keys())
   257→        for i, va in enumerate(vlist):
   258→            for vb in vlist[i + 1 :]:
   259→                ra = venue_map[va]
   260→                rb = venue_map[vb]
   261→                rate_a = float(ra["rate_pct"])
   262→                rate_b = float(rb["rate_pct"])
   263→                interval_a = float(ra.get("interval_h", 8.0) or 8.0)
   264→                interval_b = float(rb.get("interval_h", 8.0) or 8.0)
   265→
   266→                # Cheap pre-filter: raw rate spread before building model.
   267→                # This avoids ~2000 pair_pure_futures_spread calls for pairs
   268→                # that can't possibly clear min_spread.
   269→                raw_spread = abs(rate_a - rate_b)
   270→                if raw_spread < min_spread * 0.5:
   271→                    continue
   272→
   273→                # Perp-perp: short at higher rate (receives funding), long at lower (pays less).
   274→                if rate_a >= rate_b:
   275→                    short_venue, short_rate, short_info, short_interval = (
   276→                        va,
   277→                        rate_a,
   278→                        ra,
   279→                        interval_a,
   280→                    )
   281→                    long_venue, long_rate, long_info, long_interval = (
   282→                        vb,
   283→                        rate_b,
   284→                        rb,
   285→                        interval_b,
   286→                    )
   287→                else:
   288→                    short_venue, short_rate, short_info, short_interval = (
   289→                        vb,
   290→                        rate_b,
   291→                        rb,
   292→                        interval_b,
   293→                    )
   294→                    long_venue, long_rate, long_info, long_interval = (
   295→                        va,
   296→                        rate_a,
   297→                        ra,
   298→                        interval_a,
   299→                    )
   300→
   301→                now_ms_snapshot = now_ms  # use hoisted timestamp
   302→                model = pair_pure_futures_spread(
   303→                    long_rate_pct=long_rate,
   304→                    long_interval_h=long_interval,
   305→                    long_info=long_info,
   306→                    long_venue=long_venue,
   307→                    short_rate_pct=short_rate,
   308→                    short_interval_h=short_interval,
   309→                    short_info=short_info,
   310→                    short_venue=short_venue,
   311→                    now_ms=now_ms_snapshot,
   312→                )
   313→                is_mismatch = model["is_mismatch"]
   314→                long_blend = model["long_blend"]
   315→                short_blend = model["short_blend"]
   316→                eff_interval = model["eff_interval_h"]
   317→                spread = model["spread_pct"]
   318→                if spread < min_spread:
   319→                    continue
   320→
   321→                long_sym = str(long_info.get("symbol") or f"{base}USDT")
   322→                short_sym = str(short_info.get("symbol") or f"{base}USDT")
   323→                long_fee, short_fee, fee_pct = pair_open_taker_fee_pct(
   324→                    long_venue,
   325→                    long_sym,
   326→                    short_venue,
   327→                    short_sym,
   328→                    fee_cache=fee_cache,
   329→                )
   330→
   331→                net_edge = spread - fee_pct
   332→                if net_edge < min_edge:
   333→                    continue
   334→
   335→                # Categorize: forward (both positive or mixed), reverse (both negative)
   336→                if long_rate < 0 and short_rate < 0:
   337→                    direction = "reverse"
   338→                else:
   339→                    direction = "forward"
   340→
   341→                annual = _annual_from_rate(spread, eff_interval)
   342→
   343→                # Calculate mark price spread percentage
   344→                long_mark = float(long_info.get("mark_price", 0.0) or 0.0)
   345→                short_mark = float(short_info.get("mark_price", 0.0) or 0.0)
   346→                mark_spread_pct = (
   347→                    abs(long_mark - short_mark) / max(long_mark, short_mark) * 100.0
   348→                    if max(long_mark, short_mark) > 0
   349→                    else 0.0
   350→                )
   351→
   352→                # Skip if mark price spread is too large（conservative pass when mark is 0）
   353→                if mark_spread_pct > 0 and mark_spread_pct > max_mark_spread_pct:
   354→                    continue
   355→
   356→                # net_apy_pct = gross APY minus one-time cost (round-trip fees + entry basis)
   357→                one_time_cost = fee_pct * 2 + mark_spread_pct
   358→                net_apy = round(annual - one_time_cost, 1)
   359→
   360→                entry: dict[str, Any] = {
   361→                    "base": base,
   362→                    "direction": direction,
   363→                    "long_venue": long_venue,
   364→                    "short_venue": short_venue,
   365→                    "long_rate_pct": round(long_rate, 6),
   366→                    "short_rate_pct": round(short_rate, 6),
   367→                    "spread_pct": round(spread, 6),
   368→                    "long_fee_pct": round(long_fee, 4),
   369→                    "short_fee_pct": round(short_fee, 4),
   370→                    "fee_pct": round(fee_pct, 4),
   371→                    "round_trip_fee_pct": round(fee_pct * 2, 4),
   372→                    "net_edge_pct": round(net_edge, 6),
   373→                    "annual_apy_pct": round(annual, 1),
   374→                    "net_apy_pct": net_apy,
   375→                    "long_settle_ms": long_info.get("next_funding_ts", 0),
   376→                    "short_settle_ms": short_info.get("next_funding_ts", 0),
   377→                    "long_interval_h": float(long_info.get("interval_h", 8.0) or 8.0),
   378→                    "short_interval_h": float(short_info.get("interval_h", 8.0) or 8.0),
   379→                    "settle_mismatch": is_mismatch,
   380→                    "same_interval": not is_mismatch,
   381→                    "long_mark": long_mark,
   382→                    "short_mark": short_mark,
   383→                    "mark_spread_pct": round(mark_spread_pct, 6),
   384→                    # Basis-adjusted edge: net funding edge minus the cross-venue
   385→                    # mark divergence (a one-shot basis-risk discount). This is the
   386→                    # primary ranking metric — high funding edge with high mark
   387→                    # divergence (dislocated perp) is NOT a clean opportunity.
   388→                    "real_edge_pct": round(net_edge - mark_spread_pct, 6),
   389→                    "long_basis_pct": (
   390→                        round(long_blend["basis_pct"], 6)
   391→                        if long_blend.get("basis_pct") is not None
   392→                        else None
   393→                    ),
   394→                    "short_basis_pct": (
   395→                        round(short_blend["basis_pct"], 6)
   396→                        if short_blend.get("basis_pct") is not None
   397→                        else None
   398→                    ),
   399→                    "long_settle_progress": long_blend.get("settle_progress"),
   400→                    "short_settle_progress": short_blend.get("settle_progress"),
   401→                    "spread_source": model["spread_source"],
   402→                }
   403→
   404→                if direction == "forward":
   405→                    forward.append(entry)
   406→                else:
   407→                    reverse.append(entry)
   408→
   409→    # Rank by basis-adjusted real edge (not raw funding edge): a clean funding
   410→    # inefficiency beats a large edge that sits on a dislocated mark price.
   411→    forward.sort(key=lambda x: -x["real_edge_pct"])
   412→    reverse.sort(key=lambda x: -x["real_edge_pct"])
   413→    return forward, reverse

   416→def _enrich_with_depth(
   417→    entries: list[dict[str, Any]],
   418→    *,
   419→    top_n: int = DEFAULT_DEPTH_TOP_N,
   420→    window_pct: float = DEFAULT_DEPTH_WINDOW_PCT,
   421→    workers: int = DEFAULT_IO_WORKERS,
   422→) -> None:
   423→    """Add ``long_depth_usd`` / ``short_depth_usd`` / ``max_exec_usd`` to the top-N entries.
   424→
   425→    Mutates ``entries`` in place (each dict gets the four depth fields set;
   426→    entries beyond ``top_n`` get ``None`` placeholders for symmetry).
   427→
   428→    Caches (venue, base) → book so a venue/base combo shared by many pairs is
   429→    fetched at most once per scan. On fetch failure the entry's depth fields
   430→    stay ``None`` and ``depth_ok=False`` — the scan never blocks on a thin
   431→    or unresponsive book (same fail-open philosophy as ``check_pair_depth``).
   432→    """
   433→    # Rank by real_edge (the scan's primary ranking metric) and pick the top.
   434→    ranked = sorted(entries, key=lambda x: -x.get("real_edge_pct", -1e9))
   435→    targets = ranked[:top_n]
   436→
   437→    # Collect unique (venue, base) keys actually needed by the targets.
   438→    needed: set[tuple[str, str]] = set()
   439→    for e in targets:
   440→        needed.add((e["long_venue"], e["base"]))
   441→        needed.add((e["short_venue"], e["base"]))
   442→    if not needed:
   443→        for e in entries:
   444→            e.setdefault("long_depth_usd", None)
   445→            e.setdefault("short_depth_usd", None)
   446→            e.setdefault("max_exec_usd", None)
   447→            e.setdefault("depth_ok", None)
   448→        return
   449→
   450→    def _fetch_one(key: tuple[str, str]) -> tuple[tuple[str, str], dict | None]:
   451→        venue, base = key
   452→        try:
   453→            book = fetch_futures_depth(venue, base)
   454→        except Exception:
   455→            return key, None
   456→        long_usd = depth_usd_within(book, "asks", window_pct)
   457→        short_usd = depth_usd_within(book, "bids", window_pct)
   458→        return key, {"asks_usd": long_usd, "bids_usd": short_usd}
   459→
   460→    cache: dict[tuple[str, str], dict | None] = run_io_parallel(
   461→        list(needed),
   462→        _fetch_one,
   463→        max_workers=workers,
   464→        swallow_errors=True,
   465→    )
   466→
   467→    for e in entries:
   468→        long_d = cache.get((e["long_venue"], e["base"]))
   469→        short_d = cache.get((e["short_venue"], e["base"]))
   470→        if e in targets:
   471→            long_usd = long_d["asks_usd"] if long_d else None
   472→            short_usd = short_d["bids_usd"] if short_d else None
   473→            if long_usd is not None and short_usd is not None:
   474→                max_exec = min(long_usd, short_usd)
   475→                depth_ok = True
   476→            else:
   477→                max_exec = None
   478→                depth_ok = False
   479→            e["long_depth_usd"] = round(long_usd, 2) if long_usd is not None else None
   480→            e["short_depth_usd"] = (
   481→                round(short_usd, 2) if short_usd is not None else None
   482→            )
   483→            e["max_exec_usd"] = round(max_exec, 2) if max_exec is not None else None
   484→            e["depth_ok"] = depth_ok
   485→        else:
   486→            e.setdefault("long_depth_usd", None)
   487→            e.setdefault("short_depth_usd", None)
   488→            e.setdefault("max_exec_usd", None)
   489→            e.setdefault("depth_ok", None)

   493→    venues: list[str] | None = None,
   494→    min_spread: float = DEFAULT_MIN_SPREAD,
   495→    min_edge: float = DEFAULT_MIN_EDGE,

   498→    fee_policy: dict[str, Any] | None = None,
   499→    with_depth: bool = True,
   500→    depth_top_n: int = DEFAULT_DEPTH_TOP_N,
   501→    depth_window_pct: float = DEFAULT_DEPTH_WINDOW_PCT,
   502→) -> dict[str, Any]:
   503→    """Main entry point — scan and return structured results."""
   504→    if venues is None:
   505→        venues = ["binance", "bitget", "bybit", "okx"]
   506→
   507→    by_base = fetch_all_fee_rate_rows_by_base(venues, workers, cache_ttl_sec=30.0)
   508→    _backfill_missing_settle_times(by_base, venues, workers)
   509→    policy = parse_fee_policy(fee_policy)
   510→    fee_cache = build_policy_futures_cache(by_base, policy, workers=workers)
   511→
   512→    forward, reverse = _scan_spreads(
   513→        by_base, min_spread, min_edge, fee_cache, max_mark_spread_pct
   514→    )
   515→
   516→    # Liquidity context — only fetched for the top-N entries to bound latency.
   517→    # Disabled in unit tests / fast scans via with_depth=False.
   518→    if with_depth and (forward or reverse):
   519→        _enrich_with_depth(
   520→            forward + reverse,
   521→            top_n=depth_top_n,
   522→            window_pct=depth_window_pct,
   523→            workers=workers,
   524→        )
   525→
   526→    # Compute venue-level stats
   527→    venue_pairs: dict[str, int] = {}
   528→    for entry in forward + reverse:
   529→        key = f"{entry['long_venue']}↔{entry['short_venue']}"
   530→        venue_pairs[key] = venue_pairs.get(key, 0) + 1
   531→
   532→    return {
   533→        "venues": venues,
   534→        "total_assets_scanned": len(by_base),
   535→        "total_spreads_found": len(forward) + len(reverse),
   536→        "forward": forward,
   537→        "reverse": reverse,
   538→        "venue_pair_stats": sorted(
   539→            [{"pair": k, "count": v} for k, v in venue_pairs.items()],
   540→            key=lambda x: -x["count"],
   541→        ),
   542→        "timestamp": datetime.now(TZ_UTC).isoformat(),
   543→    }
