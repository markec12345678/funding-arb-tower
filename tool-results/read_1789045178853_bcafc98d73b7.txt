    92→def _safe_float(v: Any, default: float = 0.0) -> float:
    93→    try:
    94→        if v is None or v == "":
    95→            return default
    96→        return float(v)
    97→    except (TypeError, ValueError):
    98→        return default

   105→        from venues.http_util import http_get_json
   106→
   107→        coin_u = coin.upper()
   108→        url = f"https://api.bitget.com/api/v2/spot/public/coins?coin={coin_u}"
   109→        data = http_get_json(url, timeout=20)
   110→        rows = data.get("data") or []
   111→        if not rows:

   117→            out.append(
   118→                ChainRoute(
   119→                    venue=self.venue_id,
   120→                    coin=coin_u,
   121→                    native_chain=native,
   122→                    canonical=canon,
   123→                    withdraw_fee=_safe_float(ch.get("withdrawFee")),
   124→                    withdraw_fee_pct=_safe_float(ch.get("extraWithdrawFee")),
   125→                    min_withdraw=_safe_float(ch.get("minWithdrawAmount")),
   126→                    min_deposit=_safe_float(ch.get("minDepositAmount")),

   152→    def get_withdrawable_balance(self, coin: str) -> float:
   153→        from venues.bitget import BitgetSpotVenue
   154→
   155→        v = BitgetSpotVenue()
   156→        bals = v.fetch_usdt_account_balances() if coin.upper() == "USDT" else {}

   162→    def prepare_for_withdraw(self, coin: str, amount: float) -> list[str]:
   163→        from venues.bitget import BitgetSpotVenue
   164→
   165→        v = BitgetSpotVenue()
   166→        steps: list[str] = []
   167→        if coin.upper() != "USDT":
   168→            return steps
   169→        bals = v.fetch_usdt_account_balances()
   170→        spot = _safe_float(bals.get("spot"))
   171→        fut = _safe_float(bals.get("futures"))
   172→        if spot >= amount:
   173→            return steps
   174→        need = amount - spot
   175→        if fut <= 0:
   176→            return steps
   177→        xfer = min(fut, need) * 1.01
   178→        if hasattr(v, "transfer_asset") and v.transfer_asset(
   179→            "USDT", xfer, "futures", "spot"
   180→        ):
   181→            steps.append(f"bitget: futures→spot USDT {xfer:.4f}")
   182→        else:
   183→            steps.append(f"bitget: futures→spot USDT {xfer:.4f} FAILED")
   184→        return steps

   186→    def withdraw(
   187→        self,
   188→        coin: str,
   189→        amount: float,
   190→        native_chain_name: str,
   191→        address: str,
   192→        tag: str = "",
   193→    ) -> WithdrawResult:
   194→        from venues.bitget import _api_call
   195→
   196→        body: dict[str, Any] = {
   197→            "coin": coin.upper(),
   198→            "chain": native_chain_name,
   199→            "address": address,
   200→            "amount": f"{amount:.8f}".rstrip("0").rstrip("."),
   201→            "clientOid": f"w{int(time.time())}{uuid.uuid4().hex[:8]}",
   202→        }
   203→        if tag:
   204→            body["tag"] = tag
   205→        try:
   206→            data = _api_call("POST", "/api/v2/spot/wallet/withdrawal", body=body)
   207→            if str(data.get("code")) == "00000":
   208→                oid = str((data.get("data") or {}).get("orderId", ""))
   209→                return WithdrawResult(
   210→                    ok=True, order_id=oid, message="submitted", raw=data
   211→                )
   212→            return WithdrawResult(
   213→                ok=False, message=str(data.get("msg", data)), raw=data
   214→            )
   215→        except Exception as e:
   216→            return WithdrawResult(ok=False, message=str(e))

   219→        self,
   220→        coin: str,
   221→        since_ms: int = 0,
   222→        limit: int = 50,

   225→

   236→        out: list[dict[str, Any]] = []
   237→        for row in rows:

   241→            out.append(
   242→                {

   244→                    "coin": coin.upper(),

   246→                    "chain": str(row.get("chain", "")),
   247→                    "status": str(row.get("status", "")),
   248→                    "tx_id": str(row.get("tradeId", "") or row.get("txId", "")),

   475→    def prepare_for_withdraw(self, coin: str, amount: float) -> list[str]:
   476→        from venues.okx import _api_call
   477→
   478→        steps: list[str] = []
   479→        try:
   480→            bal = _api_call(
   481→                "GET", "/api/v5/account/balance", params={"ccy": coin.upper()}
   482→            )
   483→            details = (bal.get("data") or [{}])[0].get("details") or []
   484→            trading_avail = 0.0
   485→            for d in details:
   486→                if str(d.get("ccy", "")).upper() == coin.upper():
   487→                    trading_avail = _safe_float(d.get("availBal"))
   488→                    break
   489→            if trading_avail >= amount:
   490→                return steps
   491→            # Trading account -> Funding account (18=funding, 6=trading etc., OKX internal transfer)
   492→            xfer = amount - trading_avail + 0.01
   493→            _api_call(
   494→                "POST",
   495→                "/api/v5/asset/transfer",
   496→                body={
   497→                    "from": "18",
   498→                    "to": "18",
   499→                    "type": "0",
   500→                    "ccy": coin.upper(),
   501→                    "amt": f"{xfer:.8f}".rstrip("0").rstrip("."),
   502→                },
   503→            )
   504→            steps.append(f"okx: internal transfer {coin} {xfer:.4f} for withdraw")
   505→        except Exception as e:
   506→            steps.append(f"okx: prepare_for_withdraw skipped ({e})")
   507→        return steps

   702→_DEPOSIT_OK_STATUS = frozenset(
   703→    {
   704→        "success",
   705→        "1",
   706→        "2",
   707→        "completed",
   708→        "complete",
   709→        "confirmed",

   714→def _deposit_confirmed(row: dict[str, Any]) -> bool:
   715→    st = str(row.get("status", "")).lower()
   716→    if st in _DEPOSIT_OK_STATUS:
   717→        return True
   718→    if st.isdigit() and int(st) >= 1:
   719→        return True
   720→    return row.get("amount", 0) > 0 and not st

   723→def poll_deposit_until(
   724→    venue: str,
   725→    coin: str,
   726→    expected_amount: float,
   727→    since_ms: int,
   728→    *,
   729→    timeout_s: int = 600,
   730→    poll_interval_s: int = 15,
   731→    tolerance: float = 0.02,
   732→) -> tuple[bool, list[dict[str, Any]]]:
   733→    """Poll for deposit arrival. Returns (matched, records)."""
   734→    provider = get_transfer_provider(venue)
   735→    deadline = time.time() + timeout_s
   736→    seen: list[dict[str, Any]] = []
   737→    seen_ids: set[str] = set()
   738→    min_amt = max(0.0, expected_amount * (1.0 - tolerance))
   739→    while time.time() < deadline:
   740→        rows = provider.fetch_deposit_records(coin, since_ms=since_ms)
   741→        for row in rows:
   742→            rid = str(row.get("tx_id") or row.get("ts_ms") or "")
   743→            if rid and rid in seen_ids:
   744→                continue
   745→            if rid:
   746→                seen_ids.add(rid)
   747→            seen.append(row)
   748→            if not _deposit_confirmed(row):
   749→                continue
   750→            if _safe_float(row.get("amount")) >= min_amt:
   751→                return True, seen
   752→        time.sleep(poll_interval_s)
   753→    return False, seen
