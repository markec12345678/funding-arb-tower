    38→def _ensure_env() -> None:
    39→    global _env_loaded
    40→    if _env_loaded:
    41→        return
    42→    if os.environ.get("OKX_API_KEY"):
    43→        _env_loaded = True
    44→        return
    45→    try:
    46→        with open(CONFIG_PATH) as f:
    47→            for k, v in json.load(f).get("env", {}).items():
    48→                if v and k not in os.environ:
    49→                    os.environ[k] = str(v)
    50→    except (OSError, json.JSONDecodeError):
    51→        pass
    52→    _env_loaded = True

    70→def _sign(timestamp: str, method: str, path: str, body: str) -> str:
    71→    prehash = timestamp + method.upper() + path + body
    72→    return base64.b64encode(
    73→        hmac.new(_get_secret().encode(), prehash.encode(), hashlib.sha256).digest()
    74→    ).decode()

    77→def _api_call(
    78→    method: str, path: str, params: Optional[dict] = None, body: Optional[dict] = None
    79→) -> dict:
    80→    key, secret, passp = _get_key(), _get_secret(), _get_pass()
    81→    if not (key and secret and passp):
    82→        raise RuntimeError(
    83→            "OKX API credentials missing: please set OKX_API_KEY / OKX_SECRET_KEY / OKX_PASSPHRASE, "
    84→            "or configure them in ~/.funding-arb/credentials.json under env."
    85→        )
    86→
    87→    query = urllib.parse.urlencode(params) if params else ""
    88→    body_str = json.dumps(body) if body else ""
    89→    full_path = path + ("?" + query if query else "")
    90→
    91→    retries = 3 if method == "GET" else 1
    92→    last_err: Optional[Exception] = None
    93→    for attempt in range(retries):
    94→        ts = time.strftime("%Y-%m-%dT%H:%M:%S.000Z", time.gmtime())
    95→        sig = _sign(ts, method, full_path, body_str)
    96→        headers = {
    97→            "OK-ACCESS-KEY": key,
    98→            "OK-ACCESS-SIGN": sig,
    99→            "OK-ACCESS-TIMESTAMP": ts,
   100→            "OK-ACCESS-PASSPHRASE": passp,
   101→            "Content-Type": "application/json",
   102→            "User-Agent": "Mozilla/5.0 (compatible; funding-arb/1.0)",
   103→        }
   104→        url = BASE + full_path
   105→        req = urllib.request.Request(url, headers=headers, method=method)
   106→        if body_str:
   107→            req.data = body_str.encode()
   108→        try:
   109→            with urllib.request.urlopen(req, timeout=15) as resp:
   110→                data = json.loads(resp.read().decode())
   111→            if data.get("code") != "0":
   112→                raise RuntimeError(
   113→                    f"OKX API error: {data.get('msg', data.get('code'))}"
   114→                )
   115→            return data
   116→        except Exception as e:
   117→            last_err = e
   118→            if method == "GET" and attempt < retries - 1:
   119→                time.sleep(0.5 * (attempt + 1))
   120→                continue
   121→            raise
   122→    raise last_err if last_err else RuntimeError("okx _api_call failed")

   129→def _sz_precision(val: float) -> int:
   130→    """Decimal places for lotSz/tickSz; avoids str(1e-06) having no decimal point, which would give precision 0."""
   131→    s = f"{val:.12f}".rstrip("0")
   132→    if "." in s:
   133→        return len(s.split(".")[1])
   134→    return 0

   161→    def get_ticker(self, pair: str) -> float:
   162→        inst = pair
   163→        url = f"{BASE}/api/v5/market/ticker?instId={inst}"
   164→        try:
   165→            data = http_get_json(url)
   166→            return float(data.get("data", [{}])[0].get("last", 0))
   167→        except Exception:
   168→            return 0.0

   170→    def get_futures_ticker(self, pair: str) -> float:
   171→        """Perpetual last price. pair can be BTCUSDT (auto-converted to BTC-USDT-SWAP) or a full instId."""
   172→        inst = pair
   173→        if "-" not in pair:
   174→            # BTCUSDT → BTC-USDT-SWAP
   175→            base = pair.replace("USDT", "")
   176→            if base:
   177→                inst = f"{base}-USDT-SWAP"
   178→        url = f"{BASE}/api/v5/market/ticker?instId={inst}"
   179→        try:
   180→            data = http_get_json(url)
   181→            return float(data.get("data", [{}])[0].get("last", 0))
   182→        except Exception:
   183→            return 0.0

   185→    def get_all_futures_tickers(self, cache_sec: int = 5) -> dict[str, float]:
   186→        """Bulk SWAP last prices {BTCUSDT: price, BTC-USDT-SWAP: price}. Cached briefly."""
   187→        global _futures_ticker_loaded_at, _futures_ticker_prices
   188→        now = time.time()
   189→        if _futures_ticker_prices and (now - _futures_ticker_loaded_at) < cache_sec:
   190→            return dict(_futures_ticker_prices)
   191→        try:
   192→            rows = http_get_json(f"{BASE}/api/v5/market/tickers?instType=SWAP").get(
   193→                "data", []
   194→            )
   195→            new_prices: dict[str, float] = {}
   196→            for r in rows:
   197→                inst_id = str(r.get("instId", "")).upper()
   198→                last = float(r.get("last", 0) or 0)
   199→                if not inst_id or last <= 0:
   200→                    continue
   201→                # Map both formats: BTC-USDT-SWAP → BTCUSDT
   202→                new_prices[inst_id] = last  # e.g. "BTC-USDT-SWAP"
   203→                if "-USDT-SWAP" in inst_id:
   204→                    base = inst_id.replace("-USDT-SWAP", "")
   205→                    new_prices[f"{base}USDT"] = last
   206→            _futures_ticker_prices = new_prices
   207→            _futures_ticker_loaded_at = now
   208→        except Exception:
   209→            pass
   210→        return dict(_futures_ticker_prices)

   212→    def get_klines(
   213→        self, pair: str, granularity: str = "1day", limit: int = 200
   214→    ) -> list:
   215→        bar = self.GRANULARITY_MAP.get(granularity, granularity)
   216→        url = f"{BASE}/api/v5/market/candles?instId={pair}&bar={bar}&limit={limit}"
   217→        try:
   218→            data = http_get_json(url)
   219→            rows = data.get("data", [])
   220→            return list(reversed(rows))
   221→        except Exception:
   222→            return []

   224→    def fetch_symbol_rules(
   225→        self, pair: str, cache_sec: int = 3600
   226→    ) -> Optional[dict[str, Any]]:
   227→        now = time.time()
   228→        cached = _symbol_rules_cache.get(pair)
   229→        if cached and (now - cached[0]) < cache_sec:
   230→            return dict(cached[1])
   231→        url = f"{BASE}/api/v5/public/instruments?instType=SPOT&instId={pair}"
   232→        try:
   233→            payload = http_get_json(url)
   234→        except Exception:
   235→            return dict(cached[1]) if cached else None
   236→        rows = payload.get("data", [])
   237→        if not rows:
   238→            return dict(cached[1]) if cached else None
   239→        info = rows[0]
   240→        lot_sz = float(info.get("lotSz", "0.00000001"))
   241→        min_sz = float(info.get("minSz", "0"))
   242→        tick_sz = float(info.get("tickSz", "0.01"))
   243→        qty_prec = _sz_precision(lot_sz)
   244→        quote_prec = _sz_precision(tick_sz) or 2
   245→        rules = {
   246→            "symbol": pair,
   247→            "min_trade_usdt": 0,
   248→            "min_trade_base": max(min_sz, lot_sz),
   249→            "quantity_precision": qty_prec,
   250→            "quote_precision": quote_prec,
   251→            "status": "live" if info.get("state") == "live" else "",
   252→        }
   253→        _symbol_rules_cache[pair] = (now, rules)
   254→        return dict(rules)
   255→
   256→    def fetch_futures_symbol_rules(
   257→        self, pair: str, cache_sec: int = 3600
   258→    ) -> dict[str, Any] | None:
   259→        now = time.time()
   260→        cached = _futures_rules_cache.get(pair)
   261→        if cached and (now - cached[0]) < cache_sec:
   262→            return dict(cached[1])
   263→        asset = pair.replace("USDT", "") if pair.endswith("USDT") else pair
   264→        swap = _swap_inst_id(asset)
   265→        url = f"{BASE}/api/v5/public/instruments?instType=SWAP&instId={swap}"
   266→        try:
   267→            payload = http_get_json(url)
   268→        except Exception:
   269→            return dict(cached[1]) if cached else None
   270→        rows = payload.get("data", [])
   271→        if not rows:
   272→            return dict(cached[1]) if cached else None
   273→        info = rows[0]
   274→        lot_sz = float(info.get("lotSz", "1"))
   275→        min_sz = float(info.get("minSz", "1"))
   276→        ct_val = float(info.get("ctVal", "1"))
   277→        tick_sz = float(info.get("tickSz", "0.01"))
   278→        qty_prec = _sz_precision(lot_sz)
   279→        rules = {
   280→            "symbol": pair,
   281→            "min_trade_usdt": 0,
   282→            "min_trade_base": max(min_sz * ct_val, lot_sz),
   283→            "quantity_precision": qty_prec,
   284→            "quote_precision": _sz_precision(tick_sz) or 2,
   285→            "ct_val": ct_val,
   286→            "status": "live" if info.get("state") == "live" else "",
   287→        }
   288→        _futures_rules_cache[pair] = (now, rules)
   289→        return dict(rules)

   291→    def _get_account_config(self, cache_sec: int = 300) -> dict[str, Any]:
   292→        """GET /api/v5/account/config — acctLv / autoLoan / enableSpotBorrow etc. (with TTL cache)."""
   293→        global _acct_config_cache
   294→        now = time.time()
   295→        if _acct_config_cache and (now - _acct_config_cache[0]) < cache_sec:
   296→            return dict(_acct_config_cache[1])
   297→        try:
   298→            data = _api_call("GET", "/api/v5/account/config")
   299→            rows = data.get("data") or []
   300→            cfg = rows[0] if rows else {}
   301→        except Exception:
   302→            return {}
   303→        if cfg:
   304→            _acct_config_cache = (now, cfg)
   305→        return dict(cfg)

   307→    def _is_unified_margin_account(self, cfg: dict[str, Any] | None = None) -> bool:
   308→        """Under multi-currency/portfolio margin mode, spot+margin+derivatives share the trading account; no separate transfer needed."""

   312→    def transfer_asset(
   313→        self, asset: str, amount: float, from_account: str, to_account: str
   314→    ) -> bool:
   315→        # Under all OKX account modes, margin lives in the trading account (18); spot↔margin needs no transfer
   316→        if "margin" in (from_account, to_account):
   317→            return True
   318→
   319→        if from_account == "spot" and to_account == "futures":
   320→            okx_from, okx_to = "18", "27"
   321→        elif from_account == "futures" and to_account == "spot":
   322→            okx_from, okx_to = "27", "18"
   323→        else:
   324→            return False
   325→        try:
   326→            _api_call(
   327→                "POST",
   328→                "/api/v5/asset/transfer",
   329→                body={
   330→                    "from": okx_from,
   331→                    "to": okx_to,
   332→                    "currency": asset.upper(),
   333→                    "amount": f"{amount:.8f}".rstrip("0").rstrip("."),
   334→                },
   335→            )
   336→            return True
   337→        except Exception:
   338→            return False

   340→    def fetch_asset_market(
   341→        self, asset: str, quote: str = "USDT", cfg: dict[str, Any] | None = None
   342→    ) -> dict[str, Any]:
   343→        pair = _inst_id(asset, quote)
   344→        price = self.get_ticker(pair)
   345→        rules = self.fetch_symbol_rules(pair)
   346→        if rules is None:
   347→            return {
   348→                "symbol": asset,
   349→                "price": price,
   350→                "rules_error": True,
   351→                "venue": self.venue_id,
   352→            }
   353→        limits = rules_for_price(rules, price)
   354→        cfg = cfg or {}
   355→        tf = resolve_timeframes(cfg)
   356→        klines_1d = [
   357→            parse_kline_ohlcv(k)
   358→            for k in self.get_klines(pair, tf["slow"]["interval"], tf["slow"]["limit"])
   359→            if k
   360→        ]
   361→        klines_4h = [
   362→            parse_kline_ohlcv(k)
   363→            for k in self.get_klines(pair, tf["mid"]["interval"], tf["mid"]["limit"])
   364→            if k
   365→        ]
   366→        klines_1w = [
   367→            parse_kline_ohlcv(k)
   368→            for k in self.get_klines(
   369→                pair, tf["macro"]["interval"], tf["macro"]["limit"]
   370→            )
   371→            if k
   372→        ]
   373→        return {
   374→            "symbol": asset,
   375→            "pair": pair,
   376→            "price": price,
   377→            "rules_error": False,
   378→            "venue": self.venue_id,
   379→            "symbol_rules": rules,
   380→            **limits,
   381→            "klines_1d": klines_1d,
   382→            "klines_4h": klines_4h,
   383→            "klines_1w": klines_1w,
   384→        }

   386→    def fetch_balances(self, coins: list[str]) -> dict[str, float]:
   387→        balances: dict[str, float] = {c: 0.0 for c in coins}
   388→        data = _api_call("GET", "/api/v5/account/balance", params={"ccy": ""})
   389→        for detail in data.get("data", []):
   390→            for item in detail.get("details", []):
   391→                coin = str(item.get("ccy", "")).upper()
   392→                if coin in balances:
   393→                    balances[coin] = float(item.get("availBal", item.get("cashBal", 0)))
   394→        return balances

   396→    def fetch_live_state(self, assets: list[str]) -> dict[str, Any]:
   397→        balances = self.fetch_balances(assets)
   398→
   399→        positions: dict[str, dict[str, Any]] = {}
   400→        try:
   401→            pos_data = _api_call("GET", "/api/v5/account/positions")
   402→            for pos in pos_data.get("data", []):
   403→                inst = str(pos.get("instId", ""))
   404→                if not inst.endswith("-SWAP"):
   405→                    continue
   406→                base = inst.split("-")[0]
   407→                amt = float(pos.get("pos", 0))
   408→                if abs(amt) < 1e-9:
   409→                    continue
   410→                positions[base] = {
   411→                    "amount": abs(amt),
   412→                    "side": "long" if amt > 0 else "short",
   413→                    "entry_price": float(pos.get("avgPx", 0) or 0),
   414→                    "unrealized_pnl": float(pos.get("upl", 0) or 0),
   415→                    "leverage": float(pos.get("lever", 1) or 1),
   416→                }
   417→        except Exception:
   418→            pass
   419→
   420→        return {"balances": balances, "futures_positions": positions}

   422→    def fetch_futures_positions(self, quote: str = "USDT") -> list[dict[str, Any]]:
   423→        """USDT perpetual position list (single endpoint, raises on failure)."""
   424→        pos_data = _api_call("GET", "/api/v5/account/positions")
   425→        quote_u = quote.upper()
   426→        out: list[dict[str, Any]] = []
   427→        for pos in pos_data.get("data", []) or []:
   428→            inst = str(pos.get("instId", ""))
   429→            if not inst.endswith(f"-{quote_u}-SWAP"):
   430→                continue
   431→            amt = float(pos.get("pos", 0) or 0)
   432→            if abs(amt) <= 1e-12:
   433→                continue
   434→            base = inst.split("-")[0]
   435→            out.append(
   436→                {
   437→                    "symbol": f"{base}{quote_u}",
   438→                    "side": "long" if amt > 0 else "short",
   439→                    "qty": abs(amt),
   440→                    "entry_price": float(pos.get("avgPx", 0) or 0),
   441→                    "liq_price": float(pos.get("liqPx", 0) or 0),
   442→                    "leverage": float(pos.get("lever", 1) or 1),
   443→                    "unrealized_pnl": float(pos.get("upl", 0) or 0),
   444→                }
   445→            )
   446→        return out

   475→    # ── cross margin (Reverse C&C: borrow-sell / buy-repay) ──────────────────────

   477→    def _quote_ccy(self, pair: str) -> str:
   478→        if "-" in pair:
   479→            return pair.split("-")[-1].upper()
   480→        if pair.endswith("USDT"):
   481→            return "USDT"
   482→        return "USDT"
   483→
   484→    def _base_ccy(self, pair: str) -> str:
   485→        if "-" in pair:
   486→            return pair.split("-")[0].upper()
   487→        if pair.endswith("USDT"):
   488→            return pair[:-4].upper()
   489→        return pair.upper()
   490→
   491→    def _can_borrow_for_sell(self, pair: str) -> bool:
   492→        """Simple account mode: pair-level cross margin, supports reverse if max-loan side=sell has available borrow."""
   493→        try:
   494→            data = _api_call(
   495→                "GET",
   496→                "/api/v5/account/max-loan",
   497→                params={
   498→                    "instId": pair,
   499→                    "mgnMode": "cross",
   500→                    "mgnCcy": self._quote_ccy(pair),
   501→                },
   502→            )
   503→            base = self._base_ccy(pair)
   504→            for row in data.get("data", []):
   505→                if str(row.get("ccy", "")).upper() != base:
   506→                    continue
   507→                if str(row.get("side", "")).lower() == "sell":
   508→                    return float(row.get("maxLoan", 0) or 0) > 0
   509→        except Exception:
   510→            return False
   511→        return False

   513→    def supports_reverse_arbitrage(self) -> bool:
   514→        """Reverse spot leg capability: tests via actual max-loan / borrow toggle rather than inferring from acctLv alone.
   515→
   516→        Simple mode acctLv=2: pair-level cross margin orders implicitly borrow (consistent with web margin);
   517→        Multi-currency acctLv=3/4: account-level autoLoan oversell; Spot+enableSpotBorrow also supported.
   518→        Assumes the code path works when no API keys are present (for paper/dry-run scanning).
   519→        """
   520→        if not (_get_key() and _get_secret()):
   521→            return True
   522→        cfg = self._get_account_config()
   523→        if not cfg:
   524→            return False
   525→        acct_lv = str(cfg.get("acctLv", ""))
   526→        if acct_lv in ("3", "4"):
   527→            return True
   528→        if acct_lv == "1" and cfg.get("enableSpotBorrow"):
   529→            return True
   530→        if acct_lv == "2":
   531→            return self._can_borrow_for_sell(_inst_id("ETH", "USDT"))
   532→        return False

   552→    def fetch_margin_debt(self, assets: list[str]) -> dict[str, float]:
   553→        """Per-coin liabilities: balance.liab (Spot/Multi-currency/Portfolio) + MARGIN position liab (Simple mode)."""
   554→        debt: dict[str, float] = {a.upper(): 0.0 for a in assets}
   555→        try:
   556→            data = _api_call("GET", "/api/v5/account/balance", params={"ccy": ""})
   557→            for acct in data.get("data", []):
   558→                for item in acct.get("details", []):
   559→                    coin = str(item.get("ccy", "")).upper()
   560→                    if coin in debt:
   561→                        debt[coin] = abs(float(item.get("liab", 0) or 0))
   562→        except Exception as e:
   563→            print(f"okx fetch_margin_debt balance failed: {e}", file=sys.stderr)
   564→        # In Simple mode, margin liabilities are tracked on MARGIN position's liab/liabCcy
   565→        try:
   566→            pos = _api_call(
   567→                "GET", "/api/v5/account/positions", params={"instType": "MARGIN"}
   568→            )
   569→            for p in pos.get("data", []):
   570→                coin = str(p.get("liabCcy", "")).upper()
   571→                liab = abs(float(p.get("liab", 0) or 0))
   572→                if coin in debt:
   573→                    debt[coin] += liab
   574→        except Exception as e:
   575→            print(f"okx fetch_margin_debt positions failed: {e}", file=sys.stderr)
   576→        return debt

   618→    def place_margin_order(
   619→        self,
   620→        pair: str,
   621→        side: str,
   622→        amount_base: float,
   623→        quantity_precision: int = 6,
   624→        ref_price: float = 0.0,
   625→        side_effect: str = "",
   626→    ) -> tuple[bool, dict[str, Any]]:
   627→        """Cross margin order (tdMode=cross).
   628→
   629→        acctLv=2 Simple mode: requires ccy=margin coin; sell orders implicitly borrow, buy closing automatically repays principal and interest.
   630→        acctLv=3/4: depends on set-auto-loan / set-auto-repay.
   631→        sell uses market order (sz=base); buy uses IOC limit order (px with 1% buffer), because
   632→        MARGIN market buy has ambiguous sz unit and tgtCcy only applies to SPOT market orders.
   633→        """
   634→        cfg = self._get_account_config()
   635→        acct_lv = str(cfg.get("acctLv", ""))
   636→        se = side_effect.lower()
   637→        if se and acct_lv in ("3", "4"):
   638→            self._ensure_margin_auto_flags(side_effect)
   639→
   640→        client_oid = f"qmgn{int(time.time())}{random.randint(0, 9999)}"
   641→        sz = f"{amount_base:.{quantity_precision}f}".rstrip("0").rstrip(".")
   642→        body: dict[str, Any] = {
   643→            "instId": pair,
   644→            "tdMode": "cross",
   645→            "side": side,
   646→            "ordType": "market",
   647→            "sz": sz or f"{amount_base:.{quantity_precision}f}",
   648→            "clOrdId": client_oid,
   649→        }
   650→        if acct_lv == "2":
   651→            body["ccy"] = self._quote_ccy(pair)
   652→        if side == "buy":
   653→            # IOC limit with 1% buffer: precise base quantity control, effectively market execution
   654→            px_ref = ref_price if ref_price and ref_price > 0 else self.get_ticker(pair)
   655→            if not px_ref or px_ref <= 0:
   656→                return False, {
   657→                    "error": "okx margin buy requires ref_price for IOC limit"
   658→                }
   659→            rules = self.fetch_symbol_rules(pair) or {}
   660→            qp = int(rules.get("quote_precision", 4))
   661→            body["ordType"] = "ioc"
   662→            body["px"] = f"{px_ref * 1.01:.{qp}f}"
   663→            if se == "auto_repay" and acct_lv in ("2", "3"):
   664→                body["reduceOnly"] = True
   665→        submit_ts = time.time()
   666→        try:
   667→            result = _api_call("POST", "/api/v5/trade/order", body=body)
   668→            fill_ts = time.time()
   669→            order_id = result.get("data", [{}])[0].get("ordId", "?")
   670→            time.sleep(0.3)
   671→            detail = _api_call(
   672→                "GET", "/api/v5/trade/order", params={"instId": pair, "ordId": order_id}
   673→            )
   674→            od = detail.get("data", [{}])[0]
   675→            state = str(od.get("state", ""))
   676→            exec_price = float(od.get("avgPx", 0) or ref_price)
   677→            fill_sz = float(od.get("accFillSz", 0) or od.get("fillSz", 0) or 0)
   678→            if state == "filled" and fill_sz <= 0:
   679→                fill_sz = amount_base
   680→            if fill_sz <= 0:
   681→                return False, {
   682→                    "error": f"okx margin {side} {pair} not filled (state={state or '?'})",
   683→                    "order_id": order_id,
   684→                    "order_status": state,
   685→                }
   686→            exec_qty = fill_sz
   687→            exec_quote = exec_qty * exec_price
   688→            slippage = (
   689→                round((exec_price - ref_price) / ref_price, 6)
   690→                if ref_price and exec_price
   691→                else None
   692→            )
   693→            return True, {
   694→                "order_id": order_id,
   695→                "exec_price": exec_price,
   696→                "exec_qty": exec_qty,
   697→                "exec_quote_usd": exec_quote,
   698→                "ref_price": ref_price,
   699→                "slippage": slippage,
   700→                "submit_ts": round(submit_ts, 3),
   701→                "fill_ts": round(fill_ts, 3),
   702→                "latency_ms": round((fill_ts - submit_ts) * 1000),
   703→                "order_status": state,
   704→            }
   705→        except Exception as e:
   706→            return False, {"error": str(e)}

   895→    def execute_trades(
   896→        self,
   897→        trades: list[dict[str, Any]],
   898→        market: dict[str, dict[str, Any]],
   899→        dry_run: bool,
   900→    ) -> list[dict[str, Any]]:
   901→        results: list[dict[str, Any]] = []
   902→        for trade in trades:
   903→            symbol = trade["symbol"]
   904→            mkt = market.get(symbol, {})
   905→            pair = str(mkt.get("pair") or _inst_id(symbol, "USDT"))
   906→            ref_price = float(mkt.get("price", 0))
   907→            record = dict(trade)
   908→            record["dry_run"] = dry_run
   909→            record["ref_price"] = ref_price
   910→            record["venue"] = self.venue_id
   911→            if dry_run:
   912→                record["status"] = "simulated"
   913→                record["order_id"] = None
   914→                record["slippage"] = 0.0
   915→                record["latency_ms"] = 0
   916→                results.append(record)
   917→                continue
   918→            is_margin = str(trade.get("account", "")).lower() == "margin"
   919→            if trade["type"] in ("buy", "sell") and is_margin:
   920→                ok, detail = self.place_margin_order(
   921→                    pair,
   922→                    trade["type"],
   923→                    trade["amount_base"],
   924→                    int(mkt.get("quantity_precision", 6)),
   925→                    ref_price=ref_price,
   926→                    side_effect=str(trade.get("side_effect", "")),
   927→                )
   928→            elif trade["type"] == "buy":
   929→                ok, detail = self.place_buy(
   930→                    pair,
   931→                    trade["amount_usdt"],
   932→                    int(mkt.get("quote_precision", 2)),
   933→                    ref_price=ref_price,
   934→                )
   935→            elif trade["type"] == "sell":
   936→                ok, detail = self.place_sell(
   937→                    pair,
   938→                    trade["amount_base"],
   939→                    int(mkt.get("quantity_precision", 6)),
   940→                    ref_price=ref_price,
   941→                )
   942→            elif trade["type"] in (
   943→                "open_short",
   944→                "close_long",
   945→                "close_short",
   946→                "open_long",
   947→            ):
   948→                ok, detail = self.place_futures_order(
   949→                    pair,
   950→                    trade["type"],
   951→                    trade["amount_base"],
   952→                    int(
   953→                        trade.get("quantity_precision")
   954→                        or mkt.get("quantity_precision", 3)
   955→                    ),
   956→                    ref_price=ref_price,
   957→                )
   958→            else:
   959→                ok, detail = False, {"error": f"Unknown trade type {trade['type']}"}
   960→            record["status"] = "filled" if ok else "failed"
   961→            if ok:
   962→                record.update(
   963→                    {
   964→                        "order_id": detail.get("order_id"),
   965→                        "exec_price": detail.get("exec_price"),
   966→                        "exec_qty": detail.get("exec_qty"),
   967→                        "exec_quote_usd": detail.get("exec_quote_usd"),
   968→                        "slippage": detail.get("slippage"),
   969→                        "latency_ms": detail.get("latency_ms"),
   970→                        "submit_ts": detail.get("submit_ts"),
   971→                        "fill_ts": detail.get("fill_ts"),
   972→                        "order_status": detail.get("order_status"),
   973→                        "error": None,
   974→                    }
   975→                )
   976→            else:
   977→                record["order_id"] = None
   978→                record["error"] = detail.get("error", str(detail))
   979→            results.append(record)
   980→        return results
