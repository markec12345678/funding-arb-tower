   104→def venue_trade_capability(venue_id: str) -> tuple[bool, str]:
   105→    """Whether the executor can route orders (incl. dry-run) for this venue.
   106→
   107→    Returns (trade_capable, reason_if_not).
   108→    """
   109→    v = str(venue_id).strip().lower()
   110→    if v in ("binance", "bitget", "bybit", "okx", "hyperliquid", "aster"):
   111→        return True, ""
   112→    if v == "lighter":
   113→        from importlib.util import find_spec
   114→
   115→        if find_spec("lighter") is not None:
   116→            return True, ""
   117→        return False, "lighter-sdk not installed (pip install lighter-sdk)"
   118→    if v == "edgex":
   119→        from importlib.util import find_spec
   120→
   121→        if find_spec("edgex_sdk") is not None:
   122→            return True, ""
   123→        return (
   124→            False,
   125→            "edgex-python-sdk not installed (pip install edgex-python-sdk>=2.0.0)",
   126→        )
   127→    if v == "dydx":
   128→        from importlib.util import find_spec
   129→
   130→        if find_spec("dydx_v4_client") is not None:
   131→            return True, ""
   132→        return False, "dydx-v4-client not installed (pip install dydx-v4-client>=1.1.5)"
   133→    return False, f"unknown venue {v!r}"
   134→
   135→
   136→def venue_live_ready(venue_id: str) -> tuple[bool, str]:
   137→    """Whether LIVE (non-dry-run) orders can actually be signed and submitted."""
   138→    v = str(venue_id).strip().lower()
   139→    capable, reason = venue_trade_capability(v)
   140→    if not capable:
   141→        return False, reason
   142→    info = VENUES.get(v, {})
   143→    keys = info.get("trade_keys") or info.get("required_keys") or []
   144→    missing = [k for k in keys if not os.environ.get(k)]
   145→    if missing:
   146→        return False, f"missing credentials: {', '.join(missing)}"
   147→    if v == "hyperliquid" and not _HL_SKILL_DIR.exists():
   148→        return False, "sibling ../hyperliquid repo not found (needed for order signing)"
   149→    return True, ""

   243→@router.get("/settings/venues")
   244→async def get_venues():
   245→    """Return venue configuration and connection status."""
   246→    result = []
   247→    for venue_id, info in VENUES.items():
   248→        # Check if credentials are configured (no keys required = public data venue)
   249→        required = info.get("required_keys", [])
   250→        configured = not required
   251→        missing_keys: list[str] = []
   252→        for key in required:
   253→            val = os.environ.get(key)
   254→            if val:
   255→                configured = True
   256→            else:
   257→                missing_keys.append(key)
   258→
   259→        # Also check trade keys
   260→        for key in info.get("trade_keys", []):
   261→            if os.environ.get(key):
   262→                configured = True
   263→
   264→        trade_capable, trade_reason = venue_trade_capability(venue_id)
   265→        live_ready, live_reason = venue_live_ready(venue_id)
   266→        result.append(
   267→            {
   268→                "id": venue_id,
   269→                "name": info["name"],
   270→                "type": info["type"],
   271→                "configured": configured,
   272→                "missing_keys": missing_keys if not configured else [],
   273→                "status": "connected" if configured else "not_configured",
   274→                "scan_capable": True,
   275→                "trade_capable": trade_capable,
   276→                "trade_reason": trade_reason,
   277→                "live_ready": live_ready,
   278→                "live_reason": live_reason,
   279→            }
   280→        )
   281→
   282→    return {"success": True, "data": result}

   285→@router.get("/settings/credentials/status")
   286→async def credentials_status():
   287→    """Return credential backend status and venue configuration."""
   288→    backends: dict[str, Any] = {}
   289→
   290→    # Check keyring
   291→    import importlib.util
   292→
   293→    if importlib.util.find_spec("keyring") is not None:
   294→        backends["keyring"] = {
   295→            "available": True,
   296→            "description": "macOS Keychain / System Key Management",
   297→        }
   298→    else:
   299→        backends["keyring"] = {
   300→            "available": False,
   301→            "description": "keyring package not installed",
   302→        }
   303→
   304→    # Check systemd-creds
   305→    import shutil
   306→    import sys
   307→
   308→    if sys.platform == "linux" and shutil.which("systemd-creds"):
   309→        backends["systemd_creds"] = {
   310→            "available": True,
   311→            "description": "systemd-creds (Linux TPM2)",
   312→        }
   313→    else:
   314→        backends["systemd_creds"] = {"available": False, "description": "Linux only"}
   315→
   316→    # Check age
   317→    if shutil.which("age"):
   318→        backends["age"] = {"available": True, "description": "age encrypted file"}
   319→    else:
   320→        backends["age"] = {"available": False, "description": "age tool not installed"}
   321→
   322→    # Check plaintext JSON
   323→    from pathlib import Path
   324→
   325→    json_path = Path.home() / ".funding-arb" / "credentials.json"
   326→    backends["plaintext_json"] = {
   327→        "available": json_path.exists(),
   328→        "description": "Plaintext JSON (fallback)",
   329→        "path": str(json_path),
   330→    }
   331→
   332→    # Determine which venues have credentials
   333→    venues_configured: list[str] = []
   334→    venues_missing: list[str] = []
   335→    for venue_id, info in VENUES.items():
   336→        required = info.get("required_keys", [])
   337→        has_key = not required or any(os.environ.get(k) for k in required)
   338→        if not has_key:
   339→            has_key = any(os.environ.get(k) for k in info.get("trade_keys", []))
   340→        if has_key:
   341→            venues_configured.append(venue_id)
   342→        else:
   343→            venues_missing.append(venue_id)
   344→
   345→    return {
   346→        "success": True,
   347→        "data": {
   348→            "backends": backends,
   349→            "venues_configured": venues_configured,
   350→            "venues_missing": venues_missing,
   351→        },
   352→    }

   386→@router.get("/settings/fees")
   387→async def get_resolved_fees():
   388→    """Return resolved spot/futures taker fees per venue (API or VIP tier)."""
   389→    from core.fee_providers import (  # noqa: E402
   390→        parse_fee_policy,
   391→        resolve_venue_fee,
   392→        venue_has_credentials,
   393→        venue_uses_api,
   394→    )
   395→    from core.vip_fee_tiers import ALL_VENUES  # noqa: E402
   396→
   397→    policy = parse_fee_policy(_strategy_config)
   398→    venues_out: dict[str, Any] = {}
   399→    for v in ALL_VENUES:
   400→        uses_api = venue_uses_api(v, policy)
   401→        spot = resolve_venue_fee(v, leg="spot", policy=policy)
   402→        futures = resolve_venue_fee(v, leg="futures", policy=policy)
   403→        tier = policy.get("venue_tiers", {}).get(v)
   404→        if not tier:
   405→            tier = spot.get("tier") or futures.get("tier")
   406→        venues_out[v] = {
   407→            "has_credentials": venue_has_credentials(v),
   408→            "uses_api": uses_api,
   409→            "tier": tier,
   410→            "spot_taker_pct": spot["taker_pct"],
   411→            "futures_taker_pct": futures["taker_pct"],
   412→            "spot_source": spot["source"],
   413→            "futures_source": futures["source"],
   414→        }
   415→
   416→    return {
   417→        "success": True,
   418→        "data": {
   419→            "fee_mode": policy.get("mode", "auto"),
   420→            "venue_fee_tiers": policy.get("venue_tiers", {}),
   421→            "venues": venues_out,
   422→        },
   423→    }

   670→def _mask(value: str, visible: int = 4) -> str:
   671→    """Mask a secret, showing only first and last few chars."""
   672→    if not value or len(value) <= visible * 2:
   673→        return "***" if value else ""
   674→    return f"{value[:visible]}...{value[-visible:]}"

   694→    """Check connection status for all venues (CEX + DEX)."""

   696→    result = {}
   697→    for vid in venues_to_check:

   701→        # Check if all required fields have values in env

   711→            if val:
   712→                masked_fields[f["key"]] = _mask(val) if f["type"] == "password" else val
   713→
   714→        live_flag = schema.get("live_flag")
   715→        live_enabled = False
   716→        if live_flag:
   717→            live_enabled = os.environ.get(live_flag, "").strip() in ("1", "true", "yes")
   718→
   719→        # Try to get balance if connected
   720→        balance = 0.0

   724→                    from venues.dydx import DydxVenue

   732→                    v = HyperliquidVenue()

   736→                    from venues import get_venue

   746→            "chain": schema["chain"],

   755→@router.post("/settings/wallet/connect")
   756→async def connect_wallet(req: WalletConnectRequest):
   757→    """Connect a venue by setting its credentials as process env vars.
   758→
   759→    Session-scoped only: the credentials live in this server process's
   760→    environment and are NOT persisted to disk, so they are lost on restart.
   761→    Reconnect after a restart, or store them via ``setup_credentials.py``
   762→    (keyring/age) for unattended 7x24 use. This keeps secrets typed into the
   763→    browser off disk by default.
   764→    """
   765→    schema = _WALLET_SCHEMAS.get(req.venue)
   766→    if not schema:
   767→        return {"success": False, "error": f"Unknown venue: {req.venue}"}
   768→
   769→    # Validate required fields
   770→    required_keys = [f["key"] for f in schema["fields"]]
   771→    missing = [k for k in required_keys if not req.credentials.get(k, "").strip()]
   772→    if missing:
   773→        return {
   774→            "success": False,
   775→            "error": f"Missing required fields: {', '.join(missing)}",
   776→        }
   777→
   778→    # Set env vars for this session
   779→    all_field_keys = [f["key"] for f in schema["fields"]] + [
   780→        f["key"] for f in schema.get("extra_fields", [])
   781→    ]
   782→    for key in all_field_keys:
   783→        val = req.credentials.get(key, "").strip()
   784→        if val:
   785→            os.environ[key] = val
   786→
   787→    # Enable live flag if present
   788→    live_flag = schema.get("live_flag")
   789→    if live_flag and req.credentials.get(live_flag):
   790→        os.environ[live_flag] = req.credentials[live_flag]
   791→
   792→    return {"success": True, "data": {"venue": req.venue, "connected": True}}

   795→@router.post("/settings/wallet/disconnect")
   796→async def disconnect_wallet(req: WalletDisconnectRequest):
   797→    """Disconnect a wallet by clearing its credentials from the session."""
   798→    schema = _WALLET_SCHEMAS.get(req.venue)
   799→    if not schema:
   800→        return {"success": False, "error": f"Unknown venue: {req.venue}"}
   801→
   802→    all_keys = [f["key"] for f in schema["fields"]] + [
   803→        f["key"] for f in schema.get("extra_fields", [])
   804→    ]
   805→    live_flag = schema.get("live_flag")
   806→    if live_flag:
   807→        all_keys.append(live_flag)
   808→
   809→    for key in all_keys:
   810→        os.environ.pop(key, None)
   811→
   812→    return {"success": True, "data": {"venue": req.venue, "connected": False}}

   815→@router.get("/settings/trading-mode")
   816→async def get_trading_mode():
   817→    """Return current trading mode per venue: backtest, dry_run, or live."""
   818→    venues_out = {}
   819→    for vid, info in VENUES.items():
   820→        schema = _WALLET_SCHEMAS.get(vid)
   821→        if schema:
   822→            # Venue is in wallet schemas (CEX or DEX) — check via schema fields
   823→            all_keys = [f["key"] for f in schema["fields"]]
   824→            connected = all(os.environ.get(k, "").strip() for k in all_keys)
   825→            live_flag = schema.get("live_flag")
   826→            live_enabled = False
   827→            if live_flag:
   828→                live_enabled = os.environ.get(live_flag, "").strip() in (
   829→                    "1",
   830→                    "true",
   831→                    "yes",
   832→                )
   833→            mode = "live" if (connected and live_enabled) else "dry_run"
   834→            venues_out[vid] = {
   835→                "mode": mode,
   836→                "wallet_connected": connected,
   837→                "live_enabled": live_enabled,
   838→            }
   839→        else:
   840→            # Fallback: venue not in wallet schemas
   841→            trade_capable, _ = venue_trade_capability(vid)
   842→            keys = info.get("required_keys", []) or info.get("trade_keys", [])
   843→            has_creds = all(os.environ.get(k) for k in keys) if keys else False
   844→            if has_creds and trade_capable:
   845→                venues_out[vid] = {
   846→                    "mode": "dry_run",
   847→                    "wallet_connected": True,
   848→                    "live_enabled": False,
   849→                }
   850→            else:
   851→                venues_out[vid] = {
   852→                    "mode": "dry_run",
   853→                    "wallet_connected": False,
   854→                    "live_enabled": False,
   855→                }
   856→
   857→    # Overall mode: live if any venue is live, else dry_run
   858→    overall = (
   859→        "live" if any(v["mode"] == "live" for v in venues_out.values()) else "dry_run"
   860→    )
   861→
   862→    return {"success": True, "data": {"mode": overall, "venues": venues_out}}

   865→@router.get("/settings/wallet/balance")
   866→async def get_wallet_balance(venue: str, address: str, network: str | None = None):
   867→    """Query wallet balance via public APIs (no credentials needed).
   868→
   869→    Used by frontend wallet extension flow — the address comes from
   870→    Keplr/MetaMask, not from stored env vars. ``network`` (mainnet/testnet)
   871→    comes from the extension's selected chain; it overrides the env default.
   872→    """
   873→    if not address or len(address) < 10:
   874→        return {"success": False, "error": "Invalid address"}
   875→
   876→    balance = 0.0
   877→    equity = 0.0
   878→    net = (network or "").strip().lower()
   879→
   880→    try:
   881→        if venue == "dydx":
   882→            # dYdX v4 indexer: query subaccount by address
   883→            import requests as _requests
   884→
   885→            base = "https://indexer.dydx.trade"
   886→            is_testnet = net == "testnet" or (
   887→                not net and os.environ.get("DYDX_NETWORK", "").lower() == "testnet"
   888→            )
   889→            if is_testnet:
   890→                base = "https://indexer.v4testnet.dydx.exchange"
   891→            subaccount_number = int(os.environ.get("DYDX_SUBACCOUNT_NUMBER", "0") or 0)
   892→            resp = _requests.get(
   893→                f"{base}/v4/subaccounts",
   894→                params={"address": address, "subaccountNumber": subaccount_number},
   895→                timeout=15,
   896→            )
   897→            resp.raise_for_status()
   898→            data = resp.json()
   899→            # Response: { "subaccount": { "equity": "...", "freeCollateral": "..." } }
   900→            sub = data.get("subaccount", {}) if isinstance(data, dict) else {}
   901→            if not sub:
   902→                # Try alternative response format
   903→                subs = data.get("subaccounts", [])
   904→                if isinstance(subs, list) and subs:
   905→                    sub = subs[0] if isinstance(subs[0], dict) else {}
   906→            equity = float(sub.get("equity", 0) or 0)
   907→            free = float(sub.get("freeCollateral", 0) or 0)
   908→            balance = free or equity
   909→
   910→        elif venue == "hyperliquid":
   911→            import requests as _requests
   912→
   913→            hl_base = "https://api.hyperliquid.xyz"
   914→            is_testnet = net == "testnet" or (
   915→                not net
   916→                and os.environ.get("HYPERLIQUID_NETWORK", "").lower() == "testnet"
   917→            )
   918→            if is_testnet:
   919→                hl_base = "https://api.hyperliquid-testnet.xyz"
   920→            resp = _requests.post(
   921→                f"{hl_base}/info",
   922→                json={"type": "clearinghouseState", "user": address},
   923→                timeout=15,
   924→            )
   925→            resp.raise_for_status()
   926→            data = resp.json()
   927→            margin = data.get("marginSummary", {}) if isinstance(data, dict) else {}
   928→            equity = float(margin.get("accountValue", 0) or 0)
   929→            balance = equity
   930→
   931→        elif venue == "lighter":
   932→            import requests as _requests
   933→
   934→            lighter_base = "https://mainnet.zklighter.elliot.ai"
   935→            # Lighter uses L1 address to look up account
   936→            data = _requests.get(
   937→                f"{lighter_base}/api/v1/accountsByL1Address?l1_address={address}",
   938→                timeout=15,
   939→            )
   940→            data.raise_for_status()
   941→            acct = data.json()
   942→            accounts = acct.get("sub_accounts") or acct.get("accounts") or []
   943→            if accounts:
   944→                equity = float(accounts[0].get("equity", 0) or 0)
   945→                balance = equity
   946→
   947→        elif venue == "edgex":
   948→            # EdgeX V2: balance query needs authenticated SDK;
   949→            # public balance-by-address is not available yet.
   950→            balance = 0.0
   951→
   952→        elif venue == "aster":
   953→            # Aster: Binance-fapi API, balance requires API key auth.
   954→            # Wallet address alone is not sufficient for balance query.
   955→            balance = 0.0
   956→
   957→        elif venue in ("binance", "bitget", "bybit", "okx"):
   958→            # CEX — balance requires API keys; can't query by address alone
   959→            return {
   960→                "success": False,
   961→                "error": "CEX balance requires API key connection",
   962→            }
   963→
   964→        else:
   965→            return {"success": False, "error": f"Unsupported venue: {venue}"}
   966→
   967→    except Exception as e:
   968→        return {"success": False, "error": str(e)}
   969→
   970→    return {
   971→        "success": True,
   972→        "data": {
   973→            "balance": balance,
   974→            "equity": equity,
   975→            "address": address,
   976→            "venue": venue,
   977→        },
   978→    }
