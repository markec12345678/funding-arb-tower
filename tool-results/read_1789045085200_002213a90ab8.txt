   101→def normalize_arb_cfg(cfg: dict[str, Any]) -> dict[str, Any]:
   102→    """Normalize to the engine's ``crossAssetArbitrage`` config; single-asset cashAndCarry auto-maps to single slot."""
   103→    out = dict(cfg)
   104→    if cfg.get("crossAssetArbitrage"):
   105→        return out
   106→    cc = cfg.get("cashAndCarry") or {}
   107→    out["crossAssetArbitrage"] = {
   108→        "maxConcurrentPairs": 1,
   109→        "tradeUsdPerSlot": float(cc.get("tradeUsd", 1000.0)),
   110→        "entryFundingRatePct": float(cc.get("entryFundingRatePct", 0.05)),
   111→        "exitFundingRatePct": float(cc.get("exitFundingRatePct", 0.01)),
   112→        "reverseEntryFundingRatePct": float(
   113→            cc.get("reverseEntryFundingRatePct", -0.05)
   114→        ),
   115→        "reverseExitFundingRatePct": float(cc.get("reverseExitFundingRatePct", -0.01)),
   116→        "minReverseSpreadPct": float(cc.get("minReverseSpreadPct", 0.02)),
   117→        "minNetEdgePct": float(cc.get("minNetEdgePct", 0.02)),
   118→        "preemptionFrictionBufferPct": 1e9,
   119→        "maxMinutesToSettlement": float(cc.get("maxMinutesToSettlement", 0)),
   120→        "forwardRequiredCashMult": float(cc.get("forwardRequiredCashMult", 2.1)),
   121→        "reverseRequiredCashMult": float(cc.get("reverseRequiredCashMult", 1.5)),
   122→    }
   123→    return out

   127→    for key in ("crossAssetArbitrage", "cashAndCarry"):

   133→def apply_live_safety(cfg: dict[str, Any]) -> dict[str, Any]:
   134→    """Live safety gate: Reverse C&C requires margin borrow/repay, must have explicit enableReverseArbitrage to proceed.
   135→
   136→    Even if explicitly enabled, run_cycle will also verify the venue implements borrow/repay capability
   137→    (supports_reverse_arbitrage); venues that don't support it are force-disabled in live mode.
   138→    """
   139→    if cfg.get("dry_run", True):
   140→        return cfg
   141→    if cfg.get("enableReverseArbitrage", False):
   142→        return cfg
   143→    disable_reverse(cfg)
   144→    return cfg

   147→def load_config(path: Path) -> dict[str, Any]:
   148→    cfg = json.loads(json.dumps(DEFAULT_CONFIG))
   149→    with open(path, encoding="utf-8") as f:
   150→        user = json.load(f)
   151→    cc = {**cfg["cashAndCarry"], **(user.get("cashAndCarry") or {})}
   152→    cfg.update(user)
   153→    cfg["cashAndCarry"] = cc
   154→    _truthy = ("1", "true", "yes")
   155→    if (os.environ.get("FARB_DRY_RUN", "") or os.environ.get("DCA_DRY_RUN", "")).strip().lower() in _truthy:
   156→        cfg["dry_run"] = True
   157→    if (os.environ.get("FARB_LIVE", "") or os.environ.get("DCA_LIVE", "")).strip().lower() in _truthy:
   158→        cfg["dry_run"] = False
   159→
   160→    cfg = normalize_arb_cfg(cfg)
   161→    cfg = apply_live_safety(cfg)
   162→
   163→    # Validation
   164→    arb = cfg.get("crossAssetArbitrage", {})
   165→    max_min = arb.get("maxMinutesToSettlement", 0)
   166→    run_int = cfg.get("runIntervalMinutes", 30)
   167→    if max_min > 0 and max_min < run_int * 1.5:
   168→        arb["maxMinutesToSettlement"] = int(run_int * 1.5)
   169→
   170→    return cfg

   193→def load_state(state_file: Path) -> dict[str, Any]:
   194→    if not state_file.exists():
   195→        return {}
   196→    try:
   197→        return json.loads(state_file.read_text(encoding="utf-8"))
   198→    except json.JSONDecodeError as e:
   199→        backup = state_file.with_suffix(f".corrupt-{int(time.time())}.json")
   200→        try:
   201→            state_file.replace(backup)
   202→        except OSError:
   203→            backup = Path("(backup_failed)")
   204→        raise RuntimeError(
   205→            f"State file corrupted: {state_file}\nBacked up to: {backup}\nPlease inspect manually before re-running. Original error: {e}"
   206→        ) from e
   207→
   208→
   209→def save_state(state_file: Path, state: dict[str, Any]) -> None:
   210→    state_file.parent.mkdir(parents=True, exist_ok=True)
   211→    tmp = state_file.with_suffix(f".tmp.{os.getpid()}")
   212→    with tmp.open("w", encoding="utf-8") as f:
   213→        json.dump(state, f, indent=2, ensure_ascii=False)
   214→        f.flush()
   215→        os.fsync(f.fileno())
   216→    tmp.replace(state_file)

   235→def fetch_spot(cfg: dict[str, Any], venue: Any, quote: str, sym: str) -> dict[str, Any]:
   236→    mkt = venue.fetch_asset_market(sym, quote, cfg)
   237→    if mkt.get("rules_error"):
   238→        return {
   239→            "price": 0.0,
   240→            "error": f"Cannot fetch {sym} trading rules ({venue.venue_id})",
   241→        }
   242→    px = float(mkt.get("price", 0) or 0)
   243→    if px <= 0:
   244→        fb_px, fb_meta = fetch_price_with_fallback(sym, quote, primary=venue.venue_id)
   245→        if fb_px > 0:
   246→            px = fb_px
   247→            mkt["price_source"] = fb_meta.get("source", "fallback")
   248→
   249→    # Inject futures min trade rules since this strategy trades futures
   250→    if hasattr(venue, "fetch_futures_symbol_rules"):
   251→        f_rules = venue.fetch_futures_symbol_rules(make_pair(sym, quote))
   252→        if f_rules:
   253→            mkt["min_trade_usdt"] = f_rules.get(
   254→                "min_trade_usdt", mkt.get("min_trade_usdt", 0.0)
   255→            )
   256→            mkt["min_trade_base"] = f_rules.get(
   257→                "min_trade_base", mkt.get("min_trade_base", 0.0)
   258→            )
   259→
   260→    mkt["price"] = px
   261→    return mkt

   264→def asset_annual_borrow(
   265→    cfg: dict[str, Any], sym: str, live_rates: dict[str, float] | None = None
   266→) -> float:
   267→    if live_rates and live_rates.get(sym, 0) > 0:
   268→        return live_rates[sym] * 100.0  # Convert live decimal (0.12) to pct (12.0)
   269→    cfg_borrow = cfg.get("borrowAnnualRatePct", 12.0)
   270→    if isinstance(cfg_borrow, dict):
   271→        return float(cfg_borrow.get(sym, cfg_borrow.get("default", 12.0)))
   272→    return float(cfg_borrow)

   278→def build_funding_snapshots(
   279→    cfg: dict[str, Any],
   280→    quote: str,
   281→    held_syms: list[str],
   282→    funding_provider: FundingProvider | None = None,
   283→) -> tuple[list[str], dict[str, dict[str, Any]]]:
   284→    """Determine which assets to monitor this cycle + their funding rate snapshots.
   285→
   286→    - ``autoUniverse=true``: Fetch all-market perps in one go, sort by |funding| and take top ``universeTopN``
   287→      (this is "find the highest funding rate across the full market"). Currently held assets are always
   288→      included to ensure ongoing management/closing.
   289→      When ``funding_provider`` is provided, uses that exchange's rate data (not Binance).
   290→    - Otherwise: use the ``assets`` list from config (still merged with held positions).
   291→    Returns (asset_list, {asset: snapshot}).
   292→    """
   293→    quote_suf = quote.upper()
   294→    snaps: dict[str, dict[str, Any]] = {}
   295→    fp = funding_provider  # may be None, fallback to Binance
   296→
   297→    if bool(cfg.get("autoUniverse")):
   298→        if fp is not None:
   299→            rows = fp.fetch_all(quote_suf)
   300→            interval_map = fp.fetch_interval_map(quote_suf)
   301→        else:
   302→            rows = fetch_all_funding()
   303→            interval_map = fetch_funding_interval_map()
   304→        blacklist = {x.upper() for x in (cfg.get("universeBlacklist") or [])}
   305→        top_n = int(cfg.get("universeTopN", 10))
   306→        min_abs = float(cfg.get("universeMinFundingPct", 0.0))
   307→        row_by_asset: dict[str, dict[str, Any]] = {}
   308→        cand: list[tuple[str, float]] = []
   309→        for r in rows:
   310→            sym = r["symbol"].upper()
   311→            if not sym.endswith(quote_suf):
   312→                continue
   313→            asset = sym[: -len(quote_suf)]
   314→            if not asset or asset in blacklist:
   315→                continue
   316→            row_by_asset[asset] = r
   317→            if abs(r["rate_pct"]) >= min_abs:
   318→                cand.append((asset, abs(r["rate_pct"])))
   319→        cand.sort(key=lambda x: x[1], reverse=True)
   320→        chosen = [a for a, _ in cand[:top_n]]
   321→        asset_list = list(dict.fromkeys(list(held_syms) + chosen))
   322→        # For venues where fetch_all returns next_funding_ts=0 (e.g. Bitget),
   323→        # backfill via fetch_current for the chosen assets only.
   324→        needs_backfill = fp is not None and any(
   325→            row_by_asset.get(a, {}).get("next_funding_ts", 0) == 0
   326→            for a in asset_list
   327→            if a in row_by_asset
   328→        )
   329→        if needs_backfill:
   330→            workers = int(cfg.get("marketFetchWorkers", 8))
   331→            backfill_assets = [
   332→                a
   333→                for a in asset_list
   334→                if a in row_by_asset and row_by_asset[a].get("next_funding_ts", 0) == 0
   335→            ]
   336→            if backfill_assets and fp is not None:
   337→                for asset, detail in fetch_funding_snaps_for_assets(
   338→                    fp, backfill_assets, quote, workers=workers
   339→                ).items():
   340→                    if asset in row_by_asset:
   341→                        row_by_asset[asset]["next_funding_ts"] = detail.get(
   342→                            "next_funding_ts", 0
   343→                        )
   344→                        if detail.get("interval_ms", 0) > 0:
   345→                            row_by_asset[asset]["_interval_ms"] = detail["interval_ms"]
   346→        for asset in asset_list:
   347→            r = row_by_asset.get(asset)
   348→            if not r:
   349→                continue
   350→            sym_key = asset + quote_suf
   351→            default_interval_h = 8.0
   352→            if isinstance(interval_map, dict):
   353→                default_interval_h = interval_map.get(sym_key, 8.0)
   354→            interval_ms = int(r.get("_interval_ms") or 0) or int(
   355→                float(default_interval_h) * 3600 * 1000
   356→            )
   357→            next_ts = int(r["next_funding_ts"] or 0)
   358→            snaps[asset] = {
   359→                "rate_pct": r["rate_pct"],
   360→                "next_funding_ts": next_ts,
   361→                "last_settle_ts": next_ts - interval_ms if next_ts else 0,
   362→                "interval_ms": interval_ms,
   363→                "mark_price": r.get("mark_price", 0.0),
   364→            }
   365→        # Currently held but not in fetch_all list (delisted/special contracts) -> fetch snapshots individually
   366→        missing_held = [a for a in held_syms if a not in snaps]
   367→        if missing_held and fp is not None:
   368→            snaps.update(
   369→                fetch_funding_snaps_for_assets(
   370→                    fp,
   371→                    missing_held,
   372→                    quote,
   373→                    workers=int(cfg.get("marketFetchWorkers", 8)),
   374→                )
   375→            )
   376→        return asset_list, snaps
   377→
   378→    asset_list = list(dict.fromkeys(list(cfg.get("assets", ["BTC"])) + list(held_syms)))
   379→    workers = int(cfg.get("marketFetchWorkers", 8))
   380→    snaps.update(
   381→        fetch_funding_snaps_for_assets(
   382→            fp,
   383→            asset_list,
   384→            quote,
   385→            workers=workers,
   386→            fetch_current_fn=(
   387→                (lambda a: fetch_current_funding(make_pair(a, quote)))
   388→                if fp is None
   389→                else None
   390→            ),
   391→        )
   392→    )
   393→    return asset_list, snaps

   399→def run_cycle(cfg: dict[str, Any], paths: dict[str, Path]) -> dict[str, Any]:
   400→    dry_run = bool(cfg.get("dry_run", True))
   401→
   402→    cash = str(cfg.get("cash", "USDT"))
   403→    venue = get_venue(cfg)
   404→    quote = venue_quote(cfg)
   405→    venue_type = str((cfg.get("venue") or {}).get("type", "binance")).strip().lower()
   406→    fp = get_funding_provider(venue_type)
   407→
   408→    # Reverse C&C capability gate: in live mode with explicit enable, venue must implement margin borrow/repay, otherwise force-disable.
   409→    reverse_disabled_reason = None
   410→    if not dry_run and cfg.get("enableReverseArbitrage", False):
   411→        supports = getattr(venue, "supports_reverse_arbitrage", None)
   412→        if not (callable(supports) and supports()):
   413→            disable_reverse(cfg)
   414→            reverse_disabled_reason = (
   415→                f"venue {venue.venue_id} has not enabled cross margin or does not implement borrow/repay "
   416→                "(supports_reverse_arbitrage=False), Reverse C&C has been force-disabled."
   417→            )
   418→            send_notification("Reverse C&C Disabled", reverse_disabled_reason, cfg)
   419→
   420→    # 1) State first: need current positions before screener can merge held assets into candidates.
   421→    state = load_state(paths["state"])
   422→    if state.get("killed", False):
   423→        return {
   424→            "status": "skipped",
   425→            "reason": "Strategy has been KILLED by Drawdown Cutoff. Clear state to restart.",
   426→        }
   427→
   428→    futures_state = state.get("futures_state") or default_futures_state()
   429→    positions = futures_state.get("positions", {})
   430→    held_syms = [s for s, p in positions.items() if p.get("amount", 0) > 0]
   431→    settle_ts: dict[str, int] = dict(state.get("funding_settle_ts") or {})
   432→    liq_cooldown: dict[str, int] = dict(state.get("liq_cooldown") or {})
   433→    last_run_ms = int(state.get("last_run_ms", 0) or 0)
   434→    now = now_ms()
   435→    elapsed_h = max(0.0, (now - last_run_ms) / 3_600_000.0) if last_run_ms else 0.0
   436→
   437→    # 2) Full-market screening / funding rate snapshots
   438→    try:
   439→        asset_list, funding_snaps = build_funding_snapshots(
   440→            cfg, quote, held_syms, funding_provider=fp
   441→        )
   442→    except Exception as e:
   443→        return {
   444→            "status": "skipped",
   445→            "reason": f"Funding rate snapshot fetch failed: {e}, skipping entire cycle",
   446→        }
   447→    if not asset_list:
   448→        return {
   449→            "status": "hold",
   450→            "reason": "No candidate assets (universe is empty and no positions)",
   451→        }
   452→
   453→    # 3) Market data (parallel fetch for candidates + held assets; failed assets excluded from decisions this cycle but don't affect others)
   454→    prices: dict[str, float] = {}
   455→    market: dict[str, dict[str, Any]] = {}
   456→    workers = int(cfg.get("marketFetchWorkers", 8))
   457→    raw_market = fetch_assets_market_parallel(
   458→        venue, asset_list, quote, cfg, max_workers=workers
   459→    )
   460→    for sym in asset_list:
   461→        mkt = raw_market.get(sym)
   462→        if not mkt:
   463→            continue
   464→        if mkt.get("rules_error"):
   465→            continue
   466→        px = float(mkt.get("price", 0) or 0)
   467→        if px <= 0:
   468→            fb_px, fb_meta = fetch_price_with_fallback(
   469→                sym, quote, primary=venue.venue_id
   470→            )
   471→            if fb_px > 0:
   472→                px = fb_px
   473→                mkt["price_source"] = fb_meta.get("source", "fallback")
   474→        if hasattr(venue, "fetch_futures_symbol_rules"):
   475→            f_rules = venue.fetch_futures_symbol_rules(make_pair(sym, quote))
   476→            if f_rules:
   477→                mkt["min_trade_usdt"] = f_rules.get(
   478→                    "min_trade_usdt", mkt.get("min_trade_usdt", 0.0)
   479→                )
   480→                mkt["min_trade_base"] = f_rules.get(
   481→                    "min_trade_base", mkt.get("min_trade_base", 0.0)
   482→                )
   483→        mkt["price"] = px
   484→        if px > 0:
   485→            prices[sym] = px
   486→            market[sym] = mkt
   487→    if not prices:
   488→        return {
   489→            "status": "skipped",
   490→            "reason": "All candidate asset price fetches failed, skipping entire cycle",
   491→        }
   492→
   493→    # Fetch live borrow rates from venue only if not dry_run (to avoid API key noise in paper mode)
   494→    live_borrow_rates = {}
   495→    if not dry_run:
   496→        try:
   497→            live_borrow_rates = getattr(venue, "fetch_borrow_rates", lambda x: {})(
   498→                list(prices.keys())
   499→            )
   500→        except Exception:
   501→            pass
   502→
   503→    # 4) seed holdings
   504→    holdings = dict(state.get("holdings") or {})
   505→    if not holdings:
   506→        holdings = {cash: float(cfg.get("initialCapitalUsd", 10000.0))}
   507→
   508→    initial_nav = float(state.get("initial_nav") or 0)
   509→
   510→    if not dry_run:
   511→        model_nav = calculate_futures_nav(holdings, futures_state, prices, cash)
   512→        # Sync real global state from exchange to overwrite modeled holdings and positions
   513→        live_state = getattr(venue, "fetch_live_state", lambda x: {"balances": {}})(
   514→            list(prices.keys()) + [cash]
   515→        )
   516→        if "balances" in live_state and live_state["balances"]:
   517→            for k, v in live_state["balances"].items():
   518→                holdings[k] = v
   519→        if "futures_positions" in live_state:
   520→            futures_state["positions"] = live_state["futures_positions"]
   521→
   522→        real_nav = calculate_futures_nav(holdings, futures_state, prices, cash)
   523→        if initial_nav <= 0:
   524→            initial_nav = real_nav
   525→        drift = abs(real_nav - model_nav) / (model_nav or 1)
   526→        if drift > 0.05:
   527→            send_notification(
   528→                "Drift Alarm",
   529→                f"NAV Drift detected: Model NAV = {model_nav:.2f}, Real NAV = {real_nav:.2f} ({(drift * 100):.2f}%).",
   530→                cfg,
   531→            )
   532→
   533→        max_drawdown = float(cfg.get("maxDrawdownKillSwitchPct", 0.15))
   534→        if initial_nav > 0 and (initial_nav - real_nav) / initial_nav > max_drawdown:
   535→            send_notification(
   536→                "Drawdown Kill Switch",
   537→                f"Drawdown exceeded {max_drawdown * 100}%. Real NAV {real_nav:.2f} vs Initial {initial_nav:.2f}. Executing violent close.",
   538→                cfg,
   539→            )
   540→            return trigger_kill_switch(
   541→                venue,
   542→                holdings,
   543→                futures_state,
   544→                prices,
   545→                market,
   546→                cfg,
   547→                initial_nav,
   548→                real_nav,
   549→                paths,
   550→                state,
   551→                now,
   552→            )
   553→    elif initial_nav <= 0:
   554→        initial_nav = calculate_futures_nav(holdings, futures_state, prices, cash)
   555→
   556→    for sym in asset_list:
   557→        holdings.setdefault(sym, 0.0)
   558→
   559→    # 5) Borrow interest accrual (independent of funding rate fetch success, linear per elapsed hours)
   560→    borrow_info: dict[str, Any] = {}
   561→    if elapsed_h > 0:
   562→        period_rates: dict[str, float] = {}
   563→        for sym in prices:
   564→            if holdings.get(sym, 0.0) < 0:
   565→                per_h = (
   566→                    asset_annual_borrow(cfg, sym, live_borrow_rates) / HOURS_PER_YEAR
   567→                )
   568→                period_rates[sym] = per_h * elapsed_h
   569→        if period_rates:
   570→            before = holdings.get(cash, 0.0)
   571→            if dry_run:
   572→                holdings, futures_state = apply_borrow_fees(
   573→                    holdings, futures_state, prices, period_rates, cash
   574→                )
   575→                borrow_info = {
   576→                    "charged_usd": round(before - holdings.get(cash, 0.0), 6),
   577→                    "hours": round(elapsed_h, 3),
   578→                }
   579→            else:
   580→                _, futures_state = apply_borrow_fees(
   581→                    dict(holdings), futures_state, prices, period_rates, cash
   582→                )
   583→                borrow_info = {
   584→                    "charged_usd": 0.0,
   585→                    "hours": round(elapsed_h, 3),
   586→                    "note": "live_sync",
   587→                }
   588→
   589→    # 6) Funding rate snapshots + per-settlement-point catch-up settlements
   590→    funding_rates: dict[str, float] = {}
   591→    borrow_interval_rates: dict[str, float] = {}
   592→    funding_meta: dict[str, Any] = {}
   593→    pending_settle: list[tuple[str, int]] = []
   594→    for sym in prices:
   595→        snap = funding_snaps.get(sym)
   596→        if snap is None:
   597→            funding_meta[sym] = {"error": "no_funding"}
   598→            continue
   599→        rate = float(snap.get("rate_pct", 0.0))
   600→        interval_h = (
   601→            int(snap.get("interval_ms", 8 * 3600 * 1000) or 8 * 3600 * 1000)
   602→            / 3_600_000.0
   603→        )
   604→        funding_rates[sym] = rate
   605→        borrow_interval_rates[sym] = (
   606→            asset_annual_borrow(cfg, sym, live_borrow_rates) / HOURS_PER_YEAR
   607→        ) * interval_h
   608→        held = sym in positions and positions[sym].get("amount", 0) > 0
   609→        prev_ts = int(settle_ts.get(sym, 0) or 0)
   610→        if held and prev_ts > 0:
   611→            pending_settle.append((sym, prev_ts))
   612→
   613→    workers = int(cfg.get("marketFetchWorkers", 8))
   614→    held_with_ts = {s: t for s, t in pending_settle}
   615→    if held_with_ts and fp is not None:
   616→        history = fetch_funding_history_parallel(
   617→            fp, list(held_with_ts.keys()), quote, held_with_ts, workers=workers
   618→        )
   619→    elif held_with_ts:
   620→        history = fetch_funding_history_parallel(
   621→            None,
   622→            list(held_with_ts.keys()),
   623→            quote,
   624→            held_with_ts,
   625→            workers=workers,
   626→            fetch_since_fn=lambda pair, start: fetch_funding_since(pair, start),
   627→        )
   628→    else:
   629→        history = {}
   630→
   631→    for sym in prices:
   632→        snap = funding_snaps.get(sym)
   633→        if snap is None:
   634→            continue
   635→        rate = funding_rates[sym]
   636→        prev_ts = int(settle_ts.get(sym, 0) or 0)
   637→        settled_n = 0
   638→        if sym in history:
   639→            for row in sorted(history[sym], key=lambda r: int(r.get("ts", 0))):
   640→                if row["ts"] > now:
   641→                    continue
   642→                if dry_run:
   643→                    holdings, futures_state = apply_funding_fees(
   644→                        holdings, futures_state, prices, {sym: row["rate_pct"]}, cash
   645→                    )
   646→                else:
   647→                    _, futures_state = apply_funding_fees(
   648→                        dict(holdings),
   649→                        futures_state,
   650→                        prices,
   651→                        {sym: row["rate_pct"]},
   652→                        cash,
   653→                    )
   654→                settle_ts[sym] = row["ts"]
   655→                settled_n += 1
   656→        elif sym in positions and positions[sym].get("amount", 0) > 0 and prev_ts == 0:
   657→            settle_ts[sym] = int(snap.get("last_settle_ts", 0) or 0)
   658→        funding_meta[sym] = {
   659→            "rate_pct": rate,
   660→            "apr_pct": round(rate * FUNDINGS_PER_YEAR, 2),
   661→            "last_settle_ts": int(snap.get("last_settle_ts", 0) or 0),
   662→            "next_funding_ts": snap.get("next_funding_ts"),
   663→            "settled_intervals": settled_n,
   664→        }
   665→
   666→    next_funding_times = {
   667→        s: (fm.get("next_funding_ts") or 0) for s, fm in funding_meta.items()
   668→    }
   669→
   670→    leverage = float(cfg.get("leverage", 1.0))
   671→    mmr = float(cfg.get("maintenanceMarginRate", 0.005))
   672→    fee_rate = float(cfg.get("takerFeeRate", 0.0005))
   673→
   674→    # 4.5) Liquidation guard: perp leg hits liquidation price -> emergency two-leg unwind (close perp + close hedge spot) + penalty.
   675→    #      Delta-neutral under cross-margin rarely triggers; this models the worst case for isolated margin.
   676→    liquidation_events: list[dict[str, Any]] = []
   677→    liq_hits = check_liquidations(futures_state, prices)
   678→    if liq_hits:
   679→        penalty_pct = float(
   680→            cfg.get("liquidationPenaltyPct", 0.01)
   681→        )  # Liquidation penalty (including slippage)
   682→        unwind: list[dict[str, Any]] = []
   683→        liq_notional = 0.0
   684→        for hit in liq_hits:
   685→            sym = hit["symbol"]
   686→            pos = positions[sym]
   687→            px = prices[sym]
   688→            liq_notional += (
   689→                pos["amount"] * px
   690→            )  # Record notional before close zeros out amount
   691→            unwind.append(
   692→                {
   693→                    "symbol": sym,
   694→                    "type": "close_short" if pos["side"] == "short" else "close_long",
   695→                    "amount_base": round(pos["amount"], 8),
   696→                    "amount_usdt": round(pos["amount"] * px, 2),
   697→                    "reason": f"LIQUIDATION: mark {px} hit liquidation price {hit['liq_price']:.6g} ({pos['side']}).",
   698→                }
   699→            )
   700→            spot = holdings.get(sym, 0.0)
   701→            if spot > 1e-9:  # forward: close long spot
   702→                unwind.append(
   703→                    {
   704→                        "symbol": sym,
   705→                        "type": "sell",
   706→                        "amount_base": round(spot, 8),
   707→                        "amount_usdt": round(spot * px, 2),
   708→                        "reason": "Liquidation triggered close of hedge spot leg.",
   709→                    }
   710→                )
   711→            elif (
   712→                spot < -1e-9
   713→            ):  # reverse: buy back short spot (margin buy and auto-repay)
   714→                unwind.append(
   715→                    {
   716→                        "symbol": sym,
   717→                        "type": "buy",
   718→                        "amount_base": round(-spot, 8),
   719→                        "amount_usdt": round(-spot * px, 2),
   720→                        "account": "margin",
   721→                        "side_effect": "auto_repay",
   722→                        "reason": "Liquidation triggered buyback of hedge spot leg.",
   723→                    }
   724→                )
   725→        if dry_run:
   726→            ux = [
   727→                {**t, "status": "simulated", "price": prices[t["symbol"]]}
   728→                for t in unwind
   729→            ]
   730→        else:
   731→            ux = execute_delta_neutral_trades(
   732→                venue, unwind, market, dry_run=False, config=cfg
   733→            )
   734→
   735→        ux_normalized = normalize_executed_for_ledger(ux, prices)
   736→
   737→        if ux_normalized:
   738→            holdings, futures_state = apply_simulated_futures_trades(
   739→                holdings,
   740→                futures_state,
   741→                ux_normalized,
   742→                prices,
   743→                cash,
   744→                spot_fee_rate=fee_rate,
   745→                perp_fee_rate=fee_rate,
   746→                leverage=leverage,
   747→                maintenance_margin_rate=mmr,
   748→            )
   749→        # Liquidation penalty (based on liquidated notional, recorded before amount was zeroed)
   750→        # Note: This liquidation penalty is a locally estimated slippage charge. If a real exchange liquidation
   751→        # occurs in live trading, fetch_live_state will sync real balances, so we don't deduct local cash to avoid double counting.
   752→        penalty = liq_notional * penalty_pct
   753→        if not dry_run:
   754→            send_notification(
   755→                "Liquidation Event",
   756→                f"Liquidations triggered for {', '.join([h['symbol'] for h in liq_hits])}. Penalty: {penalty:.2f} USD",
   757→                cfg,
   758→            )
   759→        if dry_run:
   760→            holdings[cash] = holdings.get(cash, 0.0) - penalty
   761→        positions = futures_state.get("positions", {})
   762→        liquidation_events = [{**h, "penalty_usd": round(penalty, 2)} for h in liq_hits]
   763→        # Liquidated assets enter cooldown to avoid re-opening the same blown-up position this cycle or soon after.
   764→        cooldown_ms = int(
   765→            float(cfg.get("liquidationCooldownHours", 24.0)) * 3600 * 1000
   766→        )
   767→        for h in liq_hits:
   768→            liq_cooldown[h["symbol"]] = now + cooldown_ms
   769→
   770→    # Cooled-down assets removed from candidates (still managed by existing position settle/close logic, just no new opens).
   771→    liq_cooldown = {s: t for s, t in liq_cooldown.items() if t > now}
   772→    for sym in list(funding_rates):
   773→        if sym in liq_cooldown:
   774→            funding_rates.pop(sym, None)
   775→
   776→    # 5) Decision: candidates must have funding; held positions additionally merged into prices for engine processing (missing funding -> hold)
   777→    decision_prices = {s: prices[s] for s in funding_rates}
   778→    decision_market = {s: market[s] for s in funding_rates}
   779→    for sym, pos in positions.items():
   780→        if pos.get("amount", 0) <= 0 or sym not in prices:
   781→            continue
   782→        decision_prices.setdefault(sym, prices[sym])
   783→        decision_market.setdefault(sym, market.get(sym, {"price": prices[sym]}))
   784→
   785→    if not dry_run:
   786→        for sym in decision_prices:
   787→            pair = make_pair(sym, quote)
   788→            getattr(venue, "initialize_futures_symbol", lambda x: None)(pair)
   789→
   790→    trades, meta = decide_cross_asset_arbitrage(
   791→        holdings,
   792→        futures_state,
   793→        decision_prices,
   794→        decision_market,
   795→        cfg,
   796→        funding_rates,
   797→        borrow_interval_rates,
   798→        next_funding_times=next_funding_times,
   799→        current_time_ms=now,
   800→    )
   801→
   802→    # 6) Execute trades (real or paper)
   803→    executed: list[dict[str, Any]] = []
   804→    if trades:
   805→        executed = execute_delta_neutral_trades(
   806→            venue, trades, decision_market, dry_run=dry_run, config=cfg
   807→        )
   808→
   809→    if executed:
   810→        ux = normalize_executed_for_ledger(executed, prices)
   811→        if ux:
   812→            holdings, futures_state = apply_simulated_futures_trades(
   813→                holdings,
   814→                futures_state,
   815→                ux,
   816→                prices,
   817→                cash,
   818→                spot_fee_rate=fee_rate,
   819→                perp_fee_rate=fee_rate,
   820→                leverage=leverage,
   821→                maintenance_margin_rate=mmr,
   822→            )
   823→    # New-opened symbols: align settlement timestamp to latest settled point, avoid miscounting pre-open funding next cycle.
   824→    for ex in executed:
   825→        if str(ex.get("type", "")).startswith("open_"):
   826→            s = ex["symbol"]
   827→            settle_ts[s] = int(funding_meta.get(s, {}).get("last_settle_ts", 0) or 0)
   828→
   829→    # Clean up settlement timestamps for closed symbols
   830→    open_syms = set(futures_state.get("positions", {}).keys())
   831→    settle_ts = {s: t for s, t in settle_ts.items() if s in open_syms}
   832→
   833→    nav = calculate_futures_nav(holdings, futures_state, prices, cash)
   834→
   835→    # 7) Persist
   836→    state.update(
   837→        {
   838→            "strategyId": cfg.get("strategyId"),
   839→            "holdings": {k: round(v, 10) for k, v in holdings.items()},
   840→            "futures_state": futures_state,
   841→            "funding_settle_ts": settle_ts,
   842→            "liq_cooldown": liq_cooldown,
   843→            "last_run_ms": now,
   844→            "last_equity_usd": nav,
   845→            "dry_run": dry_run,
   846→            "initial_nav": initial_nav,
   847→        }
   848→    )
   849→    save_state(paths["state"], state)
   850→
   851→    meta.update(
   852→        {
   853→            "nav_usdt": round(nav, 2),
   854→            "venue": venue.venue_id,
   855→            "funding": funding_meta,
   856→            "borrow": borrow_info,
   857→            "margin": margin_health(holdings, futures_state, prices, cash, mmr),
   858→            "missing_funding": [s for s in prices if s not in funding_rates],
   859→            "cumulative_funding_paid": round(
   860→                futures_state.get("cumulative_funding_paid", 0.0), 4
   861→            ),
   862→            "cumulative_borrow_paid": round(
   863→                futures_state.get("cumulative_borrow_paid", 0.0), 4
   864→            ),
   865→            "cumulative_fees": round(futures_state.get("cumulative_fees", 0.0), 4),
   866→            "dry_run": dry_run,
   867→        }
   868→    )
   869→    if reverse_disabled_reason:
   870→        meta["reverse_disabled"] = reverse_disabled_reason
   871→    if liquidation_events:
   872→        meta["liquidations"] = liquidation_events
   873→        meta["status"] = "LIQUIDATION"
   874→
   875→    append_jsonl(
   876→        paths["journal"],
   877→        {
   878→            "ts": now_iso(),
   879→            "strategyId": cfg.get("strategyId"),
   880→            "strategy": "cash_and_carry",
   881→            "dry_run": dry_run,
   882→            "meta": meta,
   883→            "trades": executed,
   884→        },
   885→    )
   886→    append_jsonl(
   887→        paths["equity"],
   888→        {
   889→            "ts": now_iso(),
   890→            "equity_usd": round(nav, 4),
   891→            "funding": funding_meta,
   892→            "positions": futures_state.get("positions", {}),
   893→            "holdings": state["holdings"],
   894→            "dry_run": dry_run,
   895→        },
   896→    )
   897→    write_json(
   898→        paths["summary"],
   899→        {
   900→            "ts": now_iso(),
   901→            "strategyId": cfg.get("strategyId"),
   902→            "status": meta.get("status"),
   903→            "nav_usdt": round(nav, 2),
   904→            "holdings": state["holdings"],
   905→            "futures_state": futures_state,
   906→            "funding": funding_meta,
   907→        },
   908→    )
   909→
   910→    return {
   911→        "status": meta.get("status", "hold"),
   912→        "trades": executed,
   913→        "meta": meta,
   914→        "dry_run": dry_run,
   915→    }

   918→def trigger_kill_switch(
   919→    venue,
   920→    holdings,
   921→    futures_state,
   922→    prices,
   923→    market,
   924→    cfg,
   925→    initial_nav,
   926→    real_nav,
   927→    paths,
   928→    state,
   929→    now,
   930→):
   931→    from strategies.futures.cross_asset_arbitrage import _close_pair_trades
   932→
   933→    unwind: list[dict[str, Any]] = []
   934→    positions = futures_state.get("positions", {})
   935→    for sym, pos in positions.items():
   936→        if pos.get("amount", 0) <= 1e-9:
   937→            continue
   938→        px = float(prices.get(sym, 0) or 0)
   939→        if px <= 0:
   940→            continue
   941→        unwind.extend(
   942→            _close_pair_trades(sym, pos, holdings.get(sym, 0.0), px, "KILL SWITCH")
   943→        )
   944→
   945→    executed: list[dict[str, Any]] = []
   946→    if unwind:
   947→        executed = execute_delta_neutral_trades(
   948→            venue, unwind, market, dry_run=False, config=cfg
   949→        )
   950→
   951→    if executed:
   952→        # Kill switch only accounts for real fills; simulated should not appear in panic close path
   953→        ux = normalize_executed_for_ledger(
   954→            [ex for ex in executed if ex.get("status") == "filled"], prices
   955→        )
   956→        if ux:
   957→            holdings, futures_state = apply_simulated_futures_trades(
   958→                holdings,
   959→                futures_state,
   960→                ux,
   961→                prices,
   962→                cfg.get("cash", "USDT"),
   963→                spot_fee_rate=float(cfg.get("takerFeeRate", 0.0005)),
   964→                perp_fee_rate=float(cfg.get("takerFeeRate", 0.0005)),
   965→            )
   966→            state["holdings"] = {k: round(v, 10) for k, v in holdings.items()}
   967→            state["futures_state"] = futures_state
   968→
   969→    state["killed"] = True
   970→    state["initial_nav"] = initial_nav
   971→    save_state(paths["state"], state)
   972→
   973→    # Log to journal
   974→    append_jsonl(
   975→        paths["journal"],
   976→        {
   977→            "ts": now_iso(),
   978→            "strategyId": cfg.get("strategyId"),
   979→            "strategy": "cash_and_carry",
   980→            "dry_run": False,
   981→            "meta": {
   982→                "status": "KILLED",
   983→                "reason": "Drawdown limit reached",
   984→                "initial_nav": initial_nav,
   985→                "real_nav": real_nav,
   986→            },
   987→            "trades": executed,
   988→        },
   989→    )
   990→
   991→    return {
   992→        "status": "KILLED",
   993→        "reason": "Drawdown limit reached. Strategy killed.",
   994→        "trades": executed,
   995→    }

   998→def run_once(cfg: dict[str, Any], paths: dict[str, Path]) -> dict[str, Any]:
   999→    """Execute one cycle under file lock; if lock cannot be acquired, another instance is running, skip this cycle (prevent concurrent state writes)."""
  1000→    paths["lock"].parent.mkdir(parents=True, exist_ok=True)
  1001→    lock_fd = os.open(str(paths["lock"]), os.O_CREAT | os.O_RDWR, 0o644)
  1002→    try:
  1003→        if not lock_exclusive(lock_fd, blocking=False):
  1004→            return {
  1005→                "status": "skipped",
  1006→                "reason": "Another instance is running (lock not acquired), skipping this cycle",
  1007→            }
  1008→        return run_cycle(cfg, paths)
  1009→    finally:
  1010→        unlock(lock_fd)
  1011→        os.close(lock_fd)

  1014→def main() -> None:
  1015→    parser = argparse.ArgumentParser(
  1016→        description="Cash-and-Carry funding rate arbitrage runner (paper, multi-asset)"
  1017→    )
  1018→    parser.add_argument("--config", required=True, help="Config JSON path")
  1019→    parser.add_argument(
  1020→        "--verbose", action="store_true", help="Print summary even when no trades"
  1021→    )
  1022→    args = parser.parse_args()
  1023→
  1024→    config_path = resolve_config_path(Path(args.config))
  1025→    if not config_path.exists():
  1026→        print(f"Config not found: {config_path}", file=sys.stderr)
  1027→        sys.exit(1)
  1028→
  1029→    cfg = load_config(config_path)
  1030→    paths = resolve_paths(cfg)
  1031→
  1032→    result = run_once(cfg, paths)
  1033→    status = result.get("status")
  1034→    if status == "error":
  1035→        print(f"⚠️ {result.get('reason', 'Unknown error')}")
  1036→        sys.exit(1)
