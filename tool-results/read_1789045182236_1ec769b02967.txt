   723→   723→def poll_deposit_until(
   724→   724→    venue: str,
   725→   725→    coin: str,
   726→   726→    expected_amount: float,
   727→   727→    since_ms: int,
   728→   728→    *,
   729→   729→    timeout_s: int = 600,
   730→   730→    poll_interval_s: int = 15,
   731→   731→    tolerance: float = 0.02,
   732→   732→) -> tuple[bool, list[dict[str, Any]]]:
   733→   733→    """Poll for deposit arrival. Returns (matched, records)."""
   734→   734→    provider = get_transfer_provider(venue)
   735→   735→    deadline = time.time() + timeout_s
   736→   736→    seen: list[dict[str, Any]] = []
   737→   737→    seen_ids: set[str] = set()
   738→   738→    min_amt = max(0.0, expected_amount * (1.0 - tolerance))
   739→   739→    while time.time() < deadline:
   740→   740→        rows = provider.fetch_deposit_records(coin, since_ms=since_ms)
   741→   741→        for row in rows:
   742→   742→            rid = str(row.get("tx_id") or row.get("ts_ms") or "")
   743→   743→            if rid and rid in seen_ids:
   744→   744→                continue
   745→   745→            if rid:
   746→   746→                seen_ids.add(rid)
   747→   747→            seen.append(row)
   748→   748→            if not _deposit_confirmed(row):
   749→   749→                continue
   750→   750→            if _safe_float(row.get("amount")) >= min_amt:
   751→   751→                return True, seen
   752→   752→        time.sleep(poll_interval_s)
   753→   753→    return False, seen
