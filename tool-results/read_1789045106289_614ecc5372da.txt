   111→   111→def _get_mark_price(venue_id: str, base: str, quote: str = "USDT") -> float:
   112→   112→    """Get single-venue perp mark price. Uses short-lived cache to avoid repeated calls within a loop."""
   113→   113→    key = (str(venue_id).lower(), str(base).upper(), quote.upper())
   114→   114→    now = time.monotonic()
   115→   115→    cached = _mark_price_cache.get(key)
   116→   116→    if cached is not None:
   117→   117→        ts, price = cached
   118→   118→        if now - ts < _MARK_PRICE_TTL_SEC:
   119→   119→            return price
   120→   120→    try:
   121→   121→        v = _get_venue_cached(venue_id)
   122→   122→        pair = make_pair(base, quote)
   123→   123→        if getattr(v, "venue_id", "") == "okx" or venue_id == "okx":
   124→   124→            pair = f"{base.upper()}-{quote.upper()}-SWAP"
   125→   125→        price = float(v.get_ticker(pair) or 0.0)
   126→   126→    except Exception:
   127→   127→        price = 0.0
   128→   128→    _mark_price_cache[key] = (now, price)
   129→   129→    return price

   132→   132→def _prefetch_all_mark_prices(venue_ids: set[str], cache_sec: int = 5) -> None:
   133→   133→    """Bulk-fetch all futures tickers for the given venues in parallel.
   134→   134→
   135→   135→    Populates `_mark_price_cache` so subsequent `_get_mark_price` calls
   136→   136→    return instantly without hitting the API.
   137→   137→    """
   138→   138→    if not venue_ids:
   139→   139→        return
   140→   140→    valid: set[str] = set()
   141→   141→    for vid in venue_ids:
   142→   142→        try:
   143→   143→            v = _get_venue_cached(vid)
   144→   144→            if hasattr(v, "get_all_futures_tickers"):
   145→   145→                valid.add(vid)
   146→   146→        except Exception:
   147→   147→            continue
   148→   148→    if not valid:
   149→   149→        return
   150→   150→
   151→   151→    def _fetch_one(vid: str) -> tuple[str, dict[str, float]]:
   152→   152→        try:
   153→   153→            v = _get_venue_cached(vid)
   154→   154→            return vid, v.get_all_futures_tickers(cache_sec=cache_sec)
   155→   155→        except Exception:
   156→   156→            return vid, {}
   157→   157→
   158→   158→    results = run_io_parallel(
   159→   159→        sorted(valid), _fetch_one, max_workers=max(4, len(valid)), swallow_errors=True
   160→   160→    )
   161→   161→
   162→   162→    now = time.monotonic()
   163→   163→    for vid, tickers in results.items():
   164→   164→        for pair, price in tickers.items():
   165→   165→            if "-USDT-SWAP" in pair:
   166→   166→                base = pair.replace("-USDT-SWAP", "")
   167→   167→                key = (vid.lower(), base.upper(), "USDT")
   168→   168→            elif pair.endswith("USDT"):
   169→   169→                base = pair[:-4]
   170→   170→                key = (vid.lower(), base.upper(), "USDT")
   171→   171→            else:
   172→   172→                continue
   173→   173→            _mark_price_cache[key] = (now, float(price))

   176→   176→def check_exit(
   177→   177→    pos: dict[str, Any],
   178→   178→    scan_rates: dict[str, dict[str, dict[str, Any]]],
   179→   179→    exit_edge: float,
   180→   180→) -> tuple[bool, str]:
   181→   181→    """Check whether a position should exit (spread has narrowed).
   182→   182→
   183→   183→    Returns (should_exit, reason).
   184→   184→    """
   185→   185→    base = str(pos.get("base", "")).upper()
   186→   186→    long_venue = str(pos.get("long_venue", ""))
   187→   187→    short_venue = str(pos.get("short_venue", ""))
   188→   188→
   189→   189→    current_spread = _get_current_spread(base, long_venue, short_venue, scan_rates)
   190→   190→    if current_spread is None:
   191→   191→        # Cannot get current rate → do not actively exit (conservative approach)
   192→   192→        return False, "rate_unavailable"
   193→   193→
   194→   194→    # For forward: spread = short_rate - long_rate > 0 profitable
   195→   195→    # Exit when spread collapses below exit_edge
   196→   196→    # For reverse: spread is same formula but both rates negative
   197→   197→    if current_spread <= exit_edge:
   198→   198→        return True, f"spread_collapse: {current_spread:.4f}% ≤ {exit_edge}%"
   199→   199→
   200→   200→    return False, ""

   203→   203→def estimate_spread_pnl(
   204→   204→    pos: dict[str, Any],
   205→   205→    current_long_px: float,
   206→   206→    current_short_px: float,
   207→   207→) -> dict[str, Any]:
   208→   208→    """Estimate position spread P&L.
   209→   209→
   210→   210→    Price P&L = inter-venue spread at open - inter-venue spread now (scaled by quantity)
   211→   211→    forward: spread narrows -> profit; reverse: spread narrows -> loss
   212→   212→    """
   213→   213→    long_price = float(pos.get("long_price", 0))
   214→   214→    short_price = float(pos.get("short_price", 0))
   215→   215→    qty = float(pos.get("qty", 0))
   216→   216→    trade_usd = float(pos.get("trade_usd", 0))
   217→   217→    direction = str(pos.get("direction", "forward")).lower()
   218→   218→
   219→   219→    # Forward: long on cheaper venue, short on expensive → profit when spread narrows
   220→   220→    # Reverse: positions flipped → loss when spread narrows
   221→   221→    sign = 1.0 if direction == "forward" else -1.0
   222→   222→    open_spread = abs(long_price - short_price)
   223→   223→    close_spread = abs(current_long_px - current_short_px)
   224→   224→    spread_pnl = sign * (open_spread - close_spread) * qty
   225→   225→    spread_pnl_pct = (spread_pnl / trade_usd * 100) if trade_usd > 0 else 0.0
   226→   226→
   227→   227→    return {
   228→   228→        "open_spread": open_spread,
   229→   229→        "close_spread": close_spread,
   230→   230→        "spread_pnl": spread_pnl,
   231→   231→        "spread_pnl_pct": spread_pnl_pct,
   232→   232→    }

   235→   235→def _leg_qty_from_snapshot(
   236→   236→    venue_positions: dict[str, list[dict[str, Any]]],

   238→   238→    base: str,
   239→   239→    side: str,
   240→   240→) -> float | None:
   241→   241→    """Extract actual quantity for a leg from pre-fetched venue position snapshot."""
   242→   242→    rows = venue_positions.get(venue_id)
   243→   243→    if rows is None:
   244→   244→        return None
   245→   245→    base_u = base.upper()
   246→   246→    for p in rows:
   247→   247→        sym = str(p.get("symbol", "")).upper()
   248→   248→        if sym.startswith(base_u) and str(p.get("side", "")).lower() == side:
   249→   249→            return abs(float(p.get("qty", 0) or p.get("amount", 0)))
   250→   250→    return 0.0

   253→   253→def check_rebalance(
   254→   254→    pos: dict[str, Any],
   255→   255→    max_skew_pct: float = 1.0,
   256→   256→    venue_positions: dict[str, list[dict[str, Any]]] | None = None,
   257→   257→) -> tuple[bool, str, float, float]:
   258→   258→    """Check if leg notional values have diverged. Returns (need_rebalance, reason, long_notional, short_notional).
   259→   259→
   260→   260→    max_skew_pct: Threshold percentage for leg notional value divergence (default 1%).
   261→   261→    venue_positions: Optional exchange position snapshot; when provided, actual leg quantities
   262→   262→    are used (can detect quantity mismatch from partial liquidation/ADL, which is real delta exposure).
   263→   263→    """
   264→   264→    base = str(pos.get("base", ""))
   265→   265→    long_venue = str(pos.get("long_venue", ""))
   266→   266→    short_venue = str(pos.get("short_venue", ""))
   267→   267→    qty = float(pos.get("qty", 0))
   268→   268→
   269→   269→    long_qty = short_qty = qty
   270→   270→    if venue_positions:
   271→   271→        actual_lq = _leg_qty_from_snapshot(venue_positions, long_venue, base, "long")
   272→   272→        actual_sq = _leg_qty_from_snapshot(venue_positions, short_venue, base, "short")
   273→   273→        if actual_lq:
   274→   274→            long_qty = actual_lq
   275→   275→        if actual_sq:
   276→   276→            short_qty = actual_sq
   277→   277→
   278→   278→    long_px = _get_mark_price(long_venue, base)
   279→   279→    short_px = _get_mark_price(short_venue, base)
   280→   280→    if long_px <= 0 or short_px <= 0:
   281→   281→        return False, "price_unavailable", 0.0, 0.0
   282→   282→
   283→   283→    long_notional = long_qty * long_px
   284→   284→    short_notional = short_qty * short_px
   285→   285→    skew = (
   286→   286→        abs(long_notional - short_notional) / max(long_notional, short_notional) * 100.0
   287→   287→    )
   288→   288→
   289→   289→    if skew > max_skew_pct:
   290→   290→        qty_part = (
   291→   291→            f" qty long={long_qty} short={short_qty}"
   292→   292→            if abs(long_qty - short_qty) > 1e-12
   293→   293→            else ""
   294→   294→        )
   295→   295→        return (
   296→   296→            True,
   297→   297→            f"skew={skew:.2f}%>{max_skew_pct}%{qty_part}",
   298→   298→            long_notional,
   299→   299→            short_notional,
   300→   300→        )
   301→   301→
   302→   302→    return False, "", long_notional, short_notional

   305→   305→def check_leg_alive(
   306→   306→    pos: dict[str, Any],
   307→   307→    venue_positions: dict[str, list[dict[str, Any]]],
   308→   308→) -> tuple[bool, str]:
   309→   309→    """Check if both legs are still alive (not liquidated/unexpectedly closed).
   310→   310→
   311→   311→    venue_positions: {venue_id: [{symbol, side, qty, ...}]}.
   312→   312→    Note: caller must ensure both venues' position snapshots were fetched successfully
   313→   313→    (fetch failure != no position; confusing the two triggers wrongful close).
   314→   314→    """
   315→   315→    base = str(pos.get("base", "")).upper()
   316→   316→    qty = float(pos.get("qty", 0))
   317→   317→    long_venue = str(pos.get("long_venue", ""))
   318→   318→    short_venue = str(pos.get("short_venue", ""))
   319→   319→
   320→   320→    long_positions = venue_positions.get(long_venue, [])
   321→   321→    short_positions = venue_positions.get(short_venue, [])
   322→   322→
   323→   323→    # Check long leg
   324→   324→    long_alive = False
   325→   325→    for p in long_positions:
   326→   326→        sym = str(p.get("symbol", "")).upper()
   327→   327→        side = str(p.get("side", "")).lower()
   328→   328→        p_qty = float(p.get("qty", 0) or p.get("amount", 0))
   329→   329→        if sym.startswith(base) and side == "long" and p_qty >= qty * 0.95:
   330→   330→            long_alive = True
   331→   331→            break
   332→   332→
   333→   333→    short_alive = False
   334→   334→    for p in short_positions:
   335→   335→        sym = str(p.get("symbol", "")).upper()
   336→   336→        side = str(p.get("side", "")).lower()
   337→   337→        p_qty = float(p.get("qty", 0) or p.get("amount", 0))
   338→   338→        if sym.startswith(base) and side == "short" and abs(p_qty) >= qty * 0.95:
   339→   339→            short_alive = True
   340→   340→            break
   341→   341→
   342→   342→    if not long_alive and not short_alive:
   343→   343→        return False, "both_legs_gone"
   344→   344→    if not long_alive:
   345→   345→        return False, f"long_leg_gone@{long_venue}"
   346→   346→    if not short_alive:
   347→   347→        return False, f"short_leg_gone@{short_venue}"
   348→   348→
   349→   349→    return True, ""

   352→   352→def check_margin_distance(
   353→   353→    pos: dict[str, Any],
   354→   354→    venue_positions: dict[str, list[dict[str, Any]]],
   355→   355→    alert_distance_pct: float = 20.0,
   356→   356→) -> list[dict[str, Any]]:
   357→   357→    """Check mark-price-to-liquidation distance per leg; return alerts if too close.
   358→   358→
   359→   359→    Cross-venue hedged legs do not share P&L: when price moves unilaterally, one leg accumulates
   360→   360→    floating losses; when distance approaches liquidation, margin must be added or the position
   361→   361→    reduced proactively — waiting for liquidation causes a naked leg incident.
   362→   362→    """
   363→   363→    alerts: list[dict[str, Any]] = []
   364→   364→    base = str(pos.get("base", "")).upper()
   365→   365→    for leg, side in (("long", "long"), ("short", "short")):
   366→   366→        venue_id = str(pos.get(f"{leg}_venue", ""))
   367→   367→        rows = venue_positions.get(venue_id)
   368→   368→        if rows is None:
   369→   369→            continue
   370→   370→        for p in rows:
   371→   371→            sym = str(p.get("symbol", "")).upper()
   372→   372→            if not sym.startswith(base) or str(p.get("side", "")).lower() != side:
   373→   373→                continue
   374→   374→            liq_px = float(p.get("liq_price", 0) or 0)
   375→   375→            if liq_px <= 0:
   376→   376→                break  # Exchange did not return liquidation price (common for cross-margin low-leverage)
   377→   377→            mark = _get_mark_price(venue_id, base)
   378→   378→            if mark <= 0:
   379→   379→                break
   380→   380→            distance_pct = abs(mark - liq_px) / mark * 100.0
   381→   381→            if distance_pct < alert_distance_pct:
   382→   382→                alerts.append(
   383→   383→                    {
   384→   384→                        "leg": leg,
   385→   385→                        "venue": venue_id,
   386→   386→                        "mark_price": mark,
   387→   387→                        "liq_price": liq_px,
   388→   388→                        "distance_pct": round(distance_pct, 2),
   389→   389→                    }
   390→   390→                )
   391→   391→            break
   392→   392→    return alerts

   395→   395→def watch_cycle(
   396→   396→    cfg: dict[str, Any],
   397→   397→    *,
   398→   398→    dry_run: bool = True,
   399→   399→    verbose: bool = False,
   400→   400→    log_path: Path = WATCHER_LOG,
   401→   401→) -> dict[str, Any]:
   402→   402→    """Single watch cycle: check all open positions, decide exit/alert."""
   403→   403→    pfa = cfg.get("pureFuturesArbitrage") or {}
   404→   404→    venues = [
   405→   405→        str(v).lower() for v in pfa.get("venues", ["binance", "bitget", "bybit", "okx"])
   406→   406→    ]
   407→   407→    exit_edge = float(pfa.get("exitThresholdPct", 0.01))
   408→   408→    max_skew_pct = float(pfa.get("rebalanceSkewPct", 1.0))
   409→   409→    check_legs = bool(pfa.get("watcherCheckLegs", True))
   410→   410→    auto_rebalance = bool(pfa.get("autoRebalance", False))
   411→   411→    margin_alert_pct = float(pfa.get("marginAlertDistancePct", 20.0))
   412→   412→    workers = int(pfa.get("workers", 4))
   413→   413→
   414→   414→    cycle_result: dict[str, Any] = {
   415→   415→        "ts": _ts_str(),
   416→   416→        "ts_ms": _now_ms(),
   417→   417→        "actions": [],
   418→   418→        "alerts": [],
   419→   419→        "checked": 0,
   420→   420→    }
   421→   421→    actions: list[dict[str, Any]] = []
   422→   422→    alerts: list[Any] = []
   423→   423→    # Clear cycle-level mark price cache
   424→   424→    _mark_price_cache.clear()
   425→   425→
   426→   426→    open_positions = [
   427→   427→        p for p in load_pure_futures_positions() if p.get("status") == "open"
   428→   428→    ]
   429→   429→    if not open_positions:
   430→   430→        if verbose:
   431→   431→            print(f"[{_ts_str()}] no open positions", file=sys.stderr)
   432→   432→        return cycle_result
   433→   433→
   434→   434→    # Bulk-fetch mark prices for all venues that have open positions (parallel, one call per venue)
   435→   435→    wanted_venues: set[str] = set()
   436→   436→    for pos in open_positions:
   437→   437→        for vid in (pos.get("long_venue"), pos.get("short_venue")):
   438→   438→            if vid:
   439→   439→                wanted_venues.add(str(vid))
   440→   440→    _prefetch_all_mark_prices(wanted_venues)
   441→   441→
   442→   442→    # Fetch current funding rates for all venues
   443→   443→    try:
   444→   444→        scan_rates = fetch_all_fee_rate_rows_by_base(
   445→   445→            venues, workers, cache_ttl_sec=30.0
   446→   446→        )
   447→   447→    except Exception as e:
   448→   448→        alerts.append(f"rate_fetch_error: {e}")
   449→   449→        return cycle_result
   450→   450→
   451→   451→    # Fetch actual positions from venues (for leg/margin check)
   452→   452→    # Only keep successfully fetched venues; failures go to failed set, skip their leg checks
   453→   453→    venue_positions: dict[str, list[dict[str, Any]]] = {}
   454→   454→    failed_venues: set[str] = set()
   455→   455→    if check_legs and not dry_run:
   456→   456→        wanted: set[str] = set()
   457→   457→        for pos in open_positions:
   458→   458→            for vid in (pos.get("long_venue"), pos.get("short_venue")):
   459→   459→                if vid:
   460→   460→                    wanted.add(str(vid))
   461→   461→        if wanted:
   462→   462→            # Parallel fetch — saves V×~200ms vs sequential loop
   463→   463→            def _fetch_one(vid: str) -> tuple[str, list[dict[str, Any]] | None]:
   464→   464→                try:
   465→   465→                    return vid, _fetch_positions_from_venue(vid)
   466→   466→                except Exception as e:
   467→   467→                    return vid, None
   468→   468→
   469→   469→            results = run_io_parallel(
   470→   470→                sorted(wanted),
   471→   471→                _fetch_one,
   472→   472→                max_workers=max(4, len(wanted)),
   473→   473→                swallow_errors=True,
   474→   474→            )
   475→   475→            for vid, rows in results.items():
   476→   476→                if rows is None:
   477→   477→                    failed_venues.add(vid)
   478→   478→                    alerts.append({"alert": "positions_fetch_failed", "venue": vid})
   479→   479→                else:
   480→   480→                    venue_positions[vid] = rows
   481→   481→
   482→   482→    for pos in open_positions:
   483→   483→        pos_id = str(pos.get("id", ""))
   484→   484→        base = str(pos.get("base", ""))
   485→   485→        pos_dry_run = dry_run or bool(pos.get("dry_run", True))
   486→   486→        cycle_result["checked"] += 1
   487→   487→
   488→   488→        # 1. Check exit condition
   489→   489→        should_exit, exit_reason = check_exit(pos, scan_rates, exit_edge)
   490→   490→        if should_exit:
   491→   491→            action = {
   492→   492→                "action": "close",
   493→   493→                "position_id": pos_id,
   494→   494→                "base": base,
   495→   495→                "reason": exit_reason,
   496→   496→                "dry_run": pos_dry_run,
   497→   497→            }
   498→   498→            if verbose:
   499→   499→                print(
   500→   500→                    f"[{_ts_str()}] EXIT {pos_id} {base}: {exit_reason}",
   501→   501→                    file=sys.stderr,
   502→   502→                )
   503→   503→            if not pos_dry_run:
   504→   504→                res = close_pure_futures_pair(pos_id, dry_run=False, config=cfg)
   505→   505→                action["result"] = res.to_dict()
   506→   506→                if not res.ok:
   507→   507→                    action["error"] = f"close failed: state={res.state}"
   508→   508→                    send_notification(
   509→   509→                        "Watcher Close Failed",
   510→   510→                        f"Position {pos_id} {base}: exit triggered ({exit_reason}) but close failed: {res.state}",
   511→   511→                        cfg,
   512→   512→                    )
   513→   513→            else:
   514→   514→                action["note"] = "dry_run position, skipping live close"
   515→   515→            actions.append(action)
   516→   516→            continue
   517→   517→
   518→   518→        # 1b. PnL-based stop loss
   519→   519→        pos_long_v = str(pos.get("long_venue", ""))
   520→   520→        pos_short_v = str(pos.get("short_venue", ""))
   521→   521→        if not pos_dry_run:
   522→   522→            long_px = _get_mark_price(pos_long_v, base)
   523→   523→            short_px = _get_mark_price(pos_short_v, base)
   524→   524→            if long_px > 0 and short_px > 0:
   525→   525→                pnl_info = estimate_spread_pnl(pos, long_px, short_px)
   526→   526→                spread_loss_pct = -pnl_info.get(
   527→   527→                    "spread_pnl_pct", 0
   528→   528→                )  # Positive value = loss
   529→   529→
   530→   530→                # Estimate cumulative funding income
   531→   531→                opened_at = int(pos.get("opened_at", 0) or 0)
   532→   532→                held_hours = (_now_ms() - opened_at) / 3600000.0 if opened_at > 0 else 0
   533→   533→                interval_h = 8.0
   534→   534→                periods = max(0, held_hours / interval_h)
   535→   535→                current_spread = _get_current_spread(
   536→   536→                    base, pos_long_v, pos_short_v, scan_rates
   537→   537→                )
   538→   538→                est_funding_pct = (
   539→   539→                    (current_spread or 0) * periods if current_spread else 0
   540→   540→                )
   541→   541→
   542→   542→                max_loss_mult = float(pfa.get("maxLossVsFundingMult", 3.0))
   543→   543→                if (
   544→   544→                    spread_loss_pct > 0
   545→   545→                    and est_funding_pct > 0
   546→   546→                    and spread_loss_pct > est_funding_pct * max_loss_mult
   547→   547→                ):
   548→   548→                    action = {
   549→   549→                        "action": "close",
   550→   550→                        "position_id": pos_id,
   551→   551→                        "base": base,
   552→   552→                        "reason": f"pnl_stop_loss: spread_loss={spread_loss_pct:.4f}% > {max_loss_mult}x est_funding={est_funding_pct:.4f}%",
   553→   553→                        "pnl_info": pnl_info,
   554→   554→                    }
   555→   555→                    send_notification(
   556→   556→                        "PNL STOP LOSS",
   557→   557→                        f"Position {pos_id} {base}: "
   558→   558→                        f"spread_loss={spread_loss_pct:.4f}% > {max_loss_mult}x est_funding={est_funding_pct:.4f}%",
   559→   559→                        cfg,
   560→   560→                    )
   561→   561→                    res = close_pure_futures_pair(pos_id, dry_run=False, config=cfg)
   562→   562→                    action["result"] = res.to_dict()
   563→   563→                    actions.append(action)
   564→   564→                    continue
   565→   565→
   566→   566→        # 2. Check leg alive (only for live positions, and only when
   567→   567→        #    both venues' position snapshots fetched successfully)
   568→   568→        legs_checkable = (
   569→   569→            check_legs
   570→   570→            and not pos_dry_run
   571→   571→            and pos_long_v in venue_positions
   572→   572→            and pos_short_v in venue_positions
   573→   573→        )
   574→   574→        if legs_checkable:
   575→   575→            alive, leg_reason = check_leg_alive(pos, venue_positions)
   576→   576→            if not alive:
   577→   577→                action: dict[str, Any] = {
   578→   578→                    "action": "emergency_close",
   579→   579→                    "position_id": pos_id,
   580→   580→                    "base": base,
   581→   581→                    "reason": leg_reason,
   582→   582→                }
   583→   583→                if leg_reason == "both_legs_gone":
   584→   584→                    # Placing a close order on a vanished leg opens a new position — must not submit.
   585→   585→                    # Only alert for manual review (may be externally closed or symbol match issue).
   586→   586→                    action["action"] = "manual_review"
   587→   587→                    action["note"] = "both legs gone; no orders placed"
   588→   588→                    send_notification(
   589→   589→                        "BOTH LEGS GONE",
   590→   590→                        f"Position {pos_id} {base}: both legs missing on "
   591→   591→                        f"{pos_long_v}/{pos_short_v}. No orders placed — "
   592→   592→                        f"verify manually and mark position closed.",
   593→   593→                        cfg,
   594→   594→                    )
   595→   595→                else:
   596→   596→                    # Single leg gone: only close the surviving leg (closing a vanished leg = opening new position)
   597→   597→                    alive_leg = "short" if leg_reason.startswith("long_leg") else "long"
   598→   598→                    send_notification(
   599→   599→                        "NAKED LEG DETECTED",
   600→   600→                        f"Position {pos_id} {base}: {leg_reason}. "
   601→   601→                        f"Closing surviving {alive_leg} leg only.",
   602→   602→                        cfg,
   603→   603→                    )
   604→   604→                    res = close_pure_futures_leg(
   605→   605→                        pos_id,
   606→   606→                        alive_leg,
   607→   607→                        config=cfg,
   608→   608→                        close_reason=f"emergency: {leg_reason}",
   609→   609→                    )
   610→   610→                    action["result"] = res.to_dict()
   611→   611→                actions.append(action)
   612→   612→                continue
   613→   613→
   614→   614→            # 2b. Per-leg liquidation distance warning (no auto-action, timely notification to add margin/reduce position)
   615→   615→            margin_alerts = check_margin_distance(
   616→   616→                pos, venue_positions, margin_alert_pct
   617→   617→            )
   618→   618→            for ma in margin_alerts:
   619→   619→                alert = {
   620→   620→                    "position_id": pos_id,
   621→   621→                    "base": base,
   622→   622→                    "alert": "margin_distance_low",
   623→   623→                    **ma,
   624→   624→                }
   625→   625→                alerts.append(alert)
   626→   626→                send_notification(
   627→   627→                    "MARGIN DISTANCE LOW",
   628→   628→                    f"Position {pos_id} {base} {ma['leg']}@{ma['venue']}: "
   629→   629→                    f"mark {ma['mark_price']} vs liq {ma['liq_price']} "
   630→   630→                    f"({ma['distance_pct']}% away, threshold {margin_alert_pct}%). "
   631→   631→                    f"Add margin or reduce position.",
   632→   632→                    cfg,
   633→   633→                )
   634→   634→
   635→   635→        # 3. Check rebalance
   636→   636→        need_rebalance, rebal_reason, long_n, short_n = check_rebalance(
   637→   637→            pos, max_skew_pct, venue_positions or None
   638→   638→        )
   639→   639→        if need_rebalance:
   640→   640→            alert = {
   641→   641→                "position_id": pos_id,
   642→   642→                "base": base,
   643→   643→                "alert": "rebalance_needed",
   644→   644→                "reason": rebal_reason,
   645→   645→                "long_notional": round(long_n, 2),
   646→   646→                "short_notional": round(short_n, 2),
   647→   647→            }
   648→   648→            alerts.append(alert)
   649→   649→            if verbose:
   650→   650→                print(
   651→   651→                    f"[{_ts_str()}] REBALANCE {pos_id} {base}: {rebal_reason} "
   652→   652→                    f"(long=${long_n:.2f} short=${short_n:.2f})",
   653→   653→                    file=sys.stderr,
   654→   654→                )
   655→   655→            if auto_rebalance and not pos_dry_run:
   656→   656→                # Only trim quantity mismatch (real delta from partial liquidation/ADL).
   657→   657→                # Pure mark-price drift (equal quantities) is not tradeable away; executor returns balanced.
   658→   658→                res = rebalance_pure_futures_pair(pos_id, dry_run=False, config=cfg)
   659→   659→                action = {
   660→   660→                    "action": "rebalance",
   661→   661→                    "position_id": pos_id,
   662→   662→                    "base": base,
   663→   663→                    "reason": rebal_reason,
   664→   664→                    "result": res.to_dict(),
   665→   665→                }
   666→   666→                if res.state == "balanced":
   667→   667→                    action["note"] = "qty equal; skew is mark-price drift only"
   668→   668→                elif not res.ok:
   669→   669→                    action["error"] = f"rebalance failed: state={res.state}"
   670→   670→                elif verbose:
   671→   671→                    print(
   672→   672→                        f"[{_ts_str()}] REBALANCED {pos_id} {base}: {res.logs[-1] if res.logs else ''}",
   673→   673→                        file=sys.stderr,
   674→   674→                    )
   675→   675→                actions.append(action)
   676→   676→
   677→   677→    cycle_result["actions"] = actions
   678→   678→    cycle_result["alerts"] = alerts
   679→   679→    _append_log(log_path, cycle_result)
   680→   680→    return cycle_result

   683→   683→def main() -> int:
   684→   684→    parser = argparse.ArgumentParser(

   705→   705→    parser.add_argument("--json", action="store_true", help="JSON output each cycle")
   706→   706→    args = parser.parse_args()
   707→   707→
   708→   708→    cfg_path = Path(args.config)
   709→   709→    cfg = apply_strategy_to_pure_futures_cfg(
   710→   710→        json.loads(cfg_path.read_text(encoding="utf-8"))

   712→   712→
   713→   713→    # Determine dry-run: explicit flag > env > config
   714→   714→    if args.dry_run:
   715→   715→        dry_run = True
   716→   716→    elif os.environ.get("FARB_LIVE") == "1" or os.environ.get("DCA_LIVE") == "1":
   717→   717→        dry_run = False
   718→   718→    elif os.environ.get("FARB_DRY_RUN") == "1" or os.environ.get("DCA_DRY_RUN") == "1":
   719→   719→        dry_run = True
   720→   720→    else:
   721→   721→        dry_run = bool(cfg.get("dry_run", True))
   722→   722→
   723→   723→    if args.once:
   724→   724→        result = watch_cycle(cfg, dry_run=dry_run, verbose=args.verbose)
   725→   725→        if args.json:
   726→   726→            print(json.dumps(result, ensure_ascii=False, indent=2))
   727→   727→        return 0
   728→   728→
   729→   729→    print(
   730→   730→        f"[{_ts_str()}] Pure Futures Watcher started "
   731→   731→        f"(interval={args.interval}s, dry_run={dry_run})",
   732→   732→        file=sys.stderr,
   733→   733→    )
   734→   734→
   735→   735→    while True:
   736→   736→        try:
   737→   737→            t0 = time.time()
   738→   738→            result = watch_cycle(cfg, dry_run=dry_run, verbose=args.verbose)
   739→   739→            elapsed = time.time() - t0
   740→   740→
   741→   741→            if args.json:
   742→   742→                print(json.dumps(result, ensure_ascii=False))
   743→   743→            elif args.verbose:
   744→   744→                print(
   745→   745→                    f"[{_ts_str()}] checked={result['checked']} "
   746→   746→                    f"actions={len(result['actions'])} "
   747→   747→                    f"alerts={len(result['alerts'])} "
   748→   748→                    f"in {elapsed:.1f}s",
   749→   749→                    file=sys.stderr,
   750→   750→                )
   751→   751→
   752→   752→            sleep_time = max(0.0, args.interval - elapsed)
   753→   753→            time.sleep(sleep_time)
   754→   754→
   755→   755→        except KeyboardInterrupt:
   756→   756→            print(f"\n[{_ts_str()}] Watcher stopped.", file=sys.stderr)
   757→   757→            break
   758→   758→        except Exception as e:
   759→   759→            print(f"[{_ts_str()}] Watcher error: {e}", file=sys.stderr)
   760→   760→            time.sleep(60)
   761→   761→
   762→   762→    return 0
